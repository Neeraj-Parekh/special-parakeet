   463→# --- Circuit Breaker ---
   464→class CircuitBreaker:
   465→    """Fail open if ML is down."""
   466→    def __init__(self, failure_threshold=5, recovery_timeout=30):

   468→        self.state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN

   473→    def call(self, func, *args, **kwargs):
   474→        if self.state == "OPEN":
   475→            if (datetime.utcnow() - self.last_failure).seconds > self.timeout:
   476→                self.state = "HALF_OPEN"
   477→            else:
   478→                return None  # Fail open
   479→        
   480→        try:
   481→            result = func(*args, **kwargs)
   482→            if self.state == "HALF_OPEN":
   483→                self.state = "CLOSED"
   484→                self.failures = 0
   485→            return result
   486→        except Exception as e:
   487→            self.failures += 1
   488→            self.last_failure = datetime.utcnow()
   489→            if self.failures >= self.threshold:
   490→                self.state = "OPEN"
   491→                logger.error("circuit_breaker_opened")
   492→            raise

   574→async def score_order(
   575→    request: RiskRequest,
   576→    background_tasks: BackgroundTasks,
   577→    merchant: dict = Depends(verify_api_key),
   578→    db: Session = Depends(get_db),
   579→    rate: RateLimiter = Depends(RateLimiter(times=100, seconds=60))
   580→):
   581→    start = datetime.utcnow()
   582→    
   583→    # 1. Rules Engine First (deterministic, fast)
   584→    order_dict = request.dict()
   585→    rule_decision = rules_engine.evaluate(order_dict, db)
   586→    if rule_decision == "BLOCK":
   587→        return RiskResponse(
   588→            prediction_id=f"pred_{start.timestamp()}",
   589→            risk_score=100.0,
   590→            probability=1.0,
   591→            decision="REJECT",
   592→            explanation=[{"feature": "rule_engine", "reason": "Hard rule matched"}],
   593→            rule_fired="BLOCK",
   594→            model_version="rules_only",
   595→            latency_ms=5.0,
   596→            timestamp=start
   597→        )
   598→    
   599→    # 2. Feature Store Lookup
   600→    online_features = FeatureStore.get_online_features(
   601→        request.customer_id,
   602→        ["prior_returns", "prior_orders", "avg_order_value"]
   603→    )
   604→    
   605→    # 3. Feature Engineering (simplified)
   606→    features = np.array([[
   607→        request.amount,
   608→        1 if request.is_cod else 0,
   609→        online_features.get("prior_returns", 0),
   610→        online_features.get("prior_orders", 0),
   611→        len(request.delivery_address),
   612→        1 if request.is_repeat_customer else 0
   613→    ]])
   614→    
   615→    # 4. ML Scoring with Circuit Breaker
   616→    try:
   617→        prob, version = circuit_breaker.call(model_service.predict, features)
   618→    except Exception as e:
   619→        # Fallback to rules-only if ML down
   620→        logger.error("ml_fallback", error=str(e))
   621→        prob = 0.5
   622→        version = "fallback"
   623→    
   624→    risk_score = prob * 100
   625→    
   626→    # 5. Decision Gates (Dynamic Thresholds)
   627→    if risk_score < 15:
   628→        decision = "ACCEPT"
   629→    elif risk_score < 60:
   630→        decision = "REVIEW"
   631→    else:
   632→        decision = "REJECT"
   633→    
   634→    # 6. SHAP-like Explanation (simplified)
   635→    explanation = [
   636→        {"feature": "order_amount", "value": request.amount, "contribution": 0.3},
   637→        {"feature": "is_cod", "value": request.is_cod, "contribution": 0.25},
   638→        {"feature": "prior_returns", "value": online_features.get("prior_returns", 0), "contribution": 0.2}
   639→    ]
   640→    
   641→    # 7. Build Response
   642→    prediction_id = f"pred_{request.order_id}_{int(start.timestamp())}"
   643→    latency = (datetime.utcnow() - start).total_seconds() * 1000
   644→    
   645→    response = RiskResponse(
   646→        prediction_id=prediction_id,
   647→        risk_score=risk_score,
   648→        probability=prob,
   649→        decision=decision,
   650→        explanation=explanation,
   651→        rule_fired=rule_decision,
   652→        model_version=version,
   653→        latency_ms=round(latency, 2),
   654→        timestamp=start
   655→    )
   656→    
   657→    # 8. Async Audit Write
   658→    audit_record = {
   659→        "prediction_id": prediction_id,
   660→        "order_id": request.order_id,
   661→        "merchant_id": merchant["merchant_id"],
   662→        "risk_score": risk_score,
   663→        "probability": prob,
   664→        "decision": decision,
   665→        "features": features.tolist()[0],
   666→        "shap_values": explanation,
   667→        "top_features": explanation[:3],
   668→        "model_version": version
   669→    }
   670→    background_tasks.add_task(write_audit, audit_record, db)
   671→    
   672→    logger.info("order_scored", 
   673→                order_id=request.order_id,
   674→                merchant=merchant["merchant_id"],
   675→                decision=decision,
   676→                latency_ms=latency)

   678→    return response
