   464→   464→class CircuitBreaker:
   465→   465→    """Fail open if ML is down."""
   466→   466→    def __init__(self, failure_threshold=5, recovery_timeout=30):
   467→   467→        self.failures = 0
   468→   468→        self.state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN
   469→   469→        self.threshold = failure_threshold
   470→   470→        self.timeout = recovery_timeout
   471→   471→        self.last_failure = None
   472→   472→    
   473→   473→    def call(self, func, *args, **kwargs):
   474→   474→        if self.state == "OPEN":
   475→   475→            if (datetime.utcnow() - self.last_failure).seconds > self.timeout:
   476→   476→                self.state = "HALF_OPEN"
   477→   477→            else:
   478→   478→                return None  # Fail open
   479→   479→        
   480→   480→        try:
   481→   481→            result = func(*args, **kwargs)
   482→   482→            if self.state == "HALF_OPEN":
   483→   483→                self.state = "CLOSED"
   484→   484→                self.failures = 0
   485→   485→            return result
   486→   486→        except Exception as e:
   487→   487→            self.failures += 1
   488→   488→            self.last_failure = datetime.utcnow()
   489→   489→            if self.failures >= self.threshold:
   490→   490→                self.state = "OPEN"
   491→   491→                logger.error("circuit_breaker_opened")
   492→   492→            raise
