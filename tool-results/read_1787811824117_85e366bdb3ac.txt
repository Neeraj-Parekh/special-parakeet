    94→def create_app(
    95→    scorer_rate_per_min: int = 120, audit_path: str = "out/audit.jsonl"
    96→) -> FastAPI:
    97→    state: dict[str, Any] = {}
    98→
    99→    @asynccontextmanager
   100→    async def lifespan(_: FastAPI):
   101→        df = add_address_features(load_orders("data/raw/cod_orders.csv"))
   102→        train_df, _ = group_split(df)
   103→        X_tr, y_tr = build_feature_frame(train_df, "order+addr")
   104→        model_path = Path("out/model_api.joblib")
   105→        if not model_path.exists():
   106→            save_model(fit_model(X_tr, y_tr), str(model_path))
   107→        from src.models.train import load_model
   108→
   109→        state["model"] = load_model(str(model_path))
   110→        state["reference"] = X_tr.mode().iloc[0]
   111→        state["base_rate"] = float(y_tr.mean())
   112→        state["audit"] = AuditLogger(audit_path)
   113→        state["keys"] = default_keys()
   114→        state["bucket"] = TokenBucket(scorer_rate_per_min)
   115→        state["rules"] = RulesEngine()
   116→        state["breaker"] = CircuitBreaker()
   117→        state["metrics"] = Metrics()
   118→        state["cases"] = CaseService("out/cases.jsonl")
   119→        num_cols = [c for c in X_tr.columns if str(X_tr[c].dtype) != "category"]
   120→        state["psi_sample"] = {
   121→            c: X_tr[c].dropna().sample(n=min(2000, len(X_tr)), random_state=7).tolist()
   122→            for c in num_cols
   123→        }
   124→        state["idem"] = {}
   125→        # Pre-compute labeled (y, p) sequences for the /v1/policy/cost-curves
   126→        # endpoint (Drummond & Holte 2006). The training set carries ground-
   127→        # truth is_returned labels; running the model over it once at startup
   128→        # lets the endpoint respond in O(1) for the per-threshold sweep and
   129→        # ~3-5 sec for the 500-resample bootstrap (per-row performance only).
   130→        # Note: this is in-sample — Day 2 Track E + G will swap in a held-out
   131→        # test slice + delayed-label feedback so the curve reflects deployment
   132→        # conditions, not training fit.
   133→        try:
   134→            _cost_curve_p = state["model"].predict_proba(X_tr)[:, 1].tolist()
   135→            _cost_curve_y = [int(v) for v in y_tr.tolist()]
   136→            state["cost_curve"] = {
   137→                "y_true": _cost_curve_y,
   138→                "probs": _cost_curve_p,
   139→                "n": len(_cost_curve_y),
   140→                "source": "train_df_in_sample",
   141→            }
   142→        except Exception as e:  # pragma: no cover — startup-only, defensive
   143→            print(f"cost-curve warmup skipped: {type(e).__name__}: {e}", file=sys.stderr)
   144→            state["cost_curve"] = None
   145→        yield
   146→
   147→    app = FastAPI(title="RTO Trust Layer", version="0.2.0", lifespan=lifespan)
   148→    app.state.core = state

   155→            "OrderDay": 180,
   156→            "OrderHour": o.order_hour,
   157→            "PriorOrders": o.prior_orders,
   158→            "PriorReturns": o.prior_returns,
   159→            "is_cod": int(o.payment_method.upper() == "COD"),
   160→            "category": o.category,
   161→            "device": o.device,
   162→            "city_tier": o.city_tier,
   163→            "address_quality": o.address_quality,
   164→        }
   165→        return pd.DataFrame([row])

   167→    @app.post("/risk/score")
   168→    def score(
   169→        order: OrderIn,
   170→        authorization: str | None = Header(default=None),
   171→        idempotency_key: str | None = Header(default=None),
   172→        x_mandate: str | None = Header(default=None),
   173→        # Day 1 Track D (V3 §13): per-txn device_id + user_id for UPI Circle
   174→        # mandates (NPCI OC-201B §3.7 Issuer Bank duty + §3.3 Secondary PSP
   175→        # duty). Both default to None and are ignored for cod_order mandates.
   176→        x_device_id: str | None = Header(default=None),
   177→        x_user_id: str | None = Header(default=None),
   178→    ) -> dict:
   179→        token = bearer_token(authorization)
   180→        ok, err = check_key(token, "scorer", state["keys"])
   181→        if not ok:
   182→            raise HTTPException(status_code=401, detail=err)
   183→        client = token
   184→        if not state["bucket"].allow(client):
   185→            raise HTTPException(status_code=429, detail="rate limit exceeded")
   186→
   187→        cache_key = (idempotency_key or "", order.model_dump_json())
   188→        if idempotency_key and cache_key in state["idem"]:
   189→            return dict(state["idem"][cache_key], replayed=True)
   190→
   191→        try:
   192→            t0 = time.monotonic()
   193→            # Mandate verification (cheap; always run; enforcement happens
   194→            # later). Day 1 Track D extended the signature to pass device_id
   195→            # + user_id so UPI Circle mandates can enforce OC-201B §3.3/§3.7
   196→            # per-txn identity validation. cod_order mandates ignore both.
   197→            mandate_verdict, mandate_payload = verify_mandate(
   198→                x_mandate,
   199→                order.amount_inr,
   200→                device_id=x_device_id,
   201→                user_id=x_user_id,
   202→            )
   203→            fired = state["rules"].evaluate(order.model_dump())
   204→            rule_fired = fired.rule_id if fired else None
   205→
   206→            proba: float | None = None
   207→            reasons: list[dict] = []
   208→            degraded = False
   209→            breach_note: str | None = None
   210→            cost_breakdown: dict | None = None
   211→            decision: str | None = None
   212→            # ``decision_source`` records which layer actually chose the
   213→            # decision, for the audit trail + dashboard explainability.
   214→            # One of: rules_engine_block | mandate_breach | mandate_invalid
   215→            # | mandate_review_required | degraded_review | cost_optimal_bmr
   216→            # | cost_optimal_bmr_review_rule
   217→            decision_source: str
   218→
   219→            # ------------------------------------------------------------------
   220→            # Decision precedence (per Day 1 Track C spec):
   221→            #   1. Rules fast-path (BLOCK)         → short-circuit REJECT
   222→            #   2. Mandate enforcement (BREACH /
   223→            #      TAMPERED / EXPIRED w/ header)   → short-circuit REJECT
   224→            #   2c. Mandate REVIEW (UPI Circle 24h
   225→            #      cooling period, OC-201B)        → short-circuit REVIEW
   226→            #      (Day 1 Track D — mandate action-class expansion)
   227→            #   3. Circuit breaker OPEN            → degraded rules-only REVIEW
   228→            #   4. optimal_decision(p, ...)         → PRIMARY decision path
   229→            #      (replaces legacy static 0.15 / 0.60 thresholds)
   230→            # REVIEW rules don't short-circuit; they gate step 4 to never ACCEPT.
   231→            # ------------------------------------------------------------------
   232→            if fired is not None and fired.action == "BLOCK":
   233→                # 1. Rules-engine BLOCK short-circuits — no model invocation.
   234→                decision = "REJECT"
   235→                decision_source = "rules_engine_block"
   236→            elif mandate_verdict == MandateVerdict.BREACH:
   237→                # 2a. Mandate amount-breach short-circuits.
   238→                decision = "REJECT"
   239→                breach_note = "mandate_amount_breach"
   240→                decision_source = "mandate_breach"
   241→            elif x_mandate is not None and mandate_verdict == MandateVerdict.REVIEW:
   242→                # 2c. UPI Circle 24h cooling-period gate (OC-201B). The txn
   243→                # is permitted in principle; the cooling circuit requires a
   244→                # human to approve before the issuer debits. Routes to the
   245→                # case queue like other REVIEW decisions. verdict_reason
   246→                # (e.g. "cooling_period_active") flows into the audit trail
   247→                # for compliance traceability.
   248→                decision = "REVIEW"
   249→                breach_note = "mandate_review_required"
   250→                decision_source = "mandate_review_required"
   251→            elif x_mandate is not None and mandate_verdict != MandateVerdict.VALID:
   252→                # 2b. Tampered / expired mandate with a header present.
   253→                # Covers OC-201B 6-month inactivity auto-revoke (verdict_reason
   254→                # = "inactivity_auto_revoke") and standard TTL expiry.
   255→                decision = "REJECT"
   256→                breach_note = f"mandate_{mandate_verdict}"
   257→                decision_source = "mandate_invalid"
   258→            else:
   259→                # 3. Circuit breaker guards model invocation.
   260→                use_model = state["breaker"].allow_attempt()
   261→                if use_model:
   262→                    try:
   263→                        X, _ = build_feature_frame(to_frame(order), "order+addr")
   264→                        reasons = reason_codes_batch(
   265→                            state["model"],
   266→                            X,
   267→                            list(X.columns),
   268→                            state["base_rate"],
   269→                            state["reference"],
   270→                        )
   271→                        proba = float(state["model"].predict_proba(X)[0, 1])
   272→                        state["breaker"].record_success()
   273→                    except Exception:
   274→                        state["breaker"].record_failure()
   275→                        proba = None
   276→                if proba is None:
   277→                    # 3. Degraded path: model unavailable → rules-only REVIEW.
   278→                    degraded = True
   279→                    decision = "REVIEW"
   280→                    decision_source = "degraded_review"
   281→                else:
   282→                    # 4. Bahnsen Bayes Minimum Risk decision layer (ICMLA 2013,
   283→                    #    DOI 10.1109/ICMLA.2013.68). Per-order argmin of expected
   284→                    #    cost over {ACCEPT, REVIEW (selective-OTP), REJECT}.
   285→                    decision, costs = optimal_decision(proba, **DEFAULT_COST_WEIGHTS)
   286→                    cost_breakdown = costs
   287→                    # REVIEW rule gate: never ACCEPT when a REVIEW rule fired
   288→                    # (still uses BMR cost math, but the rule forces REVIEW).
   289→                    if fired is not None and fired.action == "REVIEW" and decision == "ACCEPT":
   290→                        decision = "REVIEW"
   291→                        decision_source = "cost_optimal_bmr_review_rule"
   292→                    else:
   293→                        decision_source = "cost_optimal_bmr"
   294→
   295→            # Legacy ``policy_hint`` is the BMR-optimal action string (kept for
   296→            # dashboard / consumer backward compat; the actual decision is the
   297→            # body's ``decision`` field — which equals ``policy_hint`` whenever
   298→            # the cost-optimizer was the decision source).
   299→            policy_hint = None
   300→            if proba is not None:
   301→                policy_hint = optimal_decision(proba, **DEFAULT_COST_WEIGHTS)[0]
   302→
   303→            features_used = (
   304→                {c: float(X.iloc[0][c]) for c in X.columns if str(X[c].dtype) != "category"}
   305→                if proba is not None
   306→                else {}
   307→            )
   308→            case_id = None
   309→            if decision == "REVIEW":
   310→                case_id = state["cases"].open_case(
   311→                    prediction_id="pending", order_id=order.order_id,
   312→                    reason=(
   313→                        f"mandate:{mandate_payload.get('verdict_reason')}"
   314→                        if decision_source == "mandate_review_required"
   315→                        else "review_gate" if fired is None
   316→                        else f"rule:{fired.rule_id}"
   317→                    ),
   318→                )
   319→            # Audit payload (Day 1 Track D — V3 §13 mandate action-class
   320→            # expansion). The new fields surface OC-201B compliance metadata
   321→            # in the tamper-evident hash chain: ``mandate_type``,
   322→            # ``bh_purpose_code``, ``device_id``, ``user_id``, and the
   323→            # machine-readable ``mandate_verdict_reason``. The audit logger
   324→            # (Track E Day 2) stores arbitrary JSON payloads; we just pass
   325→            # the new keys through — no logger.py modification needed.
   326→            audit_id = state["audit"].log(
   327→                {
   328→                    "request": {
   329→                        **order.model_dump(),
   330→                        "customer_id": redact_customer(order.customer_id),
   331→                    },
   332→                    "probability": round(proba, 5) if proba is not None else None,
   333→                    "decision": decision,
   334→                    "decision_source": decision_source,
   335→                    "cost_breakdown": cost_breakdown,
   336→                    "reason_codes": reasons[:5],
   337→                    "mandate_verdict": mandate_verdict,
   338→                    "mandate_verdict_reason": mandate_payload.get("verdict_reason"),
   339→                    "mandate_type": mandate_payload.get("mandate_type"),
   340→                    "bh_purpose_code": mandate_payload.get("bh_purpose_code"),
   341→                    "device_id": x_device_id,
   342→                    "user_id": x_user_id,
   343→                    "breach_note": breach_note,
   344→                    "rule_fired": rule_fired,
   345→                    "degraded": degraded,
   346→                    "features_used": features_used,
   347→                    "latency_ms": now_ms(t0),
   348→                    "case_id": case_id,
   349→                }
   350→            )
   351→            body = {
   352→                "prediction_id": str(uuid.uuid4()),
   353→                "risk_score": round(proba * 100, 1) if proba is not None else None,
   354→                "probability": round(proba, 4) if proba is not None else None,
   355→                "decision": decision,
   356→                # ``gate_thresholds`` now reports the policy in force. Legacy
   357→                # accept_below / reject_above kept for backward compat — but
   358→                # the live decision path no longer consults them; it consults
   359→                # ``optimal_decision(p, weights=...)``. The dashboard reads
   360→                # ``policy`` to label the decision card.
   361→                "gate_thresholds": {
   362→                    "policy": "cost_optimal_bmr",
   363→                    "weights": DEFAULT_COST_WEIGHTS,
   364→                    "legacy_accept_t": ACCEPT_T,
   365→                    "legacy_reject_t": REJECT_T,
   366→                },
   367→                "decision_source": decision_source,
   368→                "cost_breakdown": cost_breakdown,
   369→                "explanation": reasons[:5],
   370→                "rule_fired": rule_fired,
   371→                "degraded": degraded,
   372→                "policy_hint": policy_hint,
   373→                "model_version": "rules_only"
   374→                if degraded
   375→                else (current_champion() or {"version": state["audit"].model_version})["version"],
   376→                "latency_ms": now_ms(t0),
   377→                "case_id": case_id,
   378→                "mandate": {
   379→                    "verdict": mandate_verdict,
   380→                    "note": breach_note,
   381→                    # Day 1 Track D — V3 §13: surface OC-201B compliance metadata
   382→                    # to the agent / dashboard so the consumer can see WHY
   383→                    # the mandate verdict landed (e.g. "cooling_period_active"
   384→                    # vs "device_id_not_allowed") and WHICH mandate type fired.
   385→                    "verdict_reason": mandate_payload.get("verdict_reason"),
   386→                    "mandate_type": mandate_payload.get("mandate_type"),
   387→                    "bh_purpose_code": mandate_payload.get("bh_purpose_code"),
   388→                },
   389→                "audit_trail_url": f"/audit/{audit_id}",
   390→                "timestamp": datetime.now(timezone.utc).isoformat(),
   391→            }
   392→            state["metrics"].inc(
   393→                "risk_decisions_total", {"decision": decision, "degraded": str(degraded)}
   394→            )
   395→            state["metrics"].observe_latency(now_ms(t0) / 1000)
   396→            if idempotency_key:
   397→                state["idem"][cache_key] = body
   398→            return body
   399→        except HTTPException:
   400→            raise
   401→        except Exception as e:  # no internal detail leakage
   402→            incident = uuid.uuid4()
   403→            print(f"incident={incident} scoring_failed={type(e).__name__}: {e}", file=sys.stderr)
   404→            raise HTTPException(status_code=500, detail=f"internal_error incident={incident}")

   447→    @app.get("/v1/models/drift")
   448→    def models_drift(authorization: str | None = Header(default=None)) -> dict:
   449→        ok, err = check_key(bearer_token(authorization), "admin", state["keys"])
   450→        if not ok:
   451→            raise HTTPException(status_code=401, detail=err)
   452→        recent = [
   453→            r.get("features_used", {})
   454→            for r in _read_audit_tail(state["audit"], limit=300)
   455→            if r.get("features_used")
   456→        ]
   457→        if len(recent) < 30:
   458→            return {"status": "insufficient_data", "observed": len(recent), "psi": {}}
   459→        report = {}
   460→        for col, ref in state["psi_sample"].items():
   461→            obs = [r[col] for r in recent if col in r]
   462→            if len(obs) >= 10:
   463→                report[col] = round(psi(ref, obs), 4)
   464→        worst = max(report.values(), default=0.0)

   583→    @app.get("/v1/policy/cost-curves")
   584→    def policy_cost_curves(
   585→        n_resamples: int = DEFAULT_COST_CURVE_RESAMPLES,
   586→        confidence: float = DEFAULT_COST_CURVE_CONFIDENCE,
   587→        authorization: str | None = Header(default=None),
   588→    ) -> dict:
   589→        """Drummond-Holte cost-curve explorer (Machine Learning 65:95-130, 2006,
   590→        DOI 10.1007/s10994-006-8199-5).
   591→
   592→        Returns a threshold sweep (0.05 → 0.95) over the labeled dataset that
   593→        the in-process model was trained on. For each threshold we report the
   594→        confusion counts, total Bahnsen cost (Eq. 1), precision, recall, plus
   595→        a 90% bootstrap CI on cost (≥500 resamples preserving row marginals —
   596→        per Drummond-Holte ``bootstrap_performance_ci`` capability).
   597→
   598→        The dashboard cost-curve explorer consumes this endpoint to render
   599→        real bars (replacing the previously-hardcoded ``COSTS`` array).
   600→
   601→        Auth: scorer scope. Returns 503 if the model isn't loaded or no
   602→        labeled data is available — never crashes the dashboard.
   603→        """
   604→        ok, err = check_key(bearer_token(authorization), "scorer", state["keys"])
   605→        if not ok:
   606→            raise HTTPException(status_code=401, detail=err)
   607→
   608→        # 1. Guard: model must be loaded (circuit breaker may be OPEN).
   609→        if state.get("model") is None:
   610→            raise HTTPException(
   611→                status_code=503,
   612→                detail="cost curves unavailable — model not loaded",
   613→            )
   614→        # 2. Guard: pre-computed labeled data must exist (startup warmup may
   615→        # have failed silently; surface a clear error to the dashboard).
   616→        cc = state.get("cost_curve")
   617→        if not cc or not cc.get("y_true") or not cc.get("probs"):
   618→            raise HTTPException(
   619→                status_code=503,
   620→                detail="cost curves unavailable — no labeled data available",
   621→            )
   622→        if n_resamples < 1 or n_resamples > 5000:
   623→            raise HTTPException(
   624→                status_code=422,
   625→                detail="n_resamples must be in [1, 5000]",
   626→            )
   627→        if not (0.5 < confidence < 0.999):
   628→            raise HTTPException(
   629→                status_code=422,
   630→                detail="confidence must be in (0.5, 0.999)",
   631→            )
   632→
   633→        y_true = cc["y_true"]
   634→        probs = cc["probs"]
   635→        weights = DEFAULT_COST_WEIGHTS
   636→        c_fp = float(weights["c_fp"])
   637→        c_fn = float(weights["c_fn"])
   638→
   639→        # 3. Threshold sweep → per-threshold confusion + cost + precision +
   640→        #    recall (pure-Python, O(N*T); ~140k ops for 7235 rows × 19 thr).
   641→        curves = cost_curve_sweep(
   642→            y_true=y_true,
   643→            probs=probs,
   644→            c_fp=c_fp,
   645→            c_fn=c_fn,
   646→        )
   647→
   648→        # 4. Bootstrap CI per threshold (Drummond-Holte row-marginal-preserving
   649→        #    resampling; ~3-5 sec for 500 resamples × 19 thr × 7235 samples).
   650→        try:
   651→            bootstrap_ci = bootstrap_cost_ci(
   652→                y_true=y_true,
   653→                probs=probs,
   654→                c_fp=c_fp,
   655→                c_fn=c_fn,
   656→                n_resamples=n_resamples,
   657→                confidence=confidence,
   658→                seed=42,
   659→            )
   660→        except Exception as e:  # pragma: no cover — defensive
   661→            # If bootstrap fails for any reason, return degenerate CIs and a
   662→            # warning rather than 500-ing the dashboard.
   663→            bootstrap_ci = {
   664→                str(r["threshold"]): {"low": 0.0, "high": 0.0, "mean": 0.0}
   665→                for r in curves
   666→            }
   667→            print(
   668→                f"cost-curves bootstrap degraded: {type(e).__name__}: {e}",
   669→                file=sys.stderr,
   670→            )
   671→
   672→        # 5. Optimal threshold = the one minimizing total cost (this is the
   673→        #    threshold analog of what ``optimal_decision()`` does per order).
   674→        optimal_record = min(curves, key=lambda r: r["cost"])
   675→        optimal_threshold = float(optimal_record["threshold"])
   676→
   677→        # 6. Cost crossover: if a challenger model is registered in the model
   678→        #    registry, find the threshold where it beats the incumbent. For
   679→        #    now there is one in-process model so this is a degenerate case —
   680→        #    the incumbent *is* the only model. We still surface the field so
   681→        #    the dashboard wiring is forward-compatible with Day 2 Track E.
   682→        cost_crossover: dict | None = None
   683→        try:
   684→            challenger = None
   685→            champ = current_champion()
   686→            # When more than one model is registered, treat the latest
   687→            # non-champion as the challenger. (For now: single model, None.)
   688→            if champ is not None:
   689→                from src.ml.registry import load_registry
   690→                reg = load_registry()
   691→                non_champ = [m for m in reg.get("models", []) if not m.get("is_champion")]
   692→                challenger = non_champ[-1] if non_champ else None
   693→            if challenger is not None and challenger.get("version") != (champ or {}).get("version"):
   694→                # We don't have challenger probabilities pre-computed yet
   695→                # (single-model registry). Surface a placeholder that the
   696→                # dashboard renders as "single-model — no crossover yet".
   697→                cost_crossover = {
   698→                    "status": "single_model",
   699→                    "incumbent_version": (champ or {}).get("version"),
   700→                    "challenger_version": None,
   701→                    "crossover_threshold": None,
   702→                    "note": (
   703→                        "challenger predictions not pre-computed; "
   704→                        "wire Day 2 Track E model-registry to enable"
   705→                    ),
   706→                }
   707→            else:
   708→                cost_crossover = {
   709→                    "status": "single_model",
   710→                    "incumbent_version": (champ or {}).get("version"),
   711→                    "challenger_version": None,
   712→                    "crossover_threshold": None,
   713→                    "note": "only one model registered; no crossover available",
   714→                }
   715→        except Exception:  # pragma: no cover — defensive
   716→            cost_crossover = {"status": "error", "crossover_threshold": None}
   717→
   718→        return {
   719→            "thresholds": [r["threshold"] for r in curves],
   720→            "curves": curves,
   721→            "bootstrap_ci": bootstrap_ci,
   722→            "optimal_threshold": optimal_threshold,
   723→            "cost_crossover": cost_crossover,
   724→            "cost_model": {
   725→                "c_fp": c_fp,
   726→                "c_fn": c_fn,
   727→                "c_otp": float(weights["c_otp"]),
   728→                "c_block": float(weights["c_block"]),
   729→                "otp_effectiveness": float(weights["otp_effectiveness"]),
   730→                "source_paper": "Bahnsen ICMLA 2013, DOI 10.1109/ICMLA.2013.68",
   731→                "curve_paper": "Drummond & Holte 2006, DOI 10.1007/s10994-006-8199-5",
   732→            },
   733→            "data_source": cc.get("source", "unknown"),
   734→            "n_samples": int(cc.get("n", 0)),
   735→            "n_pos": int(sum(y_true)),
   736→            "n_neg": int(len(y_true) - sum(y_true)),
   737→        }
