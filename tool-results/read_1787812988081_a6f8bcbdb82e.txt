   106→def create_app(
   107→    scorer_rate_per_min: int = 120, audit_path: str = "out/audit.jsonl"
   108→) -> FastAPI:
   109→    state: dict[str, Any] = {}
   110→
   111→    # Day 2 Track E — load Settings once at app-construction time. The
   112→    # ``database_url`` field selects between Postgres mode (full DB-backed
   113→    # audit/cases/registry/idempotency) and file-mode fallback (the 63
   114→    # existing tests run this way). The Settings object itself is cached
   115→    # via ``@lru_cache`` in ``src.config`` so we don't re-read env vars on
   116→    # every TestClient build.
   117→    settings = get_settings()
   118→    state["settings"] = settings
   119→
   120→    # Idempotency cache. File mode: TTLCache (bounded + auto-expiring —
   121→    # closes §A item 2 memory leak). Postgres mode: the ``idempotency_keys``
   122→    # table; ``state["idem"]`` is unused (kept as an empty dict for backward-
   123→    # compat with any test that asserts the key exists).
   124→    if settings.is_postgres:
   125→        state["idem"] = TTLCache(maxsize=1, ttl=3600)  # placeholder, real cache is the DB
   126→    else:
   127→        state["idem"] = TTLCache(
   128→            maxsize=settings.idem_maxsize, ttl=settings.idem_ttl_seconds
   129→        )
   130→
   131→    @asynccontextmanager
   132→    async def lifespan(_: FastAPI):
   133→        df = add_address_features(load_orders("data/raw/cod_orders.csv"))
   134→        train_df, _ = group_split(df)
   135→        X_tr, y_tr = build_feature_frame(train_df, "order+addr")
   136→        model_path = Path("out/model_api.joblib")
   137→        if not model_path.exists():
   138→            save_model(fit_model(X_tr, y_tr), str(model_path))
   139→        from src.models.train import load_model
   140→
   141→        state["model"] = load_model(str(model_path))
   142→        state["reference"] = X_tr.mode().iloc[0]
   143→        state["base_rate"] = float(y_tr.mean())
   144→        # Day 2 Track E — audit path comes from Settings (defaults to the
   145→        # same ``out/audit.jsonl`` as before, but now .env-configurable +
   146→        # docker-compose-wired). The AuditLogger is dual-mode internally.
   147→        state["audit"] = AuditLogger(audit_path or settings.audit_path)
   148→        # Pass Settings-backed CSV keys to default_keys so the .env file is
   149→        # honored (Track B's Dockerfile change removed the baked ENV defaults;
   150→        # docker-compose now sets them via the environment block).
   151→        state["keys"] = default_keys(
   152→            scorer_keys=settings.rto_scorer_keys,
   153→            admin_keys=settings.rto_admin_keys,
   154→        )
   155→        state["bucket"] = TokenBucket(scorer_rate_per_min)
   156→        state["rules"] = RulesEngine()
   157→        state["breaker"] = CircuitBreaker()
   158→        state["metrics"] = Metrics()
   159→        state["cases"] = CaseService(settings.cases_path)
   160→        num_cols = [c for c in X_tr.columns if str(X_tr[c].dtype) != "category"]
   161→        state["psi_sample"] = {
   162→            c: X_tr[c].dropna().sample(n=min(2000, len(X_tr)), random_state=7).tolist()
   163→            for c in num_cols
   164→        }
   165→        # Day 2 Track E — §A item 4 (register_model dead in prod). Register
   166→        # the in-process HistGB with the model registry on every worker boot.
   167→        # File mode: this writes to ``out/model_registry.json`` (the existing
   168→        # behaviour — no-op for tests since they call register_model directly
   169→        # via tmp_path). Postgres mode: INSERT into the ``model_registry``
   170→        # table; the partial-unique index enforces single-champion. PR-AUC
   171→        # + ROC-AUC are surfaced as the metrics payload so the model-card
   172→        # + drift endpoints can read them.
   173→        if settings.is_postgres:
   174→            try:
   175→                from sklearn.metrics import average_precision_score, roc_auc_score
   176→                _p = state["model"].predict_proba(X_tr)[:, 1]
   177→                _pr = float(average_precision_score(y_tr, _p))
   178→                _roc = float(roc_auc_score(y_tr, _p))
   179→                register_model(
   180→                    version=f"v{int(time.time())}",
   181→                    model_path=str(model_path),
   182→                    metrics={"pr_auc": _pr, "roc_auc": _roc},
   183→                    champion=True,
   184→                )
   185→            except Exception as e:  # pragma: no cover — startup-only, defensive
   186→                print(
   187→                    f"model-registry registration skipped: "
   188→                    f"{type(e).__name__}: {e}",
   189→                    file=sys.stderr,
   190→                )
   191→        # Pre-compute labeled (y, p) sequences for the /v1/policy/cost-curves
   192→        # endpoint (Drummond & Holte 2006). The training set carries ground-
   193→        # truth is_returned labels; running the model over it once at startup
   194→        # lets the endpoint respond in O(1) for the per-threshold sweep and
   195→        # ~3-5 sec for the 500-resample bootstrap (per-row performance only).
   196→        # Note: this is in-sample — Day 2 Track E + G will swap in a held-out
   197→        # test slice + delayed-label feedback so the curve reflects deployment
   198→        # conditions, not training fit.
   199→        try:
   200→            _cost_curve_p = state["model"].predict_proba(X_tr)[:, 1].tolist()
   201→            _cost_curve_y = [int(v) for v in y_tr.tolist()]
   202→            state["cost_curve"] = {
   203→                "y_true": _cost_curve_y,
   204→                "probs": _cost_curve_p,
   205→                "n": len(_cost_curve_y),
   206→                "source": "train_df_in_sample",
   207→            }
   208→        except Exception as e:  # pragma: no cover — startup-only, defensive
   209→            print(f"cost-curve warmup skipped: {type(e).__name__}: {e}", file=sys.stderr)
   210→            state["cost_curve"] = None
   211→        yield
   212→        # Day 2 Track E — close the model-registry connection at shutdown so
   213→        # the worker doesn't leak a Postgres connection across hot-reloads.
   214→        if settings.is_postgres:
   215→            _close_registry_conn()
   216→
   217→    app = FastAPI(title="RTO Trust Layer", version="0.2.0", lifespan=lifespan)
   218→    app.state.core = state
   219→
   220→    def to_frame(o: OrderIn) -> pd.DataFrame:
   221→        row = {
   222→            "log_order_value": _log1p(o.amount_inr),
   223→            "discount_pct": 0.0,
   224→            "Items": o.items,
   225→            "OrderDay": 180,
   226→            "OrderHour": o.order_hour,
   227→            "PriorOrders": o.prior_orders,
   228→            "PriorReturns": o.prior_returns,
   229→            "is_cod": int(o.payment_method.upper() == "COD"),
   230→            "category": o.category,
   231→            "device": o.device,
   232→            "city_tier": o.city_tier,
   233→            "address_quality": o.address_quality,
   234→        }
   235→        return pd.DataFrame([row])

   237→    @app.post("/risk/score")
   238→    def score(
   239→        order: OrderIn,
   240→        authorization: str | None = Header(default=None),
   241→        idempotency_key: str | None = Header(default=None),
   242→        x_mandate: str | None = Header(default=None),
   243→        # Day 1 Track D (V3 §13): per-txn device_id + user_id for UPI Circle
   244→        # mandates (NPCI OC-201B §3.7 Issuer Bank duty + §3.3 Secondary PSP
   245→        # duty). Both default to None and are ignored for cod_order mandates.
   246→        x_device_id: str | None = Header(default=None),
   247→        x_user_id: str | None = Header(default=None),
   248→    ) -> dict:
   249→        token = bearer_token(authorization)
   250→        ok, err = check_key(token, "scorer", state["keys"])
   251→        if not ok:
   252→            raise HTTPException(status_code=401, detail=err)
   253→        client = token
   254→        if not state["bucket"].allow(client):
   255→            raise HTTPException(status_code=429, detail="rate limit exceeded")
   256→
   257→        # Day 2 Track E — idempotency cache (§A item 2: was an unbounded dict;
   258→        # now TTLCache in file mode, Postgres ``idempotency_keys`` table in
   259→        # db mode). File mode preserves the original (key, body) tuple key
   260→        # so two different request bodies with the same Idempotency-Key
   261→        # don't silently cross-pollute. Postgres mode uses the header value
   262→        # alone as the PK (the standard idempotency contract — same key,
   263→        # same response).
   264→        if settings.is_postgres:
   265→            cached = _idem_lookup_postgres(state, idempotency_key)
   266→            if cached is not None:
   267→                # 1% probabilistic cleanup of expired rows.
   268→                if (uuid.uuid4().int % 100) == 0:
   269→                    _idem_cleanup_postgres(state)
   270→                return dict(cached, replayed=True)
   271→        else:
   272→            cache_key = (idempotency_key or "", order.model_dump_json())
   273→            if idempotency_key and cache_key in state["idem"]:
   274→                return dict(state["idem"][cache_key], replayed=True)
   275→
   276→        try:
   277→            t0 = time.monotonic()
   278→            # Mandate verification (cheap; always run; enforcement happens
   279→            # later). Day 1 Track D extended the signature to pass device_id
   280→            # + user_id so UPI Circle mandates can enforce OC-201B §3.3/§3.7
   281→            # per-txn identity validation. cod_order mandates ignore both.
   282→            mandate_verdict, mandate_payload = verify_mandate(
   283→                x_mandate,
   284→                order.amount_inr,
   285→                device_id=x_device_id,
   286→                user_id=x_user_id,
   287→            )
   288→            fired = state["rules"].evaluate(order.model_dump())
   289→            rule_fired = fired.rule_id if fired else None
   290→
   291→            proba: float | None = None
   292→            reasons: list[dict] = []
   293→            degraded = False
   294→            breach_note: str | None = None
   295→            cost_breakdown: dict | None = None
   296→            decision: str | None = None
   297→            # ``decision_source`` records which layer actually chose the
   298→            # decision, for the audit trail + dashboard explainability.
   299→            # One of: rules_engine_block | mandate_breach | mandate_invalid
   300→            # | mandate_review_required | degraded_review | cost_optimal_bmr
   301→            # | cost_optimal_bmr_review_rule
   302→            decision_source: str
   303→
   304→            # ------------------------------------------------------------------
   305→            # Decision precedence (per Day 1 Track C spec):
   306→            #   1. Rules fast-path (BLOCK)         → short-circuit REJECT
   307→            #   2. Mandate enforcement (BREACH /
   308→            #      TAMPERED / EXPIRED w/ header)   → short-circuit REJECT
   309→            #   2c. Mandate REVIEW (UPI Circle 24h
   310→            #      cooling period, OC-201B)        → short-circuit REVIEW
   311→            #      (Day 1 Track D — mandate action-class expansion)
   312→            #   3. Circuit breaker OPEN            → degraded rules-only REVIEW
   313→            #   4. optimal_decision(p, ...)         → PRIMARY decision path
   314→            #      (replaces legacy static 0.15 / 0.60 thresholds)
   315→            # REVIEW rules don't short-circuit; they gate step 4 to never ACCEPT.
   316→            # ------------------------------------------------------------------
   317→            if fired is not None and fired.action == "BLOCK":
   318→                # 1. Rules-engine BLOCK short-circuits — no model invocation.
   319→                decision = "REJECT"
   320→                decision_source = "rules_engine_block"
   321→            elif mandate_verdict == MandateVerdict.BREACH:
   322→                # 2a. Mandate amount-breach short-circuits.
   323→                decision = "REJECT"
   324→                breach_note = "mandate_amount_breach"
   325→                decision_source = "mandate_breach"
   326→            elif x_mandate is not None and mandate_verdict == MandateVerdict.REVIEW:
   327→                # 2c. UPI Circle 24h cooling-period gate (OC-201B). The txn
   328→                # is permitted in principle; the cooling circuit requires a
   329→                # human to approve before the issuer debits. Routes to the
   330→                # case queue like other REVIEW decisions. verdict_reason
   331→                # (e.g. "cooling_period_active") flows into the audit trail
   332→                # for compliance traceability.
   333→                decision = "REVIEW"
   334→                breach_note = "mandate_review_required"
   335→                decision_source = "mandate_review_required"
   336→            elif x_mandate is not None and mandate_verdict != MandateVerdict.VALID:
   337→                # 2b. Tampered / expired mandate with a header present.
   338→                # Covers OC-201B 6-month inactivity auto-revoke (verdict_reason
   339→                # = "inactivity_auto_revoke") and standard TTL expiry.
   340→                decision = "REJECT"
   341→                breach_note = f"mandate_{mandate_verdict}"
   342→                decision_source = "mandate_invalid"
   343→            else:
   344→                # 3. Circuit breaker guards model invocation.
   345→                use_model = state["breaker"].allow_attempt()
   346→                if use_model:
   347→                    try:
   348→                        X, _ = build_feature_frame(to_frame(order), "order+addr")
   349→                        reasons = reason_codes_batch(
   350→                            state["model"],
   351→                            X,
   352→                            list(X.columns),
   353→                            state["base_rate"],
   354→                            state["reference"],
   355→                        )
   356→                        proba = float(state["model"].predict_proba(X)[0, 1])
   357→                        state["breaker"].record_success()
   358→                    except Exception:
   359→                        state["breaker"].record_failure()
   360→                        proba = None
   361→                if proba is None:
   362→                    # 3. Degraded path: model unavailable → rules-only REVIEW.
   363→                    degraded = True
   364→                    decision = "REVIEW"
   365→                    decision_source = "degraded_review"
   366→                else:
   367→                    # 4. Bahnsen Bayes Minimum Risk decision layer (ICMLA 2013,
   368→                    #    DOI 10.1109/ICMLA.2013.68). Per-order argmin of expected
   369→                    #    cost over {ACCEPT, REVIEW (selective-OTP), REJECT}.
   370→                    decision, costs = optimal_decision(proba, **DEFAULT_COST_WEIGHTS)
   371→                    cost_breakdown = costs
   372→                    # REVIEW rule gate: never ACCEPT when a REVIEW rule fired
   373→                    # (still uses BMR cost math, but the rule forces REVIEW).
   374→                    if fired is not None and fired.action == "REVIEW" and decision == "ACCEPT":
   375→                        decision = "REVIEW"
   376→                        decision_source = "cost_optimal_bmr_review_rule"
   377→                    else:
   378→                        decision_source = "cost_optimal_bmr"
   379→
   380→            # Legacy ``policy_hint`` is the BMR-optimal action string (kept for
   381→            # dashboard / consumer backward compat; the actual decision is the
   382→            # body's ``decision`` field — which equals ``policy_hint`` whenever
   383→            # the cost-optimizer was the decision source).
   384→            policy_hint = None
   385→            if proba is not None:
   386→                policy_hint = optimal_decision(proba, **DEFAULT_COST_WEIGHTS)[0]
   387→
   388→            features_used = (
   389→                {c: float(X.iloc[0][c]) for c in X.columns if str(X[c].dtype) != "category"}
   390→                if proba is not None
   391→                else {}
   392→            )
   393→            case_id = None
   394→            if decision == "REVIEW":
   395→                case_id = state["cases"].open_case(
   396→                    prediction_id="pending", order_id=order.order_id,
   397→                    reason=(
   398→                        f"mandate:{mandate_payload.get('verdict_reason')}"
   399→                        if decision_source == "mandate_review_required"
   400→                        else "review_gate" if fired is None
   401→                        else f"rule:{fired.rule_id}"
   402→                    ),
   403→                )
   404→            # Audit payload (Day 1 Track D — V3 §13 mandate action-class
   405→            # expansion). The new fields surface OC-201B compliance metadata
   406→            # in the tamper-evident hash chain: ``mandate_type``,
   407→            # ``bh_purpose_code``, ``device_id``, ``user_id``, and the
   408→            # machine-readable ``mandate_verdict_reason``. The audit logger
   409→            # (Track E Day 2) stores arbitrary JSON payloads; we just pass
   410→            # the new keys through — no logger.py modification needed.
   411→            audit_id = state["audit"].log(
   412→                {
   413→                    "request": {
   414→                        **order.model_dump(),
   415→                        "customer_id": redact_customer(order.customer_id),
   416→                    },
   417→                    "probability": round(proba, 5) if proba is not None else None,
   418→                    "decision": decision,
   419→                    "decision_source": decision_source,
   420→                    "cost_breakdown": cost_breakdown,
   421→                    "reason_codes": reasons[:5],
   422→                    "mandate_verdict": mandate_verdict,
   423→                    "mandate_verdict_reason": mandate_payload.get("verdict_reason"),
   424→                    "mandate_type": mandate_payload.get("mandate_type"),
   425→                    "bh_purpose_code": mandate_payload.get("bh_purpose_code"),
   426→                    "device_id": x_device_id,
   427→                    "user_id": x_user_id,
   428→                    "breach_note": breach_note,
   429→                    "rule_fired": rule_fired,
   430→                    "degraded": degraded,
   431→                    "features_used": features_used,
   432→                    "latency_ms": now_ms(t0),
   433→                    "case_id": case_id,
   434→                }
   435→            )
   436→            body = {
   437→                "prediction_id": str(uuid.uuid4()),
   438→                "risk_score": round(proba * 100, 1) if proba is not None else None,
   439→                "probability": round(proba, 4) if proba is not None else None,
   440→                "decision": decision,
   441→                # ``gate_thresholds`` now reports the policy in force. Legacy
   442→                # accept_below / reject_above kept for backward compat — but
   443→                # the live decision path no longer consults them; it consults
   444→                # ``optimal_decision(p, weights=...)``. The dashboard reads
   445→                # ``policy`` to label the decision card.
   446→                "gate_thresholds": {
   447→                    "policy": "cost_optimal_bmr",
   448→                    "weights": DEFAULT_COST_WEIGHTS,
   449→                    "legacy_accept_t": ACCEPT_T,
   450→                    "legacy_reject_t": REJECT_T,
   451→                },
   452→                "decision_source": decision_source,
   453→                "cost_breakdown": cost_breakdown,
   454→                "explanation": reasons[:5],
   455→                "rule_fired": rule_fired,
   456→                "degraded": degraded,
   457→                "policy_hint": policy_hint,
   458→                "model_version": "rules_only"
   459→                if degraded

   682→    @app.get("/v1/policy/cost-curves")
   683→    def policy_cost_curves(
   684→        n_resamples: int = DEFAULT_COST_CURVE_RESAMPLES,
   685→        confidence: float = DEFAULT_COST_CURVE_CONFIDENCE,
   686→        authorization: str | None = Header(default=None),
   687→    ) -> dict:
   688→        """Drummond-Holte cost-curve explorer (Machine Learning 65:95-130, 2006,
   689→        DOI 10.1007/s10994-006-8199-5).
   690→
   691→        Returns a threshold sweep (0.05 → 0.95) over the labeled dataset that
   692→        the in-process model was trained on. For each threshold we report the
   693→        confusion counts, total Bahnsen cost (Eq. 1), precision, recall, plus
   694→        a 90% bootstrap CI on cost (≥500 resamples preserving row marginals —
   695→        per Drummond-Holte ``bootstrap_performance_ci`` capability).
   696→
   697→        The dashboard cost-curve explorer consumes this endpoint to render
   698→        real bars (replacing the previously-hardcoded ``COSTS`` array).
   699→
   700→        Auth: scorer scope. Returns 503 if the model isn't loaded or no
   701→        labeled data is available — never crashes the dashboard.
   702→        """
   703→        ok, err = check_key(bearer_token(authorization), "scorer", state["keys"])
   704→        if not ok:
   705→            raise HTTPException(status_code=401, detail=err)
   706→
   707→        # 1. Guard: model must be loaded (circuit breaker may be OPEN).
   708→        if state.get("model") is None:
   709→            raise HTTPException(
   710→                status_code=503,
   711→                detail="cost curves unavailable — model not loaded",
   712→            )
   713→        # 2. Guard: pre-computed labeled data must exist (startup warmup may
   714→        # have failed silently; surface a clear error to the dashboard).
   715→        cc = state.get("cost_curve")
   716→        if not cc or not cc.get("y_true") or not cc.get("probs"):
   717→            raise HTTPException(
   718→                status_code=503,
   719→                detail="cost curves unavailable — no labeled data available",
   720→            )
   721→        if n_resamples < 1 or n_resamples > 5000:
   722→            raise HTTPException(
   723→                status_code=422,
   724→                detail="n_resamples must be in [1, 5000]",
   725→            )
   726→        if not (0.5 < confidence < 0.999):
   727→            raise HTTPException(
   728→                status_code=422,
   729→                detail="confidence must be in (0.5, 0.999)",
   730→            )
   731→
   732→        y_true = cc["y_true"]
   733→        probs = cc["probs"]
   734→        weights = DEFAULT_COST_WEIGHTS
   735→        c_fp = float(weights["c_fp"])
   736→        c_fn = float(weights["c_fn"])
   737→
   738→        # 3. Threshold sweep → per-threshold confusion + cost + precision +
   739→        #    recall (pure-Python, O(N*T); ~140k ops for 7235 rows × 19 thr).
   740→        curves = cost_curve_sweep(
   741→            y_true=y_true,
   742→            probs=probs,
   743→            c_fp=c_fp,
   744→            c_fn=c_fn,
   745→        )
   746→
   747→        # 4. Bootstrap CI per threshold (Drummond-Holte row-marginal-preserving
   748→        #    resampling; ~3-5 sec for 500 resamples × 19 thr × 7235 samples).
   749→        try:
   750→            bootstrap_ci = bootstrap_cost_ci(
   751→                y_true=y_true,
   752→                probs=probs,
   753→                c_fp=c_fp,
   754→                c_fn=c_fn,
   755→                n_resamples=n_resamples,
   756→                confidence=confidence,
   757→                seed=42,
   758→            )
   759→        except Exception as e:  # pragma: no cover — defensive
   760→            # If bootstrap fails for any reason, return degenerate CIs and a
   761→            # warning rather than 500-ing the dashboard.
   762→            bootstrap_ci = {
   763→                str(r["threshold"]): {"low": 0.0, "high": 0.0, "mean": 0.0}
   764→                for r in curves
   765→            }
   766→            print(
   767→                f"cost-curves bootstrap degraded: {type(e).__name__}: {e}",
   768→                file=sys.stderr,
   769→            )
   770→
   771→        # 5. Optimal threshold = the one minimizing total cost (this is the
   772→        #    threshold analog of what ``optimal_decision()`` does per order).
   773→        optimal_record = min(curves, key=lambda r: r["cost"])
   774→        optimal_threshold = float(optimal_record["threshold"])
   775→
   776→        # 6. Cost crossover: if a challenger model is registered in the model
   777→        #    registry, find the threshold where it beats the incumbent. For
   778→        #    now there is one in-process model so this is a degenerate case —
   779→        #    the incumbent *is* the only model. We still surface the field so
   780→        #    the dashboard wiring is forward-compatible with Day 2 Track E.
   781→        cost_crossover: dict | None = None
   782→        try:
   783→            challenger = None
   784→            champ = current_champion()
   785→            # When more than one model is registered, treat the latest
   786→            # non-champion as the challenger. (For now: single model, None.)
   787→            if champ is not None:
   788→                from src.ml.registry import load_registry
   789→                reg = load_registry()
   790→                non_champ = [m for m in reg.get("models", []) if not m.get("is_champion")]
   791→                challenger = non_champ[-1] if non_champ else None
   792→            if challenger is not None and challenger.get("version") != (champ or {}).get("version"):
   793→                # We don't have challenger probabilities pre-computed yet
   794→                # (single-model registry). Surface a placeholder that the
   795→                # dashboard renders as "single-model — no crossover yet".
   796→                cost_crossover = {
   797→                    "status": "single_model",
   798→                    "incumbent_version": (champ or {}).get("version"),
   799→                    "challenger_version": None,
   800→                    "crossover_threshold": None,
   801→                    "note": (
   802→                        "challenger predictions not pre-computed; "
   803→                        "wire Day 2 Track E model-registry to enable"
   804→                    ),
   805→                }
   806→            else:
   807→                cost_crossover = {
   808→                    "status": "single_model",
   809→                    "incumbent_version": (champ or {}).get("version"),
   810→                    "challenger_version": None,
   811→                    "crossover_threshold": None,
   812→                    "note": "only one model registered; no crossover available",
   813→                }
   814→        except Exception:  # pragma: no cover — defensive
   815→            cost_crossover = {"status": "error", "crossover_threshold": None}
   816→
   817→        return {
   818→            "thresholds": [r["threshold"] for r in curves],
   819→            "curves": curves,
   820→            "bootstrap_ci": bootstrap_ci,
   821→            "optimal_threshold": optimal_threshold,
   822→            "cost_crossover": cost_crossover,
   823→            "cost_model": {
   824→                "c_fp": c_fp,
   825→                "c_fn": c_fn,
   826→                "c_otp": float(weights["c_otp"]),
   827→                "c_block": float(weights["c_block"]),
   828→                "otp_effectiveness": float(weights["otp_effectiveness"]),
   829→                "source_paper": "Bahnsen ICMLA 2013, DOI 10.1109/ICMLA.2013.68",
   830→                "curve_paper": "Drummond & Holte 2006, DOI 10.1007/s10994-006-8199-5",
   831→            },
   832→            "data_source": cc.get("source", "unknown"),
   833→            "n_samples": int(cc.get("n", 0)),
   834→            "n_pos": int(sum(y_true)),
   835→            "n_neg": int(len(y_true) - sum(y_true)),
   836→        }

   986→def _idem_get_conn(state: dict):
   987→    """Lazily-open a dedicated psycopg connection for the idempotency cache.
   988→
   989→    Separate from the AuditLogger / CaseService / model-registry connections
   990→    so a slow idempotency lookup can't block an audit write. The connection
   991→    is cached on ``state`` so we don't pay the connect-handshake cost on
   992→    every request.
   993→    """
   994→    conn = state.get("_idem_conn")
   995→    if conn is not None:
   996→        return conn
   997→    import psycopg
   998→
   999→    conn = psycopg.connect(state["settings"].database_url, autocommit=False)
  1000→    state["_idem_conn"] = conn
  1001→    return conn

  1004→def _idem_lookup_postgres(state: dict, key: str | None) -> dict | None:
  1005→    """Read a cached idempotent response. Returns the body dict (parsed from
  1006→    the JSONB-stored response_body TEXT column) or None if not cached /
  1007→    expired. ``expires_at > NOW()`` is the TTL gate.
  1008→    """
  1009→    if not key:
  1010→        return None
  1011→    import json as _json
  1012→
  1013→    conn = _idem_get_conn(state)
  1014→    with conn.cursor() as cur:
  1015→        cur.execute(
  1016→            """
  1017→            SELECT response_body, status_code
  1018→              FROM idempotency_keys
  1019→             WHERE key = %s AND expires_at > NOW()
  1020→            """,
  1021→            (key,),
  1022→        )
  1023→        row = cur.fetchone()
  1024→    if row is None:
  1025→        return None
  1026→    body, _status = row
  1027→    return _json.loads(body)

  1030→def _idem_store_postgres(
  1031→    state: dict,
  1032→    key: str,
  1033→    request_body: str,
  1034→    response_body: dict,
  1035→    status_code: int,
  1036→) -> None:
  1037→    """INSERT an idempotency cache row. TTL = NOW() + settings.idem_ttl_seconds."""
  1038→    import json as _json
  1039→    from datetime import datetime, timedelta, timezone
  1040→
  1041→    conn = _idem_get_conn(state)
  1042→    expires = datetime.now(timezone.utc) + timedelta(seconds=state["settings"].idem_ttl_seconds)
  1043→    with conn.cursor() as cur:
  1044→        # ON CONFLICT DO UPDATE so a re-store of the same key (e.g. a slow
  1045→        # retry that lands after the original INSERT) refreshes expires_at
  1046→        # rather than erroring.
  1047→        cur.execute(
  1048→            """
  1049→            INSERT INTO idempotency_keys
  1050→              (key, request_body, response_body, status_code, expires_at)
  1051→            VALUES (%s, %s, %s, %s, %s)
  1052→            ON CONFLICT (key) DO UPDATE SET
  1053→              request_body  = EXCLUDED.request_body,
  1054→              response_body = EXCLUDED.response_body,
  1055→              status_code   = EXCLUDED.status_code,
  1056→              expires_at    = EXCLUDED.expires_at
  1057→            """,
  1058→            (key, request_body, _json.dumps(response_body), status_code, expires),
  1059→        )
  1060→        conn.commit()

  1063→def _idem_cleanup_postgres(state: dict) -> None:
  1064→    """Best-effort TTL cleanup. Called with 1% probability per request so the
  1065→    table doesn't grow forever under burst traffic.
  1066→    """
  1067→    conn = _idem_get_conn(state)
  1068→    try:
  1069→        with conn.cursor() as cur:
  1070→            cur.execute("DELETE FROM idempotency_keys WHERE expires_at < NOW()")
  1071→            conn.commit()
  1072→    except Exception:  # pragma: no cover — best-effort, never fail the request
  1073→        try:
  1074→            conn.rollback()
  1075→        except Exception:
  1076→            pass
