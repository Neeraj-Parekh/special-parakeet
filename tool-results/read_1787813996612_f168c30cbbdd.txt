   117→def create_app(
   118→    scorer_rate_per_min: int = 120, audit_path: str = "out/audit.jsonl"
   119→) -> FastAPI:
   120→    state: dict[str, Any] = {}
   121→
   122→    # Day 2 Track E — load Settings once at app-construction time. The
   123→    # ``database_url`` field selects between Postgres mode (full DB-backed
   124→    # audit/cases/registry/idempotency) and file-mode fallback (the 63
   125→    # existing tests run this way). The Settings object itself is cached
   126→    # via ``@lru_cache`` in ``src.config`` so we don't re-read env vars on
   127→    # every TestClient build.
   128→    settings = get_settings()
   129→    state["settings"] = settings
   130→
   131→    # Day 2 Track F — Redis Streams producer (fire-and-forget). Constructed
   132→    # eagerly here (cheap — just stores the URL; lazy-connects on first
   133→    # ``publish``) so the /risk/score handler can call it after the audit
   134→    # write + case open. When ``settings.redis_url`` is None (test mode,
   135→    # local dev without REDIS_URL) the producer is a no-op: ``publish()``
   136→    # returns None silently — the API response is unaffected. This is the
   137→    # pragmatic fire-and-forget contract; V3 §10.3 prescribes a full
   138→    # transactional outbox table drained by a worker (deferred; see
   139→    # worklog Day 2 Track F deferral list).
   140→    state["stream"] = StreamProducer(settings.redis_url)
   141→
   142→    # Idempotency cache. File mode: TTLCache (bounded + auto-expiring —
   143→    # closes §A item 2 memory leak). Postgres mode: the ``idempotency_keys``
   144→    # table; ``state["idem"]`` is unused (kept as an empty dict for backward-
   145→    # compat with any test that asserts the key exists).
   146→    if settings.is_postgres:
   147→        state["idem"] = TTLCache(maxsize=1, ttl=3600)  # placeholder, real cache is the DB
   148→    else:
   149→        state["idem"] = TTLCache(
   150→            maxsize=settings.idem_maxsize, ttl=settings.idem_ttl_seconds
   151→        )
   152→
   153→    @asynccontextmanager
   154→    async def lifespan(_: FastAPI):
   155→        df = add_address_features(load_orders("data/raw/cod_orders.csv"))
   156→        train_df, _ = group_split(df)
   157→        X_tr, y_tr = build_feature_frame(train_df, "order+addr")
   158→        model_path = Path("out/model_api.joblib")
   159→        if not model_path.exists():
   160→            save_model(fit_model(X_tr, y_tr), str(model_path))
   161→        from src.models.train import load_model
   162→
   163→        state["model"] = load_model(str(model_path))
   164→        state["reference"] = X_tr.mode().iloc[0]
   165→        state["base_rate"] = float(y_tr.mean())
   166→        # Day 2 Track E — audit path comes from Settings (defaults to the
   167→        # same ``out/audit.jsonl`` as before, but now .env-configurable +
   168→        # docker-compose-wired). The AuditLogger is dual-mode internally.
   169→        state["audit"] = AuditLogger(audit_path or settings.audit_path)
   170→        # Pass Settings-backed CSV keys to default_keys so the .env file is
   171→        # honored (Track B's Dockerfile change removed the baked ENV defaults;
   172→        # docker-compose now sets them via the environment block).
   173→        state["keys"] = default_keys(
   174→            scorer_keys=settings.rto_scorer_keys,
   175→            admin_keys=settings.rto_admin_keys,
   176→        )
   177→        state["bucket"] = TokenBucket(scorer_rate_per_min)
   178→        state["rules"] = RulesEngine()
   179→        state["breaker"] = CircuitBreaker()
   180→        state["metrics"] = Metrics()
   181→        state["cases"] = CaseService(settings.cases_path)
   182→        num_cols = [c for c in X_tr.columns if str(X_tr[c].dtype) != "category"]
   183→        state["psi_sample"] = {
   184→            c: X_tr[c].dropna().sample(n=min(2000, len(X_tr)), random_state=7).tolist()
   185→            for c in num_cols
   186→        }
   187→        # Day 2 Track E — §A item 4 (register_model dead in prod). Register
   188→        # the in-process HistGB with the model registry on every worker boot.
   189→        # File mode: this writes to ``out/model_registry.json`` (the existing
   190→        # behaviour — no-op for tests since they call register_model directly
   191→        # via tmp_path). Postgres mode: INSERT into the ``model_registry``
   192→        # table; the partial-unique index enforces single-champion. PR-AUC
   193→        # + ROC-AUC are surfaced as the metrics payload so the model-card
   194→        # + drift endpoints can read them.
   195→        if settings.is_postgres:
   196→            try:
   197→                from sklearn.metrics import average_precision_score, roc_auc_score
   198→                _p = state["model"].predict_proba(X_tr)[:, 1]
   199→                _pr = float(average_precision_score(y_tr, _p))
   200→                _roc = float(roc_auc_score(y_tr, _p))
   201→                register_model(
   202→                    version=f"v{int(time.time())}",
   203→                    model_path=str(model_path),
   204→                    metrics={"pr_auc": _pr, "roc_auc": _roc},
   205→                    champion=True,
   206→                )
   207→            except Exception as e:  # pragma: no cover — startup-only, defensive
   208→                print(
   209→                    f"model-registry registration skipped: "
   210→                    f"{type(e).__name__}: {e}",
   211→                    file=sys.stderr,
   212→                )
   213→        # Pre-compute labeled (y, p) sequences for the /v1/policy/cost-curves
   214→        # endpoint (Drummond & Holte 2006). The training set carries ground-
   215→        # truth is_returned labels; running the model over it once at startup
   216→        # lets the endpoint respond in O(1) for the per-threshold sweep and
   217→        # ~3-5 sec for the 500-resample bootstrap (per-row performance only).
   218→        # Note: this is in-sample — Day 2 Track E + G will swap in a held-out
   219→        # test slice + delayed-label feedback so the curve reflects deployment
   220→        # conditions, not training fit.
   221→        try:
   222→            _cost_curve_p = state["model"].predict_proba(X_tr)[:, 1].tolist()
   223→            _cost_curve_y = [int(v) for v in y_tr.tolist()]
   224→            state["cost_curve"] = {
   225→                "y_true": _cost_curve_y,
   226→                "probs": _cost_curve_p,
   227→                "n": len(_cost_curve_y),
   228→                "source": "train_df_in_sample",
   229→            }
   230→        except Exception as e:  # pragma: no cover — startup-only, defensive
   231→            print(f"cost-curve warmup skipped: {type(e).__name__}: {e}", file=sys.stderr)
   232→            state["cost_curve"] = None
   233→        yield
   234→        # Day 2 Track E — close the model-registry connection at shutdown so
   235→        # the worker doesn't leak a Postgres connection across hot-reloads.
   236→        if settings.is_postgres:
   237→            _close_registry_conn()
   238→
   239→    app = FastAPI(title="RTO Trust Layer", version="0.2.0", lifespan=lifespan)
   240→    app.state.core = state

   242→    def to_frame(o: OrderIn) -> pd.DataFrame:
   243→        row = {
   244→            "log_order_value": _log1p(o.amount_inr),
   245→            "discount_pct": 0.0,
   246→            "Items": o.items,
   247→            "OrderDay": 180,
   248→            "OrderHour": o.order_hour,
   249→            "PriorOrders": o.prior_orders,
   250→            "PriorReturns": o.prior_returns,
   251→            "is_cod": int(o.payment_method.upper() == "COD"),
   252→            "category": o.category,
   253→            "device": o.device,
   254→            "city_tier": o.city_tier,
   255→            "address_quality": o.address_quality,
   256→        }
   257→        return pd.DataFrame([row])

   259→    @app.post("/risk/score")
   260→    def score(
   261→        order: OrderIn,
   262→        authorization: str | None = Header(default=None),
   263→        idempotency_key: str | None = Header(default=None),
   264→        x_mandate: str | None = Header(default=None),
   265→        # Day 1 Track D (V3 §13): per-txn device_id + user_id for UPI Circle
   266→        # mandates (NPCI OC-201B §3.7 Issuer Bank duty + §3.3 Secondary PSP
   267→        # duty). Both default to None and are ignored for cod_order mandates.
   268→        x_device_id: str | None = Header(default=None),
   269→        x_user_id: str | None = Header(default=None),
   270→    ) -> dict:
   271→        token = bearer_token(authorization)
   272→        ok, err = check_key(token, "scorer", state["keys"])
   273→        if not ok:
   274→            raise HTTPException(status_code=401, detail=err)
   275→        client = token
   276→        if not state["bucket"].allow(client):
   277→            raise HTTPException(status_code=429, detail="rate limit exceeded")
   278→
   279→        # Day 2 Track E — idempotency cache (§A item 2: was an unbounded dict;
   280→        # now TTLCache in file mode, Postgres ``idempotency_keys`` table in
   281→        # db mode). File mode preserves the original (key, body) tuple key
   282→        # so two different request bodies with the same Idempotency-Key
   283→        # don't silently cross-pollute. Postgres mode uses the header value
   284→        # alone as the PK (the standard idempotency contract — same key,
   285→        # same response).
   286→        if settings.is_postgres:
   287→            cached = _idem_lookup_postgres(state, idempotency_key)
   288→            if cached is not None:
   289→                # 1% probabilistic cleanup of expired rows.
   290→                if (uuid.uuid4().int % 100) == 0:
   291→                    _idem_cleanup_postgres(state)
   292→                return dict(cached, replayed=True)
   293→        else:
   294→            cache_key = (idempotency_key or "", order.model_dump_json())
   295→            if idempotency_key and cache_key in state["idem"]:
   296→                return dict(state["idem"][cache_key], replayed=True)
   297→
   298→        try:
   299→            t0 = time.monotonic()
   300→            # Mandate verification (cheap; always run; enforcement happens
   301→            # later). Day 1 Track D extended the signature to pass device_id
   302→            # + user_id so UPI Circle mandates can enforce OC-201B §3.3/§3.7
   303→            # per-txn identity validation. cod_order mandates ignore both.
   304→            mandate_verdict, mandate_payload = verify_mandate(
   305→                x_mandate,
   306→                order.amount_inr,
   307→                device_id=x_device_id,
   308→                user_id=x_user_id,
   309→            )
   310→            fired = state["rules"].evaluate(order.model_dump())
   311→            rule_fired = fired.rule_id if fired else None
   312→
   313→            proba: float | None = None
   314→            reasons: list[dict] = []
   315→            degraded = False
   316→            breach_note: str | None = None
   317→            cost_breakdown: dict | None = None
   318→            decision: str | None = None
   319→            # ``decision_source`` records which layer actually chose the
   320→            # decision, for the audit trail + dashboard explainability.
   321→            # One of: rules_engine_block | mandate_breach | mandate_invalid
   322→            # | mandate_review_required | degraded_review | cost_optimal_bmr
   323→            # | cost_optimal_bmr_review_rule
   324→            decision_source: str
   325→
   326→            # ------------------------------------------------------------------
   327→            # Decision precedence (per Day 1 Track C spec):
   328→            #   1. Rules fast-path (BLOCK)         → short-circuit REJECT
   329→            #   2. Mandate enforcement (BREACH /
   330→            #      TAMPERED / EXPIRED w/ header)   → short-circuit REJECT
   331→            #   2c. Mandate REVIEW (UPI Circle 24h
   332→            #      cooling period, OC-201B)        → short-circuit REVIEW
   333→            #      (Day 1 Track D — mandate action-class expansion)
   334→            #   3. Circuit breaker OPEN            → degraded rules-only REVIEW
   335→            #   4. optimal_decision(p, ...)         → PRIMARY decision path
   336→            #      (replaces legacy static 0.15 / 0.60 thresholds)
   337→            # REVIEW rules don't short-circuit; they gate step 4 to never ACCEPT.
   338→            # ------------------------------------------------------------------
   339→            if fired is not None and fired.action == "BLOCK":
   340→                # 1. Rules-engine BLOCK short-circuits — no model invocation.
   341→                decision = "REJECT"
   342→                decision_source = "rules_engine_block"
   343→            elif mandate_verdict == MandateVerdict.BREACH:
   344→                # 2a. Mandate amount-breach short-circuits.
   345→                decision = "REJECT"
   346→                breach_note = "mandate_amount_breach"
   347→                decision_source = "mandate_breach"
   348→            elif x_mandate is not None and mandate_verdict == MandateVerdict.REVIEW:
   349→                # 2c. UPI Circle 24h cooling-period gate (OC-201B). The txn
   350→                # is permitted in principle; the cooling circuit requires a
   351→                # human to approve before the issuer debits. Routes to the
   352→                # case queue like other REVIEW decisions. verdict_reason
   353→                # (e.g. "cooling_period_active") flows into the audit trail
   354→                # for compliance traceability.
   355→                decision = "REVIEW"
   356→                breach_note = "mandate_review_required"
   357→                decision_source = "mandate_review_required"
   358→            elif x_mandate is not None and mandate_verdict != MandateVerdict.VALID:
   359→                # 2b. Tampered / expired mandate with a header present.
   360→                # Covers OC-201B 6-month inactivity auto-revoke (verdict_reason
   361→                # = "inactivity_auto_revoke") and standard TTL expiry.
   362→                decision = "REJECT"
   363→                breach_note = f"mandate_{mandate_verdict}"
   364→                decision_source = "mandate_invalid"
   365→            else:
   366→                # 3. Circuit breaker guards model invocation.
   367→                use_model = state["breaker"].allow_attempt()
   368→                if use_model:
   369→                    try:
   370→                        X, _ = build_feature_frame(to_frame(order), "order+addr")
   371→                        reasons = reason_codes_batch(
   372→                            state["model"],
   373→                            X,
   374→                            list(X.columns),
   375→                            state["base_rate"],
   376→                            state["reference"],
   377→                        )
   378→                        proba = float(state["model"].predict_proba(X)[0, 1])
   379→                        state["breaker"].record_success()
   380→                    except Exception:
   381→                        state["breaker"].record_failure()
   382→                        proba = None
   383→                if proba is None:
   384→                    # 3. Degraded path: model unavailable → rules-only REVIEW.
   385→                    degraded = True
   386→                    decision = "REVIEW"
   387→                    decision_source = "degraded_review"
   388→                else:
   389→                    # 4. Bahnsen Bayes Minimum Risk decision layer (ICMLA 2013,
   390→                    #    DOI 10.1109/ICMLA.2013.68). Per-order argmin of expected
   391→                    #    cost over {ACCEPT, REVIEW (selective-OTP), REJECT}.
   392→                    decision, costs = optimal_decision(proba, **DEFAULT_COST_WEIGHTS)
   393→                    cost_breakdown = costs
   394→                    # REVIEW rule gate: never ACCEPT when a REVIEW rule fired
   395→                    # (still uses BMR cost math, but the rule forces REVIEW).
   396→                    if fired is not None and fired.action == "REVIEW" and decision == "ACCEPT":
   397→                        decision = "REVIEW"
   398→                        decision_source = "cost_optimal_bmr_review_rule"
   399→                    else:
   400→                        decision_source = "cost_optimal_bmr"
   401→
   402→            # Legacy ``policy_hint`` is the BMR-optimal action string (kept for
   403→            # dashboard / consumer backward compat; the actual decision is the
   404→            # body's ``decision`` field — which equals ``policy_hint`` whenever
   405→            # the cost-optimizer was the decision source).
   406→            policy_hint = None
   407→            if proba is not None:
   408→                policy_hint = optimal_decision(proba, **DEFAULT_COST_WEIGHTS)[0]
   409→
   410→            features_used = (
   411→                {c: float(X.iloc[0][c]) for c in X.columns if str(X[c].dtype) != "category"}
   412→                if proba is not None
   413→                else {}
   414→            )
   415→            # Day 2 Track F — generate the prediction_id ONCE so the case
   416→            # row, the audit record's stream publish, and the response body
   417→            # all share the same identifier. Track B's original code passed
   418→            # ``prediction_id="pending"`` to ``open_case`` and generated the
   419→            # real UUID inline in the body dict — that inconsistency meant
   420→            # stream consumers couldn't correlate a ``risk.scores`` message
   421→            # to a ``cases.created`` message (different prediction_id on
   422→            # each). Now the UUID is canonical across the case, the stream,
   423→            # and the response.
   424→            prediction_id = str(uuid.uuid4())
   425→            case_id = None
   426→            if decision == "REVIEW":
   427→                case_id = state["cases"].open_case(
   428→                    prediction_id=prediction_id, order_id=order.order_id,
   429→                    reason=(
   430→                        f"mandate:{mandate_payload.get('verdict_reason')}"
   431→                        if decision_source == "mandate_review_required"
   432→                        else "review_gate" if fired is None
   433→                        else f"rule:{fired.rule_id}"
   434→                    ),
   435→                )
   436→            # Audit payload (Day 1 Track D — V3 §13 mandate action-class
   437→            # expansion). The new fields surface OC-201B compliance metadata
   438→            # in the tamper-evident hash chain: ``mandate_type``,
   439→            # ``bh_purpose_code``, ``device_id``, ``user_id``, and the
   440→            # machine-readable ``mandate_verdict_reason``. The audit logger
   441→            # (Track E Day 2) stores arbitrary JSON payloads; we just pass
   442→            # the new keys through — no logger.py modification needed.
   443→            audit_id = state["audit"].log(
   444→                {
   445→                    "request": {
   446→                        **order.model_dump(),
   447→                        "customer_id": redact_customer(order.customer_id),
   448→                    },
   449→                    "probability": round(proba, 5) if proba is not None else None,
   450→                    "decision": decision,
   451→                    "decision_source": decision_source,
   452→                    "cost_breakdown": cost_breakdown,

   764→    @app.get("/v1/policy/cost-curves")
   765→    def policy_cost_curves(
   766→        n_resamples: int = DEFAULT_COST_CURVE_RESAMPLES,
   767→        confidence: float = DEFAULT_COST_CURVE_CONFIDENCE,
   768→        authorization: str | None = Header(default=None),
   769→    ) -> dict:
   770→        """Drummond-Holte cost-curve explorer (Machine Learning 65:95-130, 2006,
   771→        DOI 10.1007/s10994-006-8199-5).
   772→
   773→        Returns a threshold sweep (0.05 → 0.95) over the labeled dataset that
   774→        the in-process model was trained on. For each threshold we report the
   775→        confusion counts, total Bahnsen cost (Eq. 1), precision, recall, plus
   776→        a 90% bootstrap CI on cost (≥500 resamples preserving row marginals —
   777→        per Drummond-Holte ``bootstrap_performance_ci`` capability).
   778→
   779→        The dashboard cost-curve explorer consumes this endpoint to render
   780→        real bars (replacing the previously-hardcoded ``COSTS`` array).
   781→
   782→        Auth: scorer scope. Returns 503 if the model isn't loaded or no
   783→        labeled data is available — never crashes the dashboard.
   784→        """
   785→        ok, err = check_key(bearer_token(authorization), "scorer", state["keys"])
   786→        if not ok:
   787→            raise HTTPException(status_code=401, detail=err)
   788→
   789→        # 1. Guard: model must be loaded (circuit breaker may be OPEN).
   790→        if state.get("model") is None:
   791→            raise HTTPException(
   792→                status_code=503,
   793→                detail="cost curves unavailable — model not loaded",
   794→            )
   795→        # 2. Guard: pre-computed labeled data must exist (startup warmup may
   796→        # have failed silently; surface a clear error to the dashboard).
   797→        cc = state.get("cost_curve")
   798→        if not cc or not cc.get("y_true") or not cc.get("probs"):
   799→            raise HTTPException(
   800→                status_code=503,
   801→                detail="cost curves unavailable — no labeled data available",
   802→            )
   803→        if n_resamples < 1 or n_resamples > 5000:
   804→            raise HTTPException(
   805→                status_code=422,
   806→                detail="n_resamples must be in [1, 5000]",
   807→            )
   808→        if not (0.5 < confidence < 0.999):
   809→            raise HTTPException(
   810→                status_code=422,
   811→                detail="confidence must be in (0.5, 0.999)",
   812→            )
   813→
   814→        y_true = cc["y_true"]
   815→        probs = cc["probs"]
   816→        weights = DEFAULT_COST_WEIGHTS
   817→        c_fp = float(weights["c_fp"])
   818→        c_fn = float(weights["c_fn"])
   819→
   820→        # 3. Threshold sweep → per-threshold confusion + cost + precision +
   821→        #    recall (pure-Python, O(N*T); ~140k ops for 7235 rows × 19 thr).
   822→        curves = cost_curve_sweep(
   823→            y_true=y_true,
   824→            probs=probs,
   825→            c_fp=c_fp,
   826→            c_fn=c_fn,
   827→        )
   828→
   829→        # 4. Bootstrap CI per threshold (Drummond-Holte row-marginal-preserving
   830→        #    resampling; ~3-5 sec for 500 resamples × 19 thr × 7235 samples).
   831→        try:
   832→            bootstrap_ci = bootstrap_cost_ci(
   833→                y_true=y_true,
   834→                probs=probs,
   835→                c_fp=c_fp,
   836→                c_fn=c_fn,
   837→                n_resamples=n_resamples,
   838→                confidence=confidence,
   839→                seed=42,
   840→            )
   841→        except Exception as e:  # pragma: no cover — defensive
   842→            # If bootstrap fails for any reason, return degenerate CIs and a
   843→            # warning rather than 500-ing the dashboard.
   844→            bootstrap_ci = {
   845→                str(r["threshold"]): {"low": 0.0, "high": 0.0, "mean": 0.0}
   846→                for r in curves
   847→            }
   848→            print(
   849→                f"cost-curves bootstrap degraded: {type(e).__name__}: {e}",
   850→                file=sys.stderr,
   851→            )
   852→
   853→        # 5. Optimal threshold = the one minimizing total cost (this is the
   854→        #    threshold analog of what ``optimal_decision()`` does per order).
   855→        optimal_record = min(curves, key=lambda r: r["cost"])
   856→        optimal_threshold = float(optimal_record["threshold"])
   857→
   858→        # 6. Cost crossover: if a challenger model is registered in the model
   859→        #    registry, find the threshold where it beats the incumbent. For
   860→        #    now there is one in-process model so this is a degenerate case —
   861→        #    the incumbent *is* the only model. We still surface the field so
   862→        #    the dashboard wiring is forward-compatible with Day 2 Track E.
   863→        cost_crossover: dict | None = None
   864→        try:
   865→            challenger = None
   866→            champ = current_champion()
   867→            # When more than one model is registered, treat the latest
   868→            # non-champion as the challenger. (For now: single model, None.)
   869→            if champ is not None:
   870→                from src.ml.registry import load_registry
   871→                reg = load_registry()
   872→                non_champ = [m for m in reg.get("models", []) if not m.get("is_champion")]
   873→                challenger = non_champ[-1] if non_champ else None
   874→            if challenger is not None and challenger.get("version") != (champ or {}).get("version"):
   875→                # We don't have challenger probabilities pre-computed yet
   876→                # (single-model registry). Surface a placeholder that the
   877→                # dashboard renders as "single-model — no crossover yet".
   878→                cost_crossover = {
   879→                    "status": "single_model",
   880→                    "incumbent_version": (champ or {}).get("version"),
   881→                    "challenger_version": None,
   882→                    "crossover_threshold": None,
   883→                    "note": (
   884→                        "challenger predictions not pre-computed; "
   885→                        "wire Day 2 Track E model-registry to enable"
   886→                    ),
   887→                }
   888→            else:
   889→                cost_crossover = {
   890→                    "status": "single_model",
   891→                    "incumbent_version": (champ or {}).get("version"),
   892→                    "challenger_version": None,
   893→                    "crossover_threshold": None,
   894→                    "note": "only one model registered; no crossover available",
   895→                }
   896→        except Exception:  # pragma: no cover — defensive
   897→            cost_crossover = {"status": "error", "crossover_threshold": None}
   898→
   899→        return {
   900→            "thresholds": [r["threshold"] for r in curves],
   901→            "curves": curves,
   902→            "bootstrap_ci": bootstrap_ci,
   903→            "optimal_threshold": optimal_threshold,
   904→            "cost_crossover": cost_crossover,
   905→            "cost_model": {
   906→                "c_fp": c_fp,
   907→                "c_fn": c_fn,
   908→                "c_otp": float(weights["c_otp"]),
   909→                "c_block": float(weights["c_block"]),
   910→                "otp_effectiveness": float(weights["otp_effectiveness"]),
   911→                "source_paper": "Bahnsen ICMLA 2013, DOI 10.1109/ICMLA.2013.68",
   912→                "curve_paper": "Drummond & Holte 2006, DOI 10.1007/s10994-006-8199-5",
   913→            },
   914→            "data_source": cc.get("source", "unknown"),
   915→            "n_samples": int(cc.get("n", 0)),
   916→            "n_pos": int(sum(y_true)),
   917→            "n_neg": int(len(y_true) - sum(y_true)),
   918→        }

  1068→def _idem_get_conn(state: dict):
  1069→    """Lazily-open a dedicated psycopg connection for the idempotency cache.
  1070→
  1071→    Separate from the AuditLogger / CaseService / model-registry connections
  1072→    so a slow idempotency lookup can't block an audit write. The connection
  1073→    is cached on ``state`` so we don't pay the connect-handshake cost on
  1074→    every request.
  1075→    """
  1076→    conn = state.get("_idem_conn")
  1077→    if conn is not None:
  1078→        return conn
  1079→    import psycopg
  1080→
  1081→    conn = psycopg.connect(state["settings"].database_url, autocommit=False)
  1082→    state["_idem_conn"] = conn
  1083→    return conn

  1086→def _idem_lookup_postgres(state: dict, key: str | None) -> dict | None:
  1087→    """Read a cached idempotent response. Returns the body dict (parsed from
  1088→    the JSONB-stored response_body TEXT column) or None if not cached /
  1089→    expired. ``expires_at > NOW()`` is the TTL gate.
  1090→    """
  1091→    if not key:
  1092→        return None
  1093→    import json as _json
  1094→
  1095→    conn = _idem_get_conn(state)
  1096→    with conn.cursor() as cur:
  1097→        cur.execute(
  1098→            """
  1099→            SELECT response_body, status_code
  1100→              FROM idempotency_keys
  1101→             WHERE key = %s AND expires_at > NOW()
  1102→            """,
  1103→            (key,),
  1104→        )
  1105→        row = cur.fetchone()
  1106→    if row is None:
  1107→        return None
  1108→    body, _status = row
  1109→    return _json.loads(body)

  1112→def _idem_store_postgres(
  1113→    state: dict,
  1114→    key: str,
  1115→    request_body: str,
  1116→    response_body: dict,
  1117→    status_code: int,
  1118→) -> None:
  1119→    """INSERT an idempotency cache row. TTL = NOW() + settings.idem_ttl_seconds."""
  1120→    import json as _json
  1121→    from datetime import datetime, timedelta, timezone
  1122→
  1123→    conn = _idem_get_conn(state)
  1124→    expires = datetime.now(timezone.utc) + timedelta(seconds=state["settings"].idem_ttl_seconds)
  1125→    with conn.cursor() as cur:
  1126→        # ON CONFLICT DO UPDATE so a re-store of the same key (e.g. a slow
  1127→        # retry that lands after the original INSERT) refreshes expires_at
  1128→        # rather than erroring.
  1129→        cur.execute(
  1130→            """
  1131→            INSERT INTO idempotency_keys
  1132→              (key, request_body, response_body, status_code, expires_at)
  1133→            VALUES (%s, %s, %s, %s, %s)
  1134→            ON CONFLICT (key) DO UPDATE SET
  1135→              request_body  = EXCLUDED.request_body,
  1136→              response_body = EXCLUDED.response_body,
  1137→              status_code   = EXCLUDED.status_code,
  1138→              expires_at    = EXCLUDED.expires_at
  1139→            """,
  1140→            (key, request_body, _json.dumps(response_body), status_code, expires),
  1141→        )
  1142→        conn.commit()

  1145→def _idem_cleanup_postgres(state: dict) -> None:
  1146→    """Best-effort TTL cleanup. Called with 1% probability per request so the
  1147→    table doesn't grow forever under burst traffic.
  1148→    """
  1149→    conn = _idem_get_conn(state)
  1150→    try:
  1151→        with conn.cursor() as cur:
  1152→            cur.execute("DELETE FROM idempotency_keys WHERE expires_at < NOW()")
  1153→            conn.commit()
  1154→    except Exception:  # pragma: no cover — best-effort, never fail the request
  1155→        try:
  1156→            conn.rollback()
  1157→        except Exception:
  1158→            pass
