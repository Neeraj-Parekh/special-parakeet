   138→def create_app(
   139→    scorer_rate_per_min: int = 120, audit_path: str = "out/audit.jsonl"
   140→) -> FastAPI:
   141→    state: dict[str, Any] = {}
   142→
   143→    # Day 2 Track E — load Settings once at app-construction time. The
   144→    # ``database_url`` field selects between Postgres mode (full DB-backed
   145→    # audit/cases/registry/idempotency) and file-mode fallback (the 63
   146→    # existing tests run this way). The Settings object itself is cached
   147→    # via ``@lru_cache`` in ``src.config`` so we don't re-read env vars on
   148→    # every TestClient build.
   149→    settings = get_settings()
   150→    state["settings"] = settings
   151→
   152→    # Day 2 Track F — Redis Streams producer (fire-and-forget). Constructed
   153→    # eagerly here (cheap — just stores the URL; lazy-connects on first
   154→    # ``publish``) so the /risk/score handler can call it after the audit
   155→    # write + case open. When ``settings.redis_url`` is None (test mode,
   156→    # local dev without REDIS_URL) the producer is a no-op: ``publish()``
   157→    # returns None silently — the API response is unaffected. This is the
   158→    # pragmatic fire-and-forget contract; V3 §10.3 prescribes a full
   159→    # transactional outbox table drained by a worker (deferred; see
   160→    # worklog Day 2 Track F deferral list).
   161→    state["stream"] = StreamProducer(settings.redis_url)
   162→
   163→    # Day 2 Track G — LabelFeedbackService (DDM + ADWIN over the delayed
   164→    # is_returned label stream). Constructed eagerly (cheap — just stores
   165→    # the URLs + creates the DDM/ADWIN instances) so the
   166→    # /v1/feedback/ingest handler can call ``ingest_label`` per request +
   167→    # the /metrics endpoint can read ``current_state()`` for the drift
   168→    # gauges. The lazy StreamProducer inside the service is constructed
   169→    # only on first DRIFT publish — so the 67 existing tests + 7 feedback
   170→    # tests still pass without a Redis fixture.
   171→    state["feedback"] = LabelFeedbackService(
   172→        redis_url=settings.redis_url,
   173→        database_url=settings.database_url,
   174→    )
   175→
   176→    # Idempotency cache. File mode: TTLCache (bounded + auto-expiring —
   177→    # closes §A item 2 memory leak). Postgres mode: the ``idempotency_keys``
   178→    # table; ``state["idem"]`` is unused (kept as an empty dict for backward-
   179→    # compat with any test that asserts the key exists).
   180→    if settings.is_postgres:
   181→        state["idem"] = TTLCache(maxsize=1, ttl=3600)  # placeholder, real cache is the DB
   182→    else:
   183→        state["idem"] = TTLCache(
   184→            maxsize=settings.idem_maxsize, ttl=settings.idem_ttl_seconds
   185→        )
   186→
   187→    @asynccontextmanager
   188→    async def lifespan(_: FastAPI):
   189→        df = add_address_features(load_orders("data/raw/cod_orders.csv"))
   190→        train_df, _ = group_split(df)
   191→        X_tr, y_tr = build_feature_frame(train_df, "order+addr")
   192→        model_path = Path("out/model_api.joblib")
   193→        if not model_path.exists():
   194→            save_model(fit_model(X_tr, y_tr), str(model_path))
   195→        from src.models.train import load_model
   196→
   197→        state["model"] = load_model(str(model_path))
   198→        state["reference"] = X_tr.mode().iloc[0]
   199→        state["base_rate"] = float(y_tr.mean())
   200→        # Day 2 Track E — audit path comes from Settings (defaults to the
   201→        # same ``out/audit.jsonl`` as before, but now .env-configurable +
   202→        # docker-compose-wired). The AuditLogger is dual-mode internally.
   203→        state["audit"] = AuditLogger(audit_path or settings.audit_path)
   204→        # Pass Settings-backed CSV keys to default_keys so the .env file is
   205→        # honored (Track B's Dockerfile change removed the baked ENV defaults;
   206→        # docker-compose now sets them via the environment block).
   207→        state["keys"] = default_keys(
   208→            scorer_keys=settings.rto_scorer_keys,
   209→            admin_keys=settings.rto_admin_keys,
   210→        )
   211→        state["bucket"] = TokenBucket(scorer_rate_per_min)
   212→        state["rules"] = RulesEngine()
   213→        state["breaker"] = CircuitBreaker()
   214→        state["metrics"] = Metrics()
   215→        state["cases"] = CaseService(settings.cases_path)
   216→        num_cols = [c for c in X_tr.columns if str(X_tr[c].dtype) != "category"]
   217→        state["psi_sample"] = {
   218→            c: X_tr[c].dropna().sample(n=min(2000, len(X_tr)), random_state=7).tolist()
   219→            for c in num_cols
   220→        }
   221→        # Day 2 Track E — §A item 4 (register_model dead in prod). Register
   222→        # the in-process HistGB with the model registry on every worker boot.
   223→        # File mode: this writes to ``out/model_registry.json`` (the existing
   224→        # behaviour — no-op for tests since they call register_model directly
   225→        # via tmp_path). Postgres mode: INSERT into the ``model_registry``
   226→        # table; the partial-unique index enforces single-champion. PR-AUC
   227→        # + ROC-AUC are surfaced as the metrics payload so the model-card
   228→        # + drift endpoints can read them.
   229→        if settings.is_postgres:
   230→            try:
   231→                from sklearn.metrics import average_precision_score, roc_auc_score
   232→                _p = state["model"].predict_proba(X_tr)[:, 1]
   233→                _pr = float(average_precision_score(y_tr, _p))
   234→                _roc = float(roc_auc_score(y_tr, _p))
   235→                register_model(
   236→                    version=f"v{int(time.time())}",
   237→                    model_path=str(model_path),
   238→                    metrics={"pr_auc": _pr, "roc_auc": _roc},
   239→                    champion=True,
   240→                )
   241→            except Exception as e:  # pragma: no cover — startup-only, defensive
   242→                print(
   243→                    f"model-registry registration skipped: "
   244→                    f"{type(e).__name__}: {e}",
   245→                    file=sys.stderr,
   246→                )
   247→        # Pre-compute labeled (y, p) sequences for the /v1/policy/cost-curves
   248→        # endpoint (Drummond & Holte 2006). The training set carries ground-
   249→        # truth is_returned labels; running the model over it once at startup
   250→        # lets the endpoint respond in O(1) for the per-threshold sweep and
   251→        # ~3-5 sec for the 500-resample bootstrap (per-row performance only).
   252→        # Note: this is in-sample — Day 2 Track E + G will swap in a held-out
   253→        # test slice + delayed-label feedback so the curve reflects deployment
   254→        # conditions, not training fit.
   255→        try:
   256→            _cost_curve_p = state["model"].predict_proba(X_tr)[:, 1].tolist()
   257→            _cost_curve_y = [int(v) for v in y_tr.tolist()]
   258→            state["cost_curve"] = {
   259→                "y_true": _cost_curve_y,
   260→                "probs": _cost_curve_p,
   261→                "n": len(_cost_curve_y),
   262→                "source": "train_df_in_sample",
   263→            }
   264→        except Exception as e:  # pragma: no cover — startup-only, defensive
   265→            print(f"cost-curve warmup skipped: {type(e).__name__}: {e}", file=sys.stderr)
   266→            state["cost_curve"] = None
   267→        yield
   268→        # Day 2 Track E — close the model-registry connection at shutdown so
   269→        # the worker doesn't leak a Postgres connection across hot-reloads.
   270→        if settings.is_postgres:
   271→            _close_registry_conn()
   272→        # Day 2 Track G — close the LabelFeedbackService's lazy StreamProducer
   273→        # so the worker doesn't leak a Redis connection across hot-reloads.
   274→        # Safe to call on a service whose producer was never constructed.
   275→        try:
   276→            state["feedback"].close()
   277→        except Exception:  # pragma: no cover — best-effort shutdown
   278→            pass
   279→
   280→    app = FastAPI(title="RTO Trust Layer", version="0.2.0", lifespan=lifespan)
   281→    app.state.core = state

   300→    @app.post("/risk/score")
   301→    def score(
   302→        order: OrderIn,
   303→        authorization: str | None = Header(default=None),
   304→        idempotency_key: str | None = Header(default=None),
   305→        x_mandate: str | None = Header(default=None),
   306→        # Day 1 Track D (V3 §13): per-txn device_id + user_id for UPI Circle
   307→        # mandates (NPCI OC-201B §3.7 Issuer Bank duty + §3.3 Secondary PSP
   308→        # duty). Both default to None and are ignored for cod_order mandates.
   309→        x_device_id: str | None = Header(default=None),
   310→        x_user_id: str | None = Header(default=None),
   311→    ) -> dict:
   312→        token = bearer_token(authorization)
   313→        ok, err = check_key(token, "scorer", state["keys"])
   314→        if not ok:
   315→            raise HTTPException(status_code=401, detail=err)
   316→        client = token
   317→        if not state["bucket"].allow(client):
   318→            raise HTTPException(status_code=429, detail="rate limit exceeded")
   319→
   320→        # Day 2 Track E — idempotency cache (§A item 2: was an unbounded dict;
   321→        # now TTLCache in file mode, Postgres ``idempotency_keys`` table in
   322→        # db mode). File mode preserves the original (key, body) tuple key
   323→        # so two different request bodies with the same Idempotency-Key
   324→        # don't silently cross-pollute. Postgres mode uses the header value
   325→        # alone as the PK (the standard idempotency contract — same key,
   326→        # same response).
   327→        if settings.is_postgres:
   328→            cached = _idem_lookup_postgres(state, idempotency_key)
   329→            if cached is not None:
   330→                # 1% probabilistic cleanup of expired rows.
   331→                if (uuid.uuid4().int % 100) == 0:
   332→                    _idem_cleanup_postgres(state)
   333→                return dict(cached, replayed=True)
   334→        else:
   335→            cache_key = (idempotency_key or "", order.model_dump_json())
   336→            if idempotency_key and cache_key in state["idem"]:
   337→                return dict(state["idem"][cache_key], replayed=True)
   338→
   339→        try:
   340→            t0 = time.monotonic()
   341→            # Mandate verification (cheap; always run; enforcement happens
   342→            # later). Day 1 Track D extended the signature to pass device_id
   343→            # + user_id so UPI Circle mandates can enforce OC-201B §3.3/§3.7
   344→            # per-txn identity validation. cod_order mandates ignore both.
   345→            mandate_verdict, mandate_payload = verify_mandate(
   346→                x_mandate,
   347→                order.amount_inr,
   348→                device_id=x_device_id,
   349→                user_id=x_user_id,
   350→            )
   351→            fired = state["rules"].evaluate(order.model_dump())
   352→            rule_fired = fired.rule_id if fired else None
   353→
   354→            proba: float | None = None
   355→            reasons: list[dict] = []
   356→            degraded = False
   357→            breach_note: str | None = None
   358→            cost_breakdown: dict | None = None
   359→            decision: str | None = None
   360→            # ``decision_source`` records which layer actually chose the
   361→            # decision, for the audit trail + dashboard explainability.
   362→            # One of: rules_engine_block | mandate_breach | mandate_invalid
   363→            # | mandate_review_required | degraded_review | cost_optimal_bmr
   364→            # | cost_optimal_bmr_review_rule
   365→            decision_source: str
   366→
   367→            # ------------------------------------------------------------------
   368→            # Decision precedence (per Day 1 Track C spec):
   369→            #   1. Rules fast-path (BLOCK)         → short-circuit REJECT
   370→            #   2. Mandate enforcement (BREACH /
   371→            #      TAMPERED / EXPIRED w/ header)   → short-circuit REJECT
   372→            #   2c. Mandate REVIEW (UPI Circle 24h
   373→            #      cooling period, OC-201B)        → short-circuit REVIEW
   374→            #      (Day 1 Track D — mandate action-class expansion)
   375→            #   3. Circuit breaker OPEN            → degraded rules-only REVIEW
   376→            #   4. optimal_decision(p, ...)         → PRIMARY decision path
   377→            #      (replaces legacy static 0.15 / 0.60 thresholds)
   378→            # REVIEW rules don't short-circuit; they gate step 4 to never ACCEPT.
   379→            # ------------------------------------------------------------------
   380→            if fired is not None and fired.action == "BLOCK":
   381→                # 1. Rules-engine BLOCK short-circuits — no model invocation.
   382→                decision = "REJECT"
   383→                decision_source = "rules_engine_block"
   384→            elif mandate_verdict == MandateVerdict.BREACH:
   385→                # 2a. Mandate amount-breach short-circuits.
   386→                decision = "REJECT"
   387→                breach_note = "mandate_amount_breach"
   388→                decision_source = "mandate_breach"
   389→            elif x_mandate is not None and mandate_verdict == MandateVerdict.REVIEW:
   390→                # 2c. UPI Circle 24h cooling-period gate (OC-201B). The txn
   391→                # is permitted in principle; the cooling circuit requires a
   392→                # human to approve before the issuer debits. Routes to the
   393→                # case queue like other REVIEW decisions. verdict_reason
   394→                # (e.g. "cooling_period_active") flows into the audit trail
   395→                # for compliance traceability.
   396→                decision = "REVIEW"
   397→                breach_note = "mandate_review_required"
   398→                decision_source = "mandate_review_required"
   399→            elif x_mandate is not None and mandate_verdict != MandateVerdict.VALID:
   400→                # 2b. Tampered / expired mandate with a header present.
   401→                # Covers OC-201B 6-month inactivity auto-revoke (verdict_reason
   402→                # = "inactivity_auto_revoke") and standard TTL expiry.
   403→                decision = "REJECT"
   404→                breach_note = f"mandate_{mandate_verdict}"
   405→                decision_source = "mandate_invalid"
   406→            else:
   407→                # 3. Circuit breaker guards model invocation.
   408→                use_model = state["breaker"].allow_attempt()
   409→                if use_model:
   410→                    try:
   411→                        X, _ = build_feature_frame(to_frame(order), "order+addr")
   412→                        reasons = reason_codes_batch(
   413→                            state["model"],
   414→                            X,
   415→                            list(X.columns),
   416→                            state["base_rate"],
   417→                            state["reference"],
   418→                        )
   419→                        proba = float(state["model"].predict_proba(X)[0, 1])
   420→                        state["breaker"].record_success()
   421→                    except Exception:
   422→                        state["breaker"].record_failure()
   423→                        proba = None
   424→                if proba is None:
   425→                    # 3. Degraded path: model unavailable → rules-only REVIEW.
   426→                    degraded = True
   427→                    decision = "REVIEW"
   428→                    decision_source = "degraded_review"
   429→                else:
   430→                    # 4. Bahnsen Bayes Minimum Risk decision layer (ICMLA 2013,
   431→                    #    DOI 10.1109/ICMLA.2013.68). Per-order argmin of expected
   432→                    #    cost over {ACCEPT, REVIEW (selective-OTP), REJECT}.
   433→                    decision, costs = optimal_decision(proba, **DEFAULT_COST_WEIGHTS)
   434→                    cost_breakdown = costs
   435→                    # REVIEW rule gate: never ACCEPT when a REVIEW rule fired
   436→                    # (still uses BMR cost math, but the rule forces REVIEW).
   437→                    if fired is not None and fired.action == "REVIEW" and decision == "ACCEPT":
   438→                        decision = "REVIEW"
   439→                        decision_source = "cost_optimal_bmr_review_rule"
   440→                    else:
   441→                        decision_source = "cost_optimal_bmr"
   442→
   443→            # Legacy ``policy_hint`` is the BMR-optimal action string (kept for
   444→            # dashboard / consumer backward compat; the actual decision is the
   445→            # body's ``decision`` field — which equals ``policy_hint`` whenever
   446→            # the cost-optimizer was the decision source).
   447→            policy_hint = None
   448→            if proba is not None:

   830→    @app.get("/v1/policy/cost-curves")
   831→    def policy_cost_curves(
   832→        n_resamples: int = DEFAULT_COST_CURVE_RESAMPLES,
   833→        confidence: float = DEFAULT_COST_CURVE_CONFIDENCE,
   834→        authorization: str | None = Header(default=None),
   835→    ) -> dict:
   836→        """Drummond-Holte cost-curve explorer (Machine Learning 65:95-130, 2006,
   837→        DOI 10.1007/s10994-006-8199-5).
   838→
   839→        Returns a threshold sweep (0.05 → 0.95) over the labeled dataset that
   840→        the in-process model was trained on. For each threshold we report the
   841→        confusion counts, total Bahnsen cost (Eq. 1), precision, recall, plus
   842→        a 90% bootstrap CI on cost (≥500 resamples preserving row marginals —
   843→        per Drummond-Holte ``bootstrap_performance_ci`` capability).
   844→
   845→        The dashboard cost-curve explorer consumes this endpoint to render
   846→        real bars (replacing the previously-hardcoded ``COSTS`` array).
   847→
   848→        Auth: scorer scope. Returns 503 if the model isn't loaded or no
   849→        labeled data is available — never crashes the dashboard.
   850→        """
   851→        ok, err = check_key(bearer_token(authorization), "scorer", state["keys"])
   852→        if not ok:
   853→            raise HTTPException(status_code=401, detail=err)
   854→
   855→        # 1. Guard: model must be loaded (circuit breaker may be OPEN).
   856→        if state.get("model") is None:
   857→            raise HTTPException(
   858→                status_code=503,
   859→                detail="cost curves unavailable — model not loaded",
   860→            )
   861→        # 2. Guard: pre-computed labeled data must exist (startup warmup may
   862→        # have failed silently; surface a clear error to the dashboard).
   863→        cc = state.get("cost_curve")
   864→        if not cc or not cc.get("y_true") or not cc.get("probs"):
   865→            raise HTTPException(
   866→                status_code=503,
   867→                detail="cost curves unavailable — no labeled data available",
   868→            )
   869→        if n_resamples < 1 or n_resamples > 5000:
   870→            raise HTTPException(
   871→                status_code=422,
   872→                detail="n_resamples must be in [1, 5000]",
   873→            )
   874→        if not (0.5 < confidence < 0.999):
   875→            raise HTTPException(
   876→                status_code=422,
   877→                detail="confidence must be in (0.5, 0.999)",
   878→            )
   879→
   880→        y_true = cc["y_true"]
   881→        probs = cc["probs"]
   882→        weights = DEFAULT_COST_WEIGHTS
   883→        c_fp = float(weights["c_fp"])
   884→        c_fn = float(weights["c_fn"])
   885→
   886→        # 3. Threshold sweep → per-threshold confusion + cost + precision +
   887→        #    recall (pure-Python, O(N*T); ~140k ops for 7235 rows × 19 thr).
   888→        curves = cost_curve_sweep(
   889→            y_true=y_true,
   890→            probs=probs,
   891→            c_fp=c_fp,
   892→            c_fn=c_fn,
   893→        )
   894→
   895→        # 4. Bootstrap CI per threshold (Drummond-Holte row-marginal-preserving
   896→        #    resampling; ~3-5 sec for 500 resamples × 19 thr × 7235 samples).
   897→        try:
   898→            bootstrap_ci = bootstrap_cost_ci(
   899→                y_true=y_true,
   900→                probs=probs,
   901→                c_fp=c_fp,
   902→                c_fn=c_fn,
   903→                n_resamples=n_resamples,
   904→                confidence=confidence,
   905→                seed=42,
   906→            )
   907→        except Exception as e:  # pragma: no cover — defensive
   908→            # If bootstrap fails for any reason, return degenerate CIs and a
   909→            # warning rather than 500-ing the dashboard.
   910→            bootstrap_ci = {
   911→                str(r["threshold"]): {"low": 0.0, "high": 0.0, "mean": 0.0}
   912→                for r in curves
   913→            }
   914→            print(
   915→                f"cost-curves bootstrap degraded: {type(e).__name__}: {e}",
   916→                file=sys.stderr,
   917→            )
   918→
   919→        # 5. Optimal threshold = the one minimizing total cost (this is the
   920→        #    threshold analog of what ``optimal_decision()`` does per order).
   921→        optimal_record = min(curves, key=lambda r: r["cost"])
   922→        optimal_threshold = float(optimal_record["threshold"])
   923→
   924→        # 6. Cost crossover: if a challenger model is registered in the model
   925→        #    registry, find the threshold where it beats the incumbent. For
   926→        #    now there is one in-process model so this is a degenerate case —

  1199→def _idem_get_conn(state: dict):

  1210→    import psycopg

  1212→    conn = psycopg.connect(state["settings"].database_url, autocommit=False)
  1213→    state["_idem_conn"] = conn
  1214→    return conn

  1217→def _idem_lookup_postgres(state: dict, key: str | None) -> dict | None:
  1218→    """Read a cached idempotent response. Returns the body dict (parsed from
  1219→    the JSONB-stored response_body TEXT column) or None if not cached /
  1220→    expired. ``expires_at > NOW()`` is the TTL gate.
  1221→    """
  1222→    if not key:
  1223→        return None
  1224→    import json as _json
  1225→
  1226→    conn = _idem_get_conn(state)
  1227→    with conn.cursor() as cur:
  1228→        cur.execute(
  1229→            """
  1230→            SELECT response_body, status_code
  1231→              FROM idempotency_keys
  1232→             WHERE key = %s AND expires_at > NOW()
  1233→            """,
  1234→            (key,),
  1235→        )
  1236→        row = cur.fetchone()
  1237→    if row is None:
  1238→        return None
  1239→    body, _status = row
  1240→    return _json.loads(body)

  1243→def _idem_store_postgres(
  1244→    state: dict,
  1245→    key: str,
  1246→    request_body: str,
  1247→    response_body: dict,
  1248→    status_code: int,
  1249→) -> None:
  1250→    """INSERT an idempotency cache row. TTL = NOW() + settings.idem_ttl_seconds."""
  1251→    import json as _json
  1252→    from datetime import datetime, timedelta, timezone
  1253→
  1254→    conn = _idem_get_conn(state)
  1255→    expires = datetime.now(timezone.utc) + timedelta(seconds=state["settings"].idem_ttl_seconds)
  1256→    with conn.cursor() as cur:
  1257→        # ON CONFLICT DO UPDATE so a re-store of the same key (e.g. a slow
  1258→        # retry that lands after the original INSERT) refreshes expires_at
  1259→        # rather than erroring.
  1260→        cur.execute(
  1261→            """
  1262→            INSERT INTO idempotency_keys
  1263→              (key, request_body, response_body, status_code, expires_at)
  1264→            VALUES (%s, %s, %s, %s, %s)
  1265→            ON CONFLICT (key) DO UPDATE SET
  1266→              request_body  = EXCLUDED.request_body,
  1267→              response_body = EXCLUDED.response_body,
  1268→              status_code   = EXCLUDED.status_code,
  1269→              expires_at    = EXCLUDED.expires_at
  1270→            """,
  1271→            (key, request_body, _json.dumps(response_body), status_code, expires),
  1272→        )
  1273→        conn.commit()

  1276→def _idem_cleanup_postgres(state: dict) -> None:
  1277→    """Best-effort TTL cleanup. Called with 1% probability per request so the
  1278→    table doesn't grow forever under burst traffic.
  1279→    """
  1280→    conn = _idem_get_conn(state)
  1281→    try:
  1282→        with conn.cursor() as cur:
  1283→            cur.execute("DELETE FROM idempotency_keys WHERE expires_at < NOW()")
  1284→            conn.commit()
  1285→    except Exception:  # pragma: no cover — best-effort, never fail the request
  1286→        try:
  1287→            conn.rollback()
  1288→        except Exception:
  1289→            pass
