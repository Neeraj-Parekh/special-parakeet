   184→def create_app(
   185→    scorer_rate_per_min: int = 120, audit_path: str = "out/audit.jsonl"
   186→) -> FastAPI:
   187→    state: dict[str, Any] = {}
   188→
   189→    # Day 2 Track E — load Settings once at app-construction time. The
   190→    # ``database_url`` field selects between Postgres mode (full DB-backed
   191→    # audit/cases/registry/idempotency) and file-mode fallback (the 63
   192→    # existing tests run this way). The Settings object itself is cached
   193→    # via ``@lru_cache`` in ``src.config`` so we don't re-read env vars on
   194→    # every TestClient build.
   195→    settings = get_settings()
   196→    state["settings"] = settings
   197→
   198→    # Day 2 Track F — Redis Streams producer (fire-and-forget). Constructed
   199→    # eagerly here (cheap — just stores the URL; lazy-connects on first
   200→    # ``publish``) so the /risk/score handler can call it after the audit
   201→    # write + case open. When ``settings.redis_url`` is None (test mode,
   202→    # local dev without REDIS_URL) the producer is a no-op: ``publish()``
   203→    # returns None silently — the API response is unaffected. This is the
   204→    # pragmatic fire-and-forget contract; V3 §10.3 prescribes a full
   205→    # transactional outbox table drained by a worker (deferred; see
   206→    # worklog Day 2 Track F deferral list).
   207→    state["stream"] = StreamProducer(settings.redis_url)
   208→
   209→    # Day 2 Track G — LabelFeedbackService (DDM + ADWIN over the delayed
   210→    # is_returned label stream). Constructed eagerly (cheap — just stores
   211→    # the URLs + creates the DDM/ADWIN instances) so the
   212→    # /v1/feedback/ingest handler can call ``ingest_label`` per request +
   213→    # the /metrics endpoint can read ``current_state()`` for the drift
   214→    # gauges. The lazy StreamProducer inside the service is constructed
   215→    # only on first DRIFT publish — so the 67 existing tests + 7 feedback
   216→    # tests still pass without a Redis fixture.
   217→    state["feedback"] = LabelFeedbackService(
   218→        redis_url=settings.redis_url,
   219→        database_url=settings.database_url,
   220→    )
   221→
   222→    # Idempotency cache. File mode: TTLCache (bounded + auto-expiring —
   223→    # closes §A item 2 memory leak). Postgres mode: the ``idempotency_keys``
   224→    # table; ``state["idem"]`` is unused (kept as an empty dict for backward-
   225→    # compat with any test that asserts the key exists).
   226→    if settings.is_postgres:
   227→        state["idem"] = TTLCache(maxsize=1, ttl=3600)  # placeholder, real cache is the DB
   228→    else:
   229→        state["idem"] = TTLCache(
   230→            maxsize=settings.idem_maxsize, ttl=settings.idem_ttl_seconds
   231→        )
   232→
   233→    @asynccontextmanager
   234→    async def lifespan(_: FastAPI):
   235→        df = add_address_features(load_orders("data/raw/cod_orders.csv"))
   236→        train_df, _ = group_split(df)
   237→        X_tr, y_tr = build_feature_frame(train_df, "order+addr")
   238→        model_path = Path("out/model_api.joblib")
   239→        if not model_path.exists():
   240→            save_model(fit_model(X_tr, y_tr), str(model_path))
   241→        from src.models.train import load_model
   242→
   243→        state["model"] = load_model(str(model_path))
   244→        state["reference"] = X_tr.mode().iloc[0]
   245→        state["base_rate"] = float(y_tr.mean())
   246→        # Day 2 Track E — audit path comes from Settings (defaults to the
   247→        # same ``out/audit.jsonl`` as before, but now .env-configurable +
   248→        # docker-compose-wired). The AuditLogger is dual-mode internally.
   249→        state["audit"] = AuditLogger(audit_path or settings.audit_path)
   250→        # Pass Settings-backed CSV keys to default_keys so the .env file is
   251→        # honored (Track B's Dockerfile change removed the baked ENV defaults;
   252→        # docker-compose now sets them via the environment block).
   253→        state["keys"] = default_keys(
   254→            scorer_keys=settings.rto_scorer_keys,
   255→            admin_keys=settings.rto_admin_keys,
   256→        )
   257→        state["bucket"] = TokenBucket(scorer_rate_per_min)
   258→        state["rules"] = RulesEngine()
   259→        state["breaker"] = CircuitBreaker()
   260→        state["metrics"] = Metrics()
   261→        state["cases"] = CaseService(settings.cases_path)
   262→        num_cols = [c for c in X_tr.columns if str(X_tr[c].dtype) != "category"]
   263→        state["psi_sample"] = {
   264→            c: X_tr[c].dropna().sample(n=min(2000, len(X_tr)), random_state=7).tolist()
   265→            for c in num_cols
   266→        }
   267→        # Day 2 Track E — §A item 4 (register_model dead in prod). Register
   268→        # the in-process HistGB with the model registry on every worker boot.
   269→        # File mode: this writes to ``out/model_registry.json`` (the existing
   270→        # behaviour — no-op for tests since they call register_model directly
   271→        # via tmp_path). Postgres mode: INSERT into the ``model_registry``
   272→        # table; the partial-unique index enforces single-champion. PR-AUC
   273→        # + ROC-AUC are surfaced as the metrics payload so the model-card
   274→        # + drift endpoints can read them.
   275→        if settings.is_postgres:
   276→            try:
   277→                from sklearn.metrics import average_precision_score, roc_auc_score
   278→                _p = state["model"].predict_proba(X_tr)[:, 1]
   279→                _pr = float(average_precision_score(y_tr, _p))
   280→                _roc = float(roc_auc_score(y_tr, _p))
   281→                register_model(
   282→                    version=f"v{int(time.time())}",
   283→                    model_path=str(model_path),
   284→                    metrics={"pr_auc": _pr, "roc_auc": _roc},
   285→                    champion=True,
   286→                )
   287→            except Exception as e:  # pragma: no cover — startup-only, defensive
   288→                print(
   289→                    f"model-registry registration skipped: "
   290→                    f"{type(e).__name__}: {e}",
   291→                    file=sys.stderr,
   292→                )
   293→        # Pre-compute labeled (y, p) sequences for the /v1/policy/cost-curves
   294→        # endpoint (Drummond & Holte 2006). The training set carries ground-
   295→        # truth is_returned labels; running the model over it once at startup
   296→        # lets the endpoint respond in O(1) for the per-threshold sweep and
   297→        # ~3-5 sec for the 500-resample bootstrap (per-row performance only).
   298→        # Note: this is in-sample — Day 2 Track E + G will swap in a held-out
   299→        # test slice + delayed-label feedback so the curve reflects deployment
   300→        # conditions, not training fit.
   301→        try:
   302→            _cost_curve_p = state["model"].predict_proba(X_tr)[:, 1].tolist()
   303→            _cost_curve_y = [int(v) for v in y_tr.tolist()]
   304→            state["cost_curve"] = {
   305→                "y_true": _cost_curve_y,
   306→                "probs": _cost_curve_p,
   307→                "n": len(_cost_curve_y),
   308→                "source": "train_df_in_sample",
   309→            }
   310→        except Exception as e:  # pragma: no cover — startup-only, defensive
   311→            print(f"cost-curve warmup skipped: {type(e).__name__}: {e}", file=sys.stderr)
   312→            state["cost_curve"] = None
   313→        yield
   314→        # Day 2 Track E — close the model-registry connection at shutdown so
   315→        # the worker doesn't leak a Postgres connection across hot-reloads.
   316→        if settings.is_postgres:
   317→            _close_registry_conn()
   318→        # Day 2 Track G — close the LabelFeedbackService's lazy StreamProducer
   319→        # so the worker doesn't leak a Redis connection across hot-reloads.
   320→        # Safe to call on a service whose producer was never constructed.
   321→        try:
   322→            state["feedback"].close()
   323→        except Exception:  # pragma: no cover — best-effort shutdown
   324→            pass
   325→
   326→    app = FastAPI(title="RTO Trust Layer", version="0.2.0", lifespan=lifespan)
   327→    app.state.core = state

   346→    @app.post("/risk/score")
   347→    def score(
   348→        order: OrderIn,
   349→        authorization: str | None = Header(default=None),
   350→        idempotency_key: str | None = Header(default=None),
   351→        x_mandate: str | None = Header(default=None),
   352→        # Day 1 Track D (V3 §13): per-txn device_id + user_id for UPI Circle
   353→        # mandates (NPCI OC-201B §3.7 Issuer Bank duty + §3.3 Secondary PSP
   354→        # duty). Both default to None and are ignored for cod_order mandates.
   355→        x_device_id: str | None = Header(default=None),
   356→        x_user_id: str | None = Header(default=None),
   357→    ) -> dict:
   358→        token = bearer_token(authorization)
   359→        ok, err = check_key(token, "scorer", state["keys"])
   360→        if not ok:
   361→            raise HTTPException(status_code=401, detail=err)
   362→        client = token
   363→        if not state["bucket"].allow(client):
   364→            raise HTTPException(status_code=429, detail="rate limit exceeded")
   365→
   366→        # Day 2 Track E — idempotency cache (§A item 2: was an unbounded dict;
   367→        # now TTLCache in file mode, Postgres ``idempotency_keys`` table in
   368→        # db mode). File mode preserves the original (key, body) tuple key
   369→        # so two different request bodies with the same Idempotency-Key
   370→        # don't silently cross-pollute. Postgres mode uses the header value
   371→        # alone as the PK (the standard idempotency contract — same key,
   372→        # same response).
   373→        if settings.is_postgres:
   374→            cached = _idem_lookup_postgres(state, idempotency_key)
   375→            if cached is not None:
   376→                # 1% probabilistic cleanup of expired rows.
   377→                if (uuid.uuid4().int % 100) == 0:
   378→                    _idem_cleanup_postgres(state)
   379→                return dict(cached, replayed=True)
   380→        else:
   381→            cache_key = (idempotency_key or "", order.model_dump_json())
   382→            if idempotency_key and cache_key in state["idem"]:
   383→                return dict(state["idem"][cache_key], replayed=True)
   384→
   385→        try:
   386→            t0 = time.monotonic()
   387→            # Mandate verification (cheap; always run; enforcement happens
   388→            # later). Day 1 Track D extended the signature to pass device_id
   389→            # + user_id so UPI Circle mandates can enforce OC-201B §3.3/§3.7
   390→            # per-txn identity validation. cod_order mandates ignore both.
   391→            mandate_verdict, mandate_payload = verify_mandate(
   392→                x_mandate,
   393→                order.amount_inr,
   394→                device_id=x_device_id,
   395→                user_id=x_user_id,
   396→            )
   397→            fired = state["rules"].evaluate(order.model_dump())
   398→            rule_fired = fired.rule_id if fired else None
   399→
   400→            proba: float | None = None
   401→            reasons: list[dict] = []
   402→            degraded = False
   403→            breach_note: str | None = None
   404→            cost_breakdown: dict | None = None
   405→            decision: str | None = None
   406→            # ``decision_source`` records which layer actually chose the
   407→            # decision, for the audit trail + dashboard explainability.
   408→            # One of: rules_engine_block | mandate_breach | mandate_invalid
   409→            # | mandate_review_required | degraded_review | cost_optimal_bmr
   410→            # | cost_optimal_bmr_review_rule
   411→            decision_source: str
   412→
   413→            # ------------------------------------------------------------------
   414→            # Decision precedence (per Day 1 Track C spec):
   415→            #   1. Rules fast-path (BLOCK)         → short-circuit REJECT
   416→            #   2. Mandate enforcement (BREACH /
   417→            #      TAMPERED / EXPIRED w/ header)   → short-circuit REJECT
   418→            #   2c. Mandate REVIEW (UPI Circle 24h
   419→            #      cooling period, OC-201B)        → short-circuit REVIEW
   420→            #      (Day 1 Track D — mandate action-class expansion)
   421→            #   3. Circuit breaker OPEN            → degraded rules-only REVIEW
   422→            #   4. optimal_decision(p, ...)         → PRIMARY decision path
   423→            #      (replaces legacy static 0.15 / 0.60 thresholds)
   424→            # REVIEW rules don't short-circuit; they gate step 4 to never ACCEPT.
   425→            # ------------------------------------------------------------------
   426→            if fired is not None and fired.action == "BLOCK":
   427→                # 1. Rules-engine BLOCK short-circuits — no model invocation.
   428→                decision = "REJECT"
   429→                decision_source = "rules_engine_block"
   430→            elif mandate_verdict == MandateVerdict.BREACH:
   431→                # 2a. Mandate amount-breach short-circuits.
   432→                decision = "REJECT"
   433→                breach_note = "mandate_amount_breach"
   434→                decision_source = "mandate_breach"
   435→            elif x_mandate is not None and mandate_verdict == MandateVerdict.REVIEW:
   436→                # 2c. UPI Circle 24h cooling-period gate (OC-201B). The txn
   437→                # is permitted in principle; the cooling circuit requires a
   438→                # human to approve before the issuer debits. Routes to the
   439→                # case queue like other REVIEW decisions. verdict_reason
   440→                # (e.g. "cooling_period_active") flows into the audit trail
   441→                # for compliance traceability.
   442→                decision = "REVIEW"
   443→                breach_note = "mandate_review_required"
   444→                decision_source = "mandate_review_required"
   445→            elif x_mandate is not None and mandate_verdict != MandateVerdict.VALID:
   446→                # 2b. Tampered / expired mandate with a header present.
   447→                # Covers OC-201B 6-month inactivity auto-revoke (verdict_reason
   448→                # = "inactivity_auto_revoke") and standard TTL expiry.
   449→                decision = "REJECT"
   450→                breach_note = f"mandate_{mandate_verdict}"
   451→                decision_source = "mandate_invalid"
   452→            else:
   453→                # 3. Circuit breaker guards model invocation.
   454→                use_model = state["breaker"].allow_attempt()
   455→                if use_model:
   456→                    try:
   457→                        X, _ = build_feature_frame(to_frame(order), "order+addr")
   458→                        reasons = reason_codes_batch(
   459→                            state["model"],
   460→                            X,
   461→                            list(X.columns),
   462→                            state["base_rate"],
   463→                            state["reference"],
   464→                        )
   465→                        proba = float(state["model"].predict_proba(X)[0, 1])
   466→                        state["breaker"].record_success()
   467→                    except Exception:
   468→                        state["breaker"].record_failure()
   469→                        proba = None
   470→                if proba is None:
   471→                    # 3. Degraded path: model unavailable → rules-only REVIEW.
   472→                    degraded = True
   473→                    decision = "REVIEW"
   474→                    decision_source = "degraded_review"
   475→                else:
   476→                    # 4. Bahnsen Bayes Minimum Risk decision layer (ICMLA 2013,
   477→                    #    DOI 10.1109/ICMLA.2013.68). Per-order argmin of expected
   478→                    #    cost over {ACCEPT, REVIEW (selective-OTP), REJECT}.
   479→                    decision, costs = optimal_decision(proba, **DEFAULT_COST_WEIGHTS)
   480→                    cost_breakdown = costs
   481→                    # REVIEW rule gate: never ACCEPT when a REVIEW rule fired
   482→                    # (still uses BMR cost math, but the rule forces REVIEW).
   483→                    if fired is not None and fired.action == "REVIEW" and decision == "ACCEPT":
   484→                        decision = "REVIEW"
   485→                        decision_source = "cost_optimal_bmr_review_rule"
   486→                    else:
   487→                        decision_source = "cost_optimal_bmr"
   488→
   489→            # Legacy ``policy_hint`` is the BMR-optimal action string (kept for
   490→            # dashboard / consumer backward compat; the actual decision is the
   491→            # body's ``decision`` field — which equals ``policy_hint`` whenever
   492→            # the cost-optimizer was the decision source).
   493→            policy_hint = None
   494→            if proba is not None:
   495→                policy_hint = optimal_decision(proba, **DEFAULT_COST_WEIGHTS)[0]
   496→
   497→            features_used = (
   498→                {c: float(X.iloc[0][c]) for c in X.columns if str(X[c].dtype) != "category"}
   499→                if proba is not None
   500→                else {}
   501→            )
   502→            # Day 2 Track F — generate the prediction_id ONCE so the case
   503→            # row, the audit record's stream publish, and the response body
   504→            # all share the same identifier. Track B's original code passed
   505→            # ``prediction_id="pending"`` to ``open_case`` and generated the
   506→            # real UUID inline in the body dict — that inconsistency meant
   507→            # stream consumers couldn't correlate a ``risk.scores`` message
   508→            # to a ``cases.created`` message (different prediction_id on
   509→            # each). Now the UUID is canonical across the case, the stream,
   510→            # and the response.
   511→            prediction_id = str(uuid.uuid4())
   512→            case_id = None
   513→            if decision == "REVIEW":
   514→                case_id = state["cases"].open_case(
   515→                    prediction_id=prediction_id, order_id=order.order_id,
   516→                    reason=(
   517→                        f"mandate:{mandate_payload.get('verdict_reason')}"
   518→                        if decision_source == "mandate_review_required"
   519→                        else "review_gate" if fired is None
   520→                        else f"rule:{fired.rule_id}"
   521→                    ),
   522→                )
   523→            # Audit payload (Day 1 Track D — V3 §13 mandate action-class
   524→            # expansion). The new fields surface OC-201B compliance metadata
   525→            # in the tamper-evident hash chain: ``mandate_type``,
   526→            # ``bh_purpose_code``, ``device_id``, ``user_id``, and the
   527→            # machine-readable ``mandate_verdict_reason``. The audit logger
   528→            # (Track E Day 2) stores arbitrary JSON payloads; we just pass
   529→            # the new keys through — no logger.py modification needed.
   530→            audit_id = state["audit"].log(
   531→                {
   532→                    "request": {
   533→                        **order.model_dump(),
   534→                        "customer_id": redact_customer(order.customer_id),
   535→                    },
   536→                    "probability": round(proba, 5) if proba is not None else None,
   537→                    "decision": decision,
   538→                    "decision_source": decision_source,
   539→                    "cost_breakdown": cost_breakdown,
   540→                    "reason_codes": reasons[:5],
   541→                    "mandate_verdict": mandate_verdict,
   542→                    "mandate_verdict_reason": mandate_payload.get("verdict_reason"),
   543→                    "mandate_type": mandate_payload.get("mandate_type"),
   544→                    "bh_purpose_code": mandate_payload.get("bh_purpose_code"),
   545→                    "device_id": x_device_id,
   546→                    "user_id": x_user_id,
   547→                    "breach_note": breach_note,
   548→                    "rule_fired": rule_fired,
   549→                    "degraded": degraded,
   550→                    "features_used": features_used,
   551→                    "latency_ms": now_ms(t0),
   552→                    # Day 2 Track G — store the prediction_id in the audit
   553→                    # body so /v1/feedback/ingest can look up the predicted
   554→                    # P(RTO) for a given prediction_id (file mode:
   555→                    # AuditLogger.tail() scan; Postgres mode:
   556→                    # jsonb_path_query on the audit_records.body column).
   557→                    # Additive — doesn't touch the decision logic, just
   558→                    # enriches the audit record with the canonical
   559→                    # correlation key the feedback endpoint needs.
   560→                    "prediction_id": prediction_id,
   561→                    "case_id": case_id,
   562→                }
   563→            )
   564→            # Day 2 Track F — fire-and-forget Redis Streams publishes. After
   565→            # the audit hash-chain append + case open, publish to:
   566→            #   * ``risk.scores`` — every decision (the streaming-processor
   567→            #     worker consumes this for TFX-style ``generate_data_statistics``
   568→            #     + anomaly → ``model.drift``)
   569→            #   * ``audit.records`` — every audit record (audit-tail worker
   570→            #     for fan-out to dashboard / compliance export — Track I/H)
   571→            #   * ``cases.created`` — REVIEW decisions only (case-queue
   572→            #     worker for SLA timers + assignment — Track H)
   573→            # All three are fire-and-forget: ``StreamProducer.publish``
   574→            # returns None silently if REDIS_URL is unset (test mode) or
   575→            # Redis is down. The response body is unaffected. The full
   576→            # transactional outbox (V3 §10.3) is deferred — see worklog.
   577→            state["stream"].publish(
   578→                STREAM_RISK_SCORES,
   579→                {
   580→                    "prediction_id": prediction_id,
   581→                    "order_id": order.order_id,
   582→                    "decision": decision or "",
   583→                    "score": "" if proba is None else f"{float(proba):.6f}",
   584→                    "decision_source": decision_source,
   585→                    "model_version": state["audit"].model_version,
   586→                    "ts": datetime.now(timezone.utc).isoformat(),
   587→                },
   588→            )
   589→            state["stream"].publish(
   590→                STREAM_AUDIT_RECORDS,
   591→                {
   592→                    "audit_id": audit_id,
   593→                    "prediction_id": prediction_id,
   594→                    "decision": decision or "",
   595→                    "ts": datetime.now(timezone.utc).isoformat(),
   596→                },
   597→            )
   598→            if decision == "REVIEW" and case_id is not None:
   599→                state["stream"].publish(
   600→                    STREAM_CASES_CREATED,
   601→                    {
   602→                        "case_id": case_id,
   603→                        "prediction_id": prediction_id,
   604→                        "order_id": order.order_id,
   605→                        "reason": (
   606→                            f"mandate:{mandate_payload.get('verdict_reason')}"
   607→                            if decision_source == "mandate_review_required"
   608→                            else "review_gate" if fired is None
   609→                            else f"rule:{fired.rule_id}"
   610→                        ),
   611→                        "ts": datetime.now(timezone.utc).isoformat(),
   612→                    },
   613→                )
   614→            body = {
   615→                "prediction_id": prediction_id,
   616→                "risk_score": round(proba * 100, 1) if proba is not None else None,
   617→                "probability": round(proba, 4) if proba is not None else None,
   618→                "decision": decision,
   619→                # ``gate_thresholds`` now reports the policy in force. Legacy
   620→                # accept_below / reject_above kept for backward compat — but
   621→                # the live decision path no longer consults them; it consults
   622→                # ``optimal_decision(p, weights=...)``. The dashboard reads
   623→                # ``policy`` to label the decision card.
   624→                "gate_thresholds": {
   625→                    "policy": "cost_optimal_bmr",
   626→                    "weights": DEFAULT_COST_WEIGHTS,
   627→                    "legacy_accept_t": ACCEPT_T,
   628→                    "legacy_reject_t": REJECT_T,
   629→                },
   630→                "decision_source": decision_source,
   631→                "cost_breakdown": cost_breakdown,
   632→                "explanation": reasons[:5],
   633→                "rule_fired": rule_fired,
   634→                "degraded": degraded,
   635→                "policy_hint": policy_hint,
   636→                "model_version": "rules_only"
   637→                if degraded
   638→                else (current_champion() or {"version": state["audit"].model_version})["version"],
   639→                "latency_ms": now_ms(t0),
   640→                "case_id": case_id,
   641→                "mandate": {
   642→                    "verdict": mandate_verdict,
   643→                    "note": breach_note,
   644→                    # Day 1 Track D — V3 §13: surface OC-201B compliance metadata
   645→                    # to the agent / dashboard so the consumer can see WHY
   646→                    # the mandate verdict landed (e.g. "cooling_period_active"
   647→                    # vs "device_id_not_allowed") and WHICH mandate type fired.
   648→                    "verdict_reason": mandate_payload.get("verdict_reason"),
   649→                    "mandate_type": mandate_payload.get("mandate_type"),
   650→                    "bh_purpose_code": mandate_payload.get("bh_purpose_code"),
   651→                },
   652→                "audit_trail_url": f"/audit/{audit_id}",
   653→                "timestamp": datetime.now(timezone.utc).isoformat(),
   654→            }
   655→            state["metrics"].inc(
   656→                "risk_decisions_total", {"decision": decision, "degraded": str(degraded)}
   657→            )
   658→            state["metrics"].observe_latency(now_ms(t0) / 1000)
   659→            # Day 2 Track E — store the response in the idempotency cache.
   660→            # File mode: TTLCache (auto-expires after settings.idem_ttl_seconds;
   661→            # bounded to settings.idem_maxsize). Postgres mode: INSERT into
   662→            # ``idempotency_keys`` with expires_at = NOW() + ttl.
   663→            if idempotency_key:
   664→                if settings.is_postgres:
   665→                    _idem_store_postgres(
   666→                        state,
   667→                        idempotency_key,
   668→                        order.model_dump_json(),
   669→                        body,
   670→                        200,
   671→                    )
   672→                else:
   673→                    cache_key = (idempotency_key, order.model_dump_json())
   674→                    state["idem"][cache_key] = body
   675→            return body
   676→        except HTTPException:
   677→            raise
   678→        except Exception as e:  # no internal detail leakage
   679→            incident = uuid.uuid4()
   680→            print(f"incident={incident} scoring_failed={type(e).__name__}: {e}", file=sys.stderr)
   681→            raise HTTPException(status_code=500, detail=f"internal_error incident={incident}")

   876→    @app.get("/v1/policy/cost-curves")
   877→    def policy_cost_curves(
   878→        n_resamples: int = DEFAULT_COST_CURVE_RESAMPLES,
   879→        confidence: float = DEFAULT_COST_CURVE_CONFIDENCE,
   880→        authorization: str | None = Header(default=None),
   881→    ) -> dict:
   882→        """Drummond-Holte cost-curve explorer (Machine Learning 65:95-130, 2006,
   883→        DOI 10.1007/s10994-006-8199-5).
   884→
   885→        Returns a threshold sweep (0.05 → 0.95) over the labeled dataset that
   886→        the in-process model was trained on. For each threshold we report the
   887→        confusion counts, total Bahnsen cost (Eq. 1), precision, recall, plus
   888→        a 90% bootstrap CI on cost (≥500 resamples preserving row marginals —
   889→        per Drummond-Holte ``bootstrap_performance_ci`` capability).
   890→
   891→        The dashboard cost-curve explorer consumes this endpoint to render
   892→        real bars (replacing the previously-hardcoded ``COSTS`` array).
   893→
   894→        Auth: scorer scope. Returns 503 if the model isn't loaded or no
   895→        labeled data is available — never crashes the dashboard.
   896→        """
   897→        ok, err = check_key(bearer_token(authorization), "scorer", state["keys"])
   898→        if not ok:
   899→            raise HTTPException(status_code=401, detail=err)
   900→
   901→        # 1. Guard: model must be loaded (circuit breaker may be OPEN).
   902→        if state.get("model") is None:
   903→            raise HTTPException(
   904→                status_code=503,
   905→                detail="cost curves unavailable — model not loaded",
   906→            )
   907→        # 2. Guard: pre-computed labeled data must exist (startup warmup may
   908→        # have failed silently; surface a clear error to the dashboard).
   909→        cc = state.get("cost_curve")
   910→        if not cc or not cc.get("y_true") or not cc.get("probs"):
   911→            raise HTTPException(
   912→                status_code=503,
   913→                detail="cost curves unavailable — no labeled data available",
   914→            )
   915→        if n_resamples < 1 or n_resamples > 5000:
   916→            raise HTTPException(
   917→                status_code=422,

   929→        c_fp = float(weights["c_fp"])
   930→        c_fn = float(weights["c_fn"])

   962→                file=sys.stderr,

   966→        #    threshold analog of what ``optimal_decision()`` does per order).

   970→        # 6. Cost crossover: if a challenger model is registered in the model

  1532→    def usage(
  1533→        authorization: str | None = Header(default=None),
  1534→        since_hours: str = "24,168,720",
  1535→    ) -> dict:
  1536→        """Per-merchant request counts (Day 2 Track H — closes §A item 15 +
  1537→        §C T10; metering endpoint for billing / quota enforcement).
  1538→
  1539→        Auth: admin scope. Returns audit-record counts for each window
  1540→        in ``since_hours`` (default ``24,168,720`` = 24h / 7d / 30d).
  1541→        Per-window = count of audit_records with ``created_at > now() -
  1542→        interval '<H> hours'`` in Postgres mode, or a JSONL timestamp
  1543→        scan in file mode.
  1544→
  1545→        Note: multi-tenant merchant_id is not yet implemented — the
  1546→        counts are aggregate (all merchants combined). The
  1547→        ``merchant_id`` column on audit_records.body is the future
  1548→        filter key (deferred — Track E's schema migration 001 already
  1549→        has the JSONB column ready; a per-merchant GROUP BY is a
  1550→        one-line change once the merchant_id header is wired into
  1551→        /risk/score).
  1552→
  1553→        The response also surfaces the Merkle interval sealing cadence
  1554→        (last 100 intervals) so a billing auditor can verify the audit
  1555→        trail's tamper-evidence layer is up-to-date alongside the
  1556→        metering counts.

  1558→        ok, err = check_key(bearer_token(authorization), "admin", state["keys"])
  1559→        if not ok:
  1560→            raise HTTPException(status_code=401, detail=err)
  1561→        # Parse + clamp the since_hours CSV. Each value must be a positive
  1562→        # int (hours); we cap at 87600 (10 years) so a stray huge value
  1563→        # doesn't make the Postgres interval scan pathological.
  1564→        try:
  1565→            hours = tuple(
  1566→                max(1, min(87600, int(h.strip())))
  1567→                for h in since_hours.split(",")
  1568→                if h.strip()
  1569→            )
  1570→        except ValueError:
  1571→            raise HTTPException(
  1572→                status_code=422,
  1573→                detail="since_hours must be a CSV of positive integers (e.g. '24,168,720')",

  1583→        intervals = state["audit"].merkle_intervals(limit=100)

  1593→        except (ValueError, TypeError):

  1597→            "since_hours": list(hours),
  1598→            "intervals_sealed_total": len(intervals),

  1600→            "latest_interval": intervals[0] if intervals else None,

  1643→def _idem_get_conn(state: dict):
  1644→    """Lazily-open a dedicated psycopg connection for the idempotency cache.
  1645→
  1646→    Separate from the AuditLogger / CaseService / model-registry connections
  1647→    so a slow idempotency lookup can't block an audit write. The connection
  1648→    is cached on ``state`` so we don't pay the connect-handshake cost on
  1649→    every request.
  1650→    """
  1651→    conn = state.get("_idem_conn")
  1652→    if conn is not None:
  1653→        return conn
  1654→    import psycopg
  1655→
  1656→    conn = psycopg.connect(state["settings"].database_url, autocommit=False)
  1657→    state["_idem_conn"] = conn
  1658→    return conn

  1661→def _idem_lookup_postgres(state: dict, key: str | None) -> dict | None:
  1662→    """Read a cached idempotent response. Returns the body dict (parsed from
  1663→    the JSONB-stored response_body TEXT column) or None if not cached /
  1664→    expired. ``expires_at > NOW()`` is the TTL gate.
  1665→    """
  1666→    if not key:
  1667→        return None
  1668→    import json as _json
  1669→
  1670→    conn = _idem_get_conn(state)
  1671→    with conn.cursor() as cur:
  1672→        cur.execute(
  1673→            """
  1674→            SELECT response_body, status_code
  1675→              FROM idempotency_keys
  1676→             WHERE key = %s AND expires_at > NOW()
  1677→            """,
  1678→            (key,),
  1679→        )
  1680→        row = cur.fetchone()
  1681→    if row is None:
  1682→        return None
  1683→    body, _status = row
  1684→    return _json.loads(body)

  1687→def _idem_store_postgres(
  1688→    state: dict,
  1689→    key: str,
  1690→    request_body: str,
  1691→    response_body: dict,
  1692→    status_code: int,
  1693→) -> None:
  1694→    """INSERT an idempotency cache row. TTL = NOW() + settings.idem_ttl_seconds."""
  1695→    import json as _json
  1696→    from datetime import datetime, timedelta, timezone
  1697→
  1698→    conn = _idem_get_conn(state)
  1699→    expires = datetime.now(timezone.utc) + timedelta(seconds=state["settings"].idem_ttl_seconds)
  1700→    with conn.cursor() as cur:
  1701→        # ON CONFLICT DO UPDATE so a re-store of the same key (e.g. a slow
  1702→        # retry that lands after the original INSERT) refreshes expires_at
  1703→        # rather than erroring.
  1704→        cur.execute(
  1705→            """
  1706→            INSERT INTO idempotency_keys
  1707→              (key, request_body, response_body, status_code, expires_at)
  1708→            VALUES (%s, %s, %s, %s, %s)
  1709→            ON CONFLICT (key) DO UPDATE SET
  1710→              request_body  = EXCLUDED.request_body,
  1711→              response_body = EXCLUDED.response_body,
  1712→              status_code   = EXCLUDED.status_code,
  1713→              expires_at    = EXCLUDED.expires_at
  1714→            """,
  1715→            (key, request_body, _json.dumps(response_body), status_code, expires),
  1716→        )
  1717→        conn.commit()

  1720→def _idem_cleanup_postgres(state: dict) -> None:
  1721→    """Best-effort TTL cleanup. Called with 1% probability per request so the
  1722→    table doesn't grow forever under burst traffic.
  1723→    """
  1724→    conn = _idem_get_conn(state)
  1725→    try:
  1726→        with conn.cursor() as cur:
  1727→            cur.execute("DELETE FROM idempotency_keys WHERE expires_at < NOW()")
  1728→            conn.commit()
  1729→    except Exception:  # pragma: no cover — best-effort, never fail the request
  1730→        try:
  1731→            conn.rollback()
  1732→        except Exception:
  1733→            pass
