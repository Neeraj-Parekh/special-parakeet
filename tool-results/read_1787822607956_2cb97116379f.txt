   193→def create_app(
   194→    scorer_rate_per_min: int = 120, audit_path: str = "out/audit.jsonl"
   195→) -> FastAPI:
   196→    state: dict[str, Any] = {}
   197→
   198→    # Day 2 Track E — load Settings once at app-construction time. The
   199→    # ``database_url`` field selects between Postgres mode (full DB-backed
   200→    # audit/cases/registry/idempotency) and file-mode fallback (the 63
   201→    # existing tests run this way). The Settings object itself is cached
   202→    # via ``@lru_cache`` in ``src.config`` so we don't re-read env vars on
   203→    # every TestClient build.
   204→    settings = get_settings()
   205→    state["settings"] = settings
   206→
   207→    # Day 2 Track F — Redis Streams producer (fire-and-forget). Constructed
   208→    # eagerly here (cheap — just stores the URL; lazy-connects on first
   209→    # ``publish``) so the /risk/score handler can call it after the audit
   210→    # write + case open. When ``settings.redis_url`` is None (test mode,
   211→    # local dev without REDIS_URL) the producer is a no-op: ``publish()``
   212→    # returns None silently — the API response is unaffected. This is the
   213→    # pragmatic fire-and-forget contract; V3 §10.3 prescribes a full
   214→    # transactional outbox table drained by a worker (deferred; see
   215→    # worklog Day 2 Track F deferral list).
   216→    state["stream"] = StreamProducer(settings.redis_url)
   217→
   218→    # Day 2 Track G — LabelFeedbackService (DDM + ADWIN over the delayed
   219→    # is_returned label stream). Constructed eagerly (cheap — just stores
   220→    # the URLs + creates the DDM/ADWIN instances) so the
   221→    # /v1/feedback/ingest handler can call ``ingest_label`` per request +
   222→    # the /metrics endpoint can read ``current_state()`` for the drift
   223→    # gauges. The lazy StreamProducer inside the service is constructed
   224→    # only on first DRIFT publish — so the 67 existing tests + 7 feedback
   225→    # tests still pass without a Redis fixture.
   226→    state["feedback"] = LabelFeedbackService(
   227→        redis_url=settings.redis_url,
   228→        database_url=settings.database_url,
   229→    )
   230→
   231→    # Day 4 Track M — OpenTelemetry tracer. Dual-mode like Track E's
   232→    # DATABASE_URL + Track F's REDIS_URL: setup_otel() returns None when
   233→    # OTEL_EXPORTER_OTLP_ENDPOINT is unset (test mode, local dev without
   234→    # Jaeger). The /risk/score handler checks ``state["tracer"] is not
   235→    # None`` before starting a span so the disabled-mode path is a no-op.
   236→    # The 93 existing tests pass without Jaeger.
   237→    state["tracer"] = setup_otel()
   238→
   239→    # Idempotency cache. File mode: TTLCache (bounded + auto-expiring —
   240→    # closes §A item 2 memory leak). Postgres mode: the ``idempotency_keys``
   241→    # table; ``state["idem"]`` is unused (kept as an empty dict for backward-
   242→    # compat with any test that asserts the key exists).
   243→    if settings.is_postgres:
   244→        state["idem"] = TTLCache(maxsize=1, ttl=3600)  # placeholder, real cache is the DB
   245→    else:
   246→        state["idem"] = TTLCache(
   247→            maxsize=settings.idem_maxsize, ttl=settings.idem_ttl_seconds
   248→        )
   249→
   250→    @asynccontextmanager
   251→    async def lifespan(_: FastAPI):
   252→        df = add_address_features(load_orders("data/raw/cod_orders.csv"))
   253→        train_df, _ = group_split(df)
   254→        X_tr, y_tr = build_feature_frame(train_df, "order+addr")
   255→        model_path = Path("out/model_api.joblib")
   256→        if not model_path.exists():
   257→            save_model(fit_model(X_tr, y_tr), str(model_path))
   258→        from src.models.train import load_model
   259→
   260→        state["model"] = load_model(str(model_path))
   261→        state["reference"] = X_tr.mode().iloc[0]
   262→        state["base_rate"] = float(y_tr.mean())
   263→        # Day 2 Track E — audit path comes from Settings (defaults to the
   264→        # same ``out/audit.jsonl`` as before, but now .env-configurable +
   265→        # docker-compose-wired). The AuditLogger is dual-mode internally.
   266→        state["audit"] = AuditLogger(audit_path or settings.audit_path)
   267→        # Pass Settings-backed CSV keys to default_keys so the .env file is
   268→        # honored (Track B's Dockerfile change removed the baked ENV defaults;
   269→        # docker-compose now sets them via the environment block).
   270→        state["keys"] = default_keys(
   271→            scorer_keys=settings.rto_scorer_keys,
   272→            admin_keys=settings.rto_admin_keys,
   273→        )
   274→        state["bucket"] = TokenBucket(scorer_rate_per_min)
   275→        state["rules"] = RulesEngine()
   276→        state["breaker"] = CircuitBreaker()
   277→        state["metrics"] = Metrics()
   278→        state["cases"] = CaseService(settings.cases_path)
   279→        num_cols = [c for c in X_tr.columns if str(X_tr[c].dtype) != "category"]
   280→        state["psi_sample"] = {
   281→            c: X_tr[c].dropna().sample(n=min(2000, len(X_tr)), random_state=7).tolist()
   282→            for c in num_cols
   283→        }
   284→        # Day 2 Track E — §A item 4 (register_model dead in prod). Register
   285→        # the in-process HistGB with the model registry on every worker boot.
   286→        # File mode: this writes to ``out/model_registry.json`` (the existing
   287→        # behaviour — no-op for tests since they call register_model directly
   288→        # via tmp_path). Postgres mode: INSERT into the ``model_registry``
   289→        # table; the partial-unique index enforces single-champion. PR-AUC
   290→        # + ROC-AUC are surfaced as the metrics payload so the model-card
   291→        # + drift endpoints can read them.
   292→        if settings.is_postgres:
   293→            try:
   294→                from sklearn.metrics import average_precision_score, roc_auc_score
   295→                _p = state["model"].predict_proba(X_tr)[:, 1]
   296→                _pr = float(average_precision_score(y_tr, _p))
   297→                _roc = float(roc_auc_score(y_tr, _p))
   298→                register_model(
   299→                    version=f"v{int(time.time())}",
   300→                    model_path=str(model_path),
   301→                    metrics={"pr_auc": _pr, "roc_auc": _roc},
   302→                    champion=True,
   303→                )
   304→            except Exception as e:  # pragma: no cover — startup-only, defensive
   305→                print(
   306→                    f"model-registry registration skipped: "
   307→                    f"{type(e).__name__}: {e}",
   308→                    file=sys.stderr,
   309→                )
   310→        # Pre-compute labeled (y, p) sequences for the /v1/policy/cost-curves
   311→        # endpoint (Drummond & Holte 2006). The training set carries ground-
   312→        # truth is_returned labels; running the model over it once at startup
   313→        # lets the endpoint respond in O(1) for the per-threshold sweep and
   314→        # ~3-5 sec for the 500-resample bootstrap (per-row performance only).
   315→        # Note: this is in-sample — Day 2 Track E + G will swap in a held-out
   316→        # test slice + delayed-label feedback so the curve reflects deployment
   317→        # conditions, not training fit.
   318→        try:
   319→            _cost_curve_p = state["model"].predict_proba(X_tr)[:, 1].tolist()
   320→            _cost_curve_y = [int(v) for v in y_tr.tolist()]
   321→            # Day 4 Track N — store the dataset median order amount so the
   322→            # /v1/policy/cost-curves endpoint can default the 5-way
   323→            # intervention sweep (Bahnsen Eq.(5): per-amount FN cost) to a
   324→            # representative order value when the caller doesn't pass
   325→            # ``amount_inr``. The raw dataset carries ``OrderValue``; the
   326→            # cleaning step adds ``order_value_inr``; the ``OrderIn`` API
   327→            # surface exposes ``amount_inr`` (all three refer to the same
   328→            # field). Try all three column names, fall back to None if
   329→            # none are present (the endpoint then falls back to 12400 INR —
   330→            # the API_SPEC example value).
   331→            try:
   332→                _amount_col = next(
   333→                    (c for c in ("amount_inr", "order_value_inr", "OrderValue")
   334→                     if c in train_df.columns),
   335→                    None,
   336→                )
   337→                _median_amount = (
   338→                    float(train_df[_amount_col].median()) if _amount_col else None
   339→                )
   340→            except Exception:
   341→                _median_amount = None
   342→            state["cost_curve"] = {
   343→                "y_true": _cost_curve_y,
   344→                "probs": _cost_curve_p,
   345→                "n": len(_cost_curve_y),
   346→                "source": "train_df_in_sample",
   347→                "median_amount_inr": _median_amount,
   348→            }
   349→        except Exception as e:  # pragma: no cover — startup-only, defensive
   350→            print(f"cost-curve warmup skipped: {type(e).__name__}: {e}", file=sys.stderr)
   351→            state["cost_curve"] = None
   352→        yield
   353→        # Day 2 Track E — close the model-registry connection at shutdown so
   354→        # the worker doesn't leak a Postgres connection across hot-reloads.
   355→        if settings.is_postgres:
   356→            _close_registry_conn()
   357→        # Day 2 Track G — close the LabelFeedbackService's lazy StreamProducer
   358→        # so the worker doesn't leak a Redis connection across hot-reloads.
   359→        # Safe to call on a service whose producer was never constructed.
   360→        try:
   361→            state["feedback"].close()
   362→        except Exception:  # pragma: no cover — best-effort shutdown
   363→            pass
   364→
   365→    app = FastAPI(title="RTO Trust Layer", version="0.2.0", lifespan=lifespan)
   366→    app.state.core = state

   385→    @app.post("/risk/score")
   386→    def score(
   387→        order: OrderIn,
   388→        authorization: str | None = Header(default=None),
   389→        idempotency_key: str | None = Header(default=None),
   390→        x_mandate: str | None = Header(default=None),
   391→        # Day 1 Track D (V3 §13): per-txn device_id + user_id for UPI Circle
   392→        # mandates (NPCI OC-201B §3.7 Issuer Bank duty + §3.3 Secondary PSP
   393→        # duty). Both default to None and are ignored for cod_order mandates.
   394→        x_device_id: str | None = Header(default=None),
   395→        x_user_id: str | None = Header(default=None),
   396→        # Day 4 Track M — multi-source ingest channel tag (Microsoft Fabric
   397→        # reference: mobile banking / ATM / e-commerce / call center). The
   398→        # ``channel`` discriminator surfaces in the audit record's ``channel``
   399→        # field + drives per-channel drift detection via TFX
   400→        # generate_data_statistics (Kandula 2021 paper: Payment_Type as a
   401→        # discriminator feature → here ``channel`` is the discriminator).
   402→        # Defaults to ``ecommerce`` for backward-compat with the existing
   403→        # merchant web-checkout path. The 4 ingest simulators in
   404→        # ``src/ingest/`` post with the appropriate channel header.
   405→        x_channel: str | None = Header(default=None),
   406→    ) -> dict:
   407→        token = bearer_token(authorization)
   408→        ok, err = check_key(token, "scorer", state["keys"])
   409→        if not ok:
   410→            raise HTTPException(status_code=401, detail=err)
   411→        client = token
   412→        if not state["bucket"].allow(client):
   413→            raise HTTPException(status_code=429, detail="rate limit exceeded")
   414→
   415→        # Day 2 Track E — idempotency cache (§A item 2: was an unbounded dict;
   416→        # now TTLCache in file mode, Postgres ``idempotency_keys`` table in
   417→        # db mode). File mode preserves the original (key, body) tuple key
   418→        # so two different request bodies with the same Idempotency-Key
   419→        # don't silently cross-pollute. Postgres mode uses the header value
   420→        # alone as the PK (the standard idempotency contract — same key,
   421→        # same response).
   422→        if settings.is_postgres:
   423→            cached = _idem_lookup_postgres(state, idempotency_key)
   424→            if cached is not None:
   425→                # 1% probabilistic cleanup of expired rows.
   426→                if (uuid.uuid4().int % 100) == 0:
   427→                    _idem_cleanup_postgres(state)
   428→                return dict(cached, replayed=True)
   429→        else:
   430→            cache_key = (idempotency_key or "", order.model_dump_json())
   431→            if idempotency_key and cache_key in state["idem"]:
   432→                return dict(state["idem"][cache_key], replayed=True)
   433→
   434→        # Day 4 Track M — OpenTelemetry span around the decision pipeline.
   435→        # Dual-mode: when ``state["tracer"]`` is None (test mode, local dev
   436→        # without Jaeger), the ExitStack stays empty + the inline
   437→        # ``if span is not None:`` checks short-circuit, so the OTel SDK
   438→        # is never imported on the disabled path. The span covers the
   439→        # mandate check → rules engine → cost-optimizer → audit → stream
   440→        # publish path so a Jaeger trace surfaces the end-to-end decision
   441→        # pipeline + the Jaeger UI (:16686) shows order_id / amount /
   442→        # decision / score / decision_source on every span. Source: OTel
   443→        # Python SDK §"Manual instrumentation" + the V2 §9.2 observability

   445→        _span_stack = ExitStack()
   446→        span = (
   447→            _span_stack.enter_context(state["tracer"].start_as_current_span("risk.score"))
   448→            if state["tracer"] is not None
   449→            else None
   450→        )
   451→        if span is not None:
   452→            span.set_attribute("order_id", order.order_id)
   453→            span.set_attribute("amount", float(order.amount_inr))
   454→
   455→        try:
   456→            t0 = time.monotonic()
   457→            # Mandate verification (cheap; always run; enforcement happens
   458→            # later). Day 1 Track D extended the signature to pass device_id
   459→            # + user_id so UPI Circle mandates can enforce OC-201B §3.3/§3.7
   460→            # per-txn identity validation. cod_order mandates ignore both.
   461→            mandate_verdict, mandate_payload = verify_mandate(
   462→                x_mandate,
   463→                order.amount_inr,
   464→                device_id=x_device_id,
   465→                user_id=x_user_id,
   466→            )
   467→            fired = state["rules"].evaluate(order.model_dump())
   468→            rule_fired = fired.rule_id if fired else None
   469→
   470→            proba: float | None = None
   471→            reasons: list[dict] = []
   472→            degraded = False
   473→            breach_note: str | None = None
   474→            cost_breakdown: dict | None = None
   475→            decision: str | None = None
   476→            # Day 4 Track N — 5-way intervention policy fields. Track C's
   477→            # 3-way decision still drives the primary ``decision`` field;
   478→            # ``intervention`` is the cost-optimal V3 §11.6 recommendation
   479→            # (ship / otp_verify / partial_cod / address_check / hold) computed
   480→            # from the per-amount FN cost (Bahnsen Eq.(5)). Surfaced for REVIEW
   481→            # decisions where the operator needs a granular next-step beyond
   482→            # "REVIEW = selective OTP". ``intervention_costs`` carries the full
   483→            # 5-way breakdown for explainability + dashboard rendering.
   484→            intervention: str | None = None
   485→            intervention_costs: dict | None = None
   486→            # ``decision_source`` records which layer actually chose the
   487→            # decision, for the audit trail + dashboard explainability.
   488→            # One of: rules_engine_block | mandate_breach | mandate_invalid
   489→            # | mandate_review_required | degraded_review | cost_optimal_bmr
   490→            # | cost_optimal_bmr_review_rule
   491→            decision_source: str
   492→
   493→            # ------------------------------------------------------------------
   494→            # Decision precedence (per Day 1 Track C spec):
   495→            #   1. Rules fast-path (BLOCK)         → short-circuit REJECT
   496→            #   2. Mandate enforcement (BREACH /
   497→            #      TAMPERED / EXPIRED w/ header)   → short-circuit REJECT
   498→            #   2c. Mandate REVIEW (UPI Circle 24h
   499→            #      cooling period, OC-201B)        → short-circuit REVIEW
   500→            #      (Day 1 Track D — mandate action-class expansion)
   501→            #   3. Circuit breaker OPEN            → degraded rules-only REVIEW

   503→            #      (replaces legacy static 0.15 / 0.60 thresholds)
   504→            # REVIEW rules don't short-circuit; they gate step 4 to never ACCEPT.
   505→            # ------------------------------------------------------------------
   506→            if fired is not None and fired.action == "BLOCK":
   507→                # 1. Rules-engine BLOCK short-circuits — no model invocation.
   508→                decision = "REJECT"
   509→                decision_source = "rules_engine_block"
   510→            elif mandate_verdict == MandateVerdict.BREACH:
   511→                # 2a. Mandate amount-breach short-circuits.
   512→                decision = "REJECT"
   513→                breach_note = "mandate_amount_breach"
   514→                decision_source = "mandate_breach"
   515→            elif x_mandate is not None and mandate_verdict == MandateVerdict.REVIEW:

   517→                # is permitted in principle; the cooling circuit requires a
   518→                # human to approve before the issuer debits. Routes to the

   520→                # (e.g. "cooling_period_active") flows into the audit trail
   521→                # for compliance traceability.
   522→                decision = "REVIEW"
   523→                breach_note = "mandate_review_required"
   524→                decision_source = "mandate_review_required"
   525→            elif x_mandate is not None and mandate_verdict != MandateVerdict.VALID:

   527→                # Covers OC-201B 6-month inactivity auto-revoke (verdict_reason
   528→                # = "inactivity_auto_revoke") and standard TTL expiry.
   529→                decision = "REJECT"
   530→                breach_note = f"mandate_{mandate_verdict}"
   531→                decision_source = "mandate_invalid"
   532→            else:

   534→                use_model = state["breaker"].allow_attempt()

   537→                        X, _ = build_feature_frame(to_frame(order), "order+addr")

   540→                            X,
   541→                            list(X.columns),
   542→                            state["base_rate"],
   543→                            state["reference"],

   558→                    #    cost over {ACCEPT, REVIEW (selective-OTP), REJECT}.

   560→                    #    default (backward compat with Track C's tests). Day 4
   561→                    #    Track N's per-amount FN cost (Bahnsen Eq.(5)) is wired
   562→                    #    into the 5-way ``optimal_intervention`` call below —
   563→                    #    the 3-way decision remains the primary authorization
   564→                    #    signal; the 5-way intervention is the operator's

   576→                    # ``intervention`` is the operator's next-step hint within

   578→                    intervention, intervention_costs = optimal_intervention(
   579→                        proba, order.amount_inr,
   580→                    )
   581→                    # REVIEW rule gate: never ACCEPT when a REVIEW rule fired
   582→                    # (still uses BMR cost math, but the rule forces REVIEW).
   583→                    if fired is not None and fired.action == "REVIEW" and decision == "ACCEPT":
   584→                        decision = "REVIEW"
   585→                        decision_source = "cost_optimal_bmr_review_rule"
   586→                    else:
   587→                        decision_source = "cost_optimal_bmr"
   588→

   590→            # dashboard / consumer backward compat; the actual decision is the
   591→            # body's ``decision`` field — which equals ``policy_hint`` whenever

   593→            # constant-c_fn 3-way path as the live decision above (Track C
   594→            # behaviour; per-amount FN cost is wired into the 5-way
   595→            # ``optimal_intervention`` only).
   596→            policy_hint = None
   597→            if proba is not None:
   598→                policy_hint = optimal_decision(proba, **DEFAULT_COST_WEIGHTS)[0]
   599→
   600→            features_used = (
   601→                {c: float(X.iloc[0][c]) for c in X.columns if str(X[c].dtype) != "category"}
   602→                if proba is not None
   603→                else {}
   604→            )
   605→            # Day 2 Track F — generate the prediction_id ONCE so the case
   606→            # row, the audit record's stream publish, and the response body
   607→            # all share the same identifier. Track B's original code passed

   610→            # stream consumers couldn't correlate a ``risk.scores`` message
   611→            # to a ``cases.created`` message (different prediction_id on

   613→            # and the response.
   614→            prediction_id = str(uuid.uuid4())
   615→            case_id = None
   616→            if decision == "REVIEW":
   617→                case_id = state["cases"].open_case(
   618→                    prediction_id=prediction_id, order_id=order.order_id,
   619→                    reason=(
   620→                        f"mandate:{mandate_payload.get('verdict_reason')}"

   622→                        else "review_gate" if fired is None
   623→                        else f"rule:{fired.rule_id}"
   624→                    ),
   625→                )
   626→            # Audit payload (Day 1 Track D — V3 §13 mandate action-class
   627→            # expansion). The new fields surface OC-201B compliance metadata

   629→            # ``bh_purpose_code``, ``device_id``, ``user_id``, and the

   633→            audit_id = state["audit"].log(
   634→                {
   635→                    "request": {
   636→                        **order.model_dump(),
   637→                        "customer_id": redact_customer(order.customer_id),

   640→                    "decision": decision,
   641→                    "decision_source": decision_source,
   642→                    "cost_breakdown": cost_breakdown,
   643→                    # Day 4 Track N — V3 §11.6 5-way intervention policy fields

   648→                    # cost-optimizer suggested vs. what the operator did, which

   651→                    "intervention_costs": intervention_costs,
   652→                    "reason_codes": reasons[:5],
   653→                    "mandate_verdict": mandate_verdict,
   654→                    "mandate_verdict_reason": mandate_payload.get("verdict_reason"),
   655→                    "mandate_type": mandate_payload.get("mandate_type"),
   656→                    "bh_purpose_code": mandate_payload.get("bh_purpose_code"),
   657→                    "device_id": x_device_id,
   658→                    "user_id": x_user_id,
   659→                    "breach_note": breach_note,
   660→                    "rule_fired": rule_fired,
   661→                    "degraded": degraded,
   662→                    "features_used": features_used,

   823→                span.set_attribute(
   824→                    "score", float(proba) if proba is not None else 0.0
   825→                )

   828→        except HTTPException:

   830→        except Exception as e:  # no internal detail leakage

   851→            # value is computed but before control returns to caller).
   852→            _span_stack.close()

  1047→    @app.get("/v1/policy/cost-curves")
  1048→    def policy_cost_curves(
  1049→        n_resamples: int = DEFAULT_COST_CURVE_RESAMPLES,
  1050→        confidence: float = DEFAULT_COST_CURVE_CONFIDENCE,
  1051→        amount_inr: float | None = None,
  1052→        authorization: str | None = Header(default=None),
  1053→    ) -> dict:
  1054→        """Drummond-Holte cost-curve explorer (Machine Learning 65:95-130, 2006,
  1055→        DOI 10.1007/s10994-006-8199-5) + V3 §11.6 5-way intervention curves
  1056→        (Day 4 Track N — Bahnsen Eq.(5) per-amount FN cost).
  1057→
  1058→        Returns a threshold sweep (0.05 → 0.95) over the labeled dataset that
  1059→        the in-process model was trained on. For each threshold we report the
  1060→        confusion counts, total Bahnsen cost (Eq. 1), precision, recall, plus
  1061→        a 90% bootstrap CI on cost (≥500 resamples preserving row marginals —
  1062→        per Drummond-Holte ``bootstrap_performance_ci`` capability).
  1063→
  1064→        Day 4 Track N — also returns ``intervention_curves`` (the 5-way
  1065→        intervention sweep: ship/otp_verify/partial_cod/address_check/hold
  1066→        per-threshold cost breakdown) + ``intervention_crossover`` (the
  1067→        threshold(s) where the cost-optimal intervention changes). The
  1068→        intervention sweep uses the per-amount FN cost (Bahnsen Eq.(5)):
  1069→        ``amount_inr`` param overrides the default representative order value
  1070→        so the dashboard can render the cost-optimal intervention for any
  1071→        order-amount bracket.
  1072→

  1074→        real bars (replacing the previously-hardcoded ``COSTS`` array).
  1075→
  1076→        Auth: scorer scope. Returns 503 if the model isn't loaded or no
  1077→        labeled data is available — never crashes the dashboard.
  1078→        """
  1079→        ok, err = check_key(bearer_token(authorization), "scorer", state["keys"])
  1080→        if not ok:
  1081→            raise HTTPException(status_code=401, detail=err)
  1082→
  1083→        # 1. Guard: model must be loaded (circuit breaker may be OPEN).
  1084→        if state.get("model") is None:
  1085→            raise HTTPException(
  1086→                status_code=503,
  1087→                detail="cost curves unavailable — model not loaded",
  1088→            )
  1089→        # 2. Guard: pre-computed labeled data must exist (startup warmup may
  1090→        # have failed silently; surface a clear error to the dashboard).
  1091→        cc = state.get("cost_curve")
  1092→        if not cc or not cc.get("y_true") or not cc.get("probs"):
  1093→            raise HTTPException(
  1094→                status_code=503,
  1095→                detail="cost curves unavailable — no labeled data available",
  1096→            )
  1097→        if n_resamples < 1 or n_resamples > 5000:
  1098→            raise HTTPException(
  1099→                status_code=422,
  1100→                detail="n_resamples must be in [1, 5000]",
  1101→            )
  1102→        if not (0.5 < confidence < 0.999):
  1103→            raise HTTPException(
  1104→                status_code=422,
  1105→                detail="confidence must be in (0.5, 0.999)",
  1106→            )
  1107→        # Day 4 Track N — validate the optional amount_inr param. None →
  1108→        # use the dataset median as the representative order value for the
  1109→        # 5-way intervention sweep (Bahnsen Eq.(5): FN cost = amount). The
  1110→        # operator can pass any positive amount to render the intervention
  1111→        # curves for a specific order-value bracket.
  1112→        if amount_inr is not None and (not (1.0 <= float(amount_inr) <= 1_000_000.0)):
  1113→            raise HTTPException(
  1114→                status_code=422,
  1115→                detail="amount_inr must be in [1, 1_000_000]",
  1116→            )
  1117→
  1118→        y_true = cc["y_true"]
  1119→        probs = cc["probs"]
  1120→        weights = DEFAULT_COST_WEIGHTS
  1121→        c_fp = float(weights["c_fp"])
  1122→        c_fn = float(weights["c_fn"])
  1123→
  1124→        # 3. Threshold sweep → per-threshold confusion + cost + precision +
  1125→        #    recall (pure-Python, O(N*T); ~140k ops for 7235 rows × 19 thr).
  1126→        curves = cost_curve_sweep(
  1127→            y_true=y_true,
  1128→            probs=probs,
  1129→            c_fp=c_fp,
  1130→            c_fn=c_fn,
  1131→        )
  1132→
  1133→        # 4. Bootstrap CI per threshold (Drummond-Holte row-marginal-preserving
  1134→        #    resampling; ~3-5 sec for 500 resamples × 19 thr × 7235 samples).
  1135→        try:
  1136→            bootstrap_ci = bootstrap_cost_ci(
  1137→                y_true=y_true,
  1138→                probs=probs,
  1139→                c_fp=c_fp,
  1140→                c_fn=c_fn,
  1141→                n_resamples=n_resamples,
  1142→                confidence=confidence,
  1143→                seed=42,
  1144→            )
  1145→        except Exception as e:  # pragma: no cover — defensive
  1146→            # If bootstrap fails for any reason, return degenerate CIs and a
  1147→            # warning rather than 500-ing the dashboard.
  1148→            bootstrap_ci = {
  1149→                str(r["threshold"]): {"low": 0.0, "high": 0.0, "mean": 0.0}
  1150→                for r in curves
  1151→            }

  1153→                f"cost-curves bootstrap degraded: {type(e).__name__}: {e}",

  1179→                # We don't have challenger probabilities pre-computed yet

  1182→                cost_crossover = {
  1183→                    "status": "single_model",
  1184→                    "incumbent_version": (champ or {}).get("version"),
  1185→                    "challenger_version": None,
  1186→                    "crossover_threshold": None,
  1187→                    "note": (
  1188→                        "challenger predictions not pre-computed; "

  1192→            else:
  1193→                cost_crossover = {
  1194→                    "status": "single_model",
  1195→                    "incumbent_version": (champ or {}).get("version"),
  1196→                    "challenger_version": None,

  1207→        #    3-way threshold sweep so the operator sees both views.

  1209→            # Use the labeled dataset's amount distribution if available; else
  1210→            # fall back to a sensible default (₹12400 — the API_SPEC example).
  1211→            representative_amount = float(amount_inr) if amount_inr is not None else float(
  1212→                cc.get("median_amount_inr") or 12400.0
  1213→            )
  1214→            intervention_curves = intervention_curve_sweep(representative_amount)

  1216→        except Exception as e:  # pragma: no cover — defensive

  1219→                "crossover_thresholds": [],
  1220→                "per_region_intervention": [],
  1221→                "regions": [],
  1222→                "error": f"{type(e).__name__}: {e}",

  1893→def _idem_get_conn(state: dict):
  1894→    """Lazily-open a dedicated psycopg connection for the idempotency cache.
  1895→
  1896→    Separate from the AuditLogger / CaseService / model-registry connections
  1897→    so a slow idempotency lookup can't block an audit write. The connection
  1898→    is cached on ``state`` so we don't pay the connect-handshake cost on
  1899→    every request.
  1900→    """
  1901→    conn = state.get("_idem_conn")
  1902→    if conn is not None:
  1903→        return conn
  1904→    import psycopg
  1905→
  1906→    conn = psycopg.connect(state["settings"].database_url, autocommit=False)
  1907→    state["_idem_conn"] = conn
  1908→    return conn

  1911→def _idem_lookup_postgres(state: dict, key: str | None) -> dict | None:
  1912→    """Read a cached idempotent response. Returns the body dict (parsed from
  1913→    the JSONB-stored response_body TEXT column) or None if not cached /
  1914→    expired. ``expires_at > NOW()`` is the TTL gate.
  1915→    """
  1916→    if not key:
  1917→        return None
  1918→    import json as _json
  1919→
  1920→    conn = _idem_get_conn(state)
  1921→    with conn.cursor() as cur:
  1922→        cur.execute(
  1923→            """
  1924→            SELECT response_body, status_code
  1925→              FROM idempotency_keys
  1926→             WHERE key = %s AND expires_at > NOW()
  1927→            """,
  1928→            (key,),
  1929→        )
  1930→        row = cur.fetchone()
  1931→    if row is None:
  1932→        return None
  1933→    body, _status = row
  1934→    return _json.loads(body)

  1937→def _idem_store_postgres(
  1938→    state: dict,
  1939→    key: str,
  1940→    request_body: str,
  1941→    response_body: dict,
  1942→    status_code: int,
  1943→) -> None:
  1944→    """INSERT an idempotency cache row. TTL = NOW() + settings.idem_ttl_seconds."""
  1945→    import json as _json
  1946→    from datetime import datetime, timedelta, timezone
  1947→
  1948→    conn = _idem_get_conn(state)
  1949→    expires = datetime.now(timezone.utc) + timedelta(seconds=state["settings"].idem_ttl_seconds)
  1950→    with conn.cursor() as cur:
  1951→        # ON CONFLICT DO UPDATE so a re-store of the same key (e.g. a slow
  1952→        # retry that lands after the original INSERT) refreshes expires_at
  1953→        # rather than erroring.
  1954→        cur.execute(
  1955→            """
  1956→            INSERT INTO idempotency_keys
  1957→              (key, request_body, response_body, status_code, expires_at)
  1958→            VALUES (%s, %s, %s, %s, %s)
  1959→            ON CONFLICT (key) DO UPDATE SET
  1960→              request_body  = EXCLUDED.request_body,
  1961→              response_body = EXCLUDED.response_body,
  1962→              status_code   = EXCLUDED.status_code,
  1963→              expires_at    = EXCLUDED.expires_at
  1964→            """,
  1965→            (key, request_body, _json.dumps(response_body), status_code, expires),
  1966→        )
  1967→        conn.commit()

  1970→def _idem_cleanup_postgres(state: dict) -> None:
  1971→    """Best-effort TTL cleanup. Called with 1% probability per request so the
  1972→    table doesn't grow forever under burst traffic.
  1973→    """
  1974→    conn = _idem_get_conn(state)
  1975→    try:
  1976→        with conn.cursor() as cur:
  1977→            cur.execute("DELETE FROM idempotency_keys WHERE expires_at < NOW()")
  1978→            conn.commit()
  1979→    except Exception:  # pragma: no cover — best-effort, never fail the request
  1980→        try:
  1981→            conn.rollback()
  1982→        except Exception:
  1983→            pass
