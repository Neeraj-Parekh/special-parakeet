    61→class MerkleSealer:
    62→    """Merkle interval sealing per V3 §10.3.
    63→
    64→    Every N records (default 1000) or T seconds (default 3600), compute
    65→    a Merkle root of the batch's ``raw_hash`` leaves, chain it to the
    66→    previous interval's root, store the interval.
    67→
    68→    This is ADDITIONAL to the per-record hash chain (Track E Day 2) —
    69→    it's a coarser tamper-evidence layer that enables O(log N) proof
    70→    generation per record (vs O(N) full-chain recompute via
    71→    ``AuditLogger.verify_chain()``). The per-record ``raw_hash`` is the
    72→    leaf; ``audit_merkle_intervals.merkle_root`` is the interval root;
    73→    ``audit_merkle_intervals.prev_interval_root`` chains intervals
    74→    together so cross-interval tampering is detected (same model as
    75→    Track E's per-record ``prev_hash``).
    76→
    77→    Source: SoK (Mao 2026) capability ``recommend_layered_defenses``
    78→    layer 5 — "market & compliance monitoring with tamper-evident audit
    79→    trails"; ``audit_agent_mandate_scoping`` (the dual-control override
    80→    in routes.py consumes this tamper-evidence layer so both admin
    81→    signatures are anchored to a verifiable root, not just the live
    82→    hash chain tip).
    83→
    84→    File mode is a no-op: the per-record hash chain is enough for tests
    85→    + local dev without Postgres. Merkle is a Postgres-mode enhancement.
    86→    """
    87→
    88→    def __init__(
    89→        self,
    90→        conn=None,
    91→        interval_size: int = 1000,
    92→        interval_seconds: int = 3600,
    93→    ):
    94→        # Postgres connection (None in file mode → no-op). Shared with
    95→        # the parent AuditLogger's connection — no extra pool needed.
    96→        self.conn = conn
    97→        self.interval_size = interval_size
    98→        self.interval_seconds = interval_seconds
    99→        # Pending (record_id, raw_hash) pairs since the last seal.
   100→        # Bounded by interval_size; flushed to audit_merkle_intervals on
   101→        # add() when len(self._pending) >= interval_size or by an
   102→        # explicit seal_interval() call (e.g. from a cron / shutdown hook).
   103→        self._pending: list[tuple[int, str]] = []
   104→        # First-record timestamp so the time-based seal trigger can compute
   105→        # elapsed since the interval opened (independent of record count).
   106→        self._interval_started_at: datetime | None = None
   107→
   108→    # ------------------------------------------------------------------ #
   109→    # Public API — add / seal / proof                                    #
   110→    # ------------------------------------------------------------------ #
   111→
   112→    def add(self, record_id: int, raw_hash: str) -> dict | None:
   113→        """Add a record to the pending interval. Seal if threshold reached.
   114→
   115→        Called by ``AuditLogger._log_postgres`` after every INSERT so the
   116→        sealer tracks new records in real time. The first record's
   117→        timestamp seeds ``_interval_started_at`` for the time-based
   118→        trigger (so an idle period after a partial interval still flushes
   119→        on the next ``seal_interval()`` even with no fresh traffic).
   120→        """
   121→        if self.conn is None:
   122→            # File mode — no-op (per-record hash chain is enough there).
   123→            return None
   124→        if not self._pending:
   125→            self._interval_started_at = datetime.now(timezone.utc)
   126→        self._pending.append((record_id, raw_hash))
   127→        # Seal when EITHER threshold trips (count OR elapsed). The count
   128→        # trigger fires inline; the time trigger is checked here too so
   129→        # a long quiet gap after a partial batch doesn't leave the
   130→        # interval open forever (the cron hook calls seal_interval() too).
   131→        if len(self._pending) >= self.interval_size:
   132→            return self.seal()
   133→        if self._interval_started_at is not None:
   134→            elapsed = (datetime.now(timezone.utc) - self._interval_started_at).total_seconds()
   135→            if elapsed >= self.interval_seconds and len(self._pending) > 0:
   136→                return self.seal()
   137→        return None
   138→
   139→    def seal(self) -> dict | None:
   140→        """Seal the current pending batch into a Merkle interval.
   141→
   142→        Returns the interval metadata dict (merkle_root, prev_root,
   143→        leaf_count) on success, None if there's nothing to seal or the
   144→        sealer is in file mode (conn is None). After sealing, the
   145→        pending batch is cleared + the next add() starts a fresh
   146→        interval. Idempotent — calling seal() with an empty pending
   147→        list is a no-op.
   148→        """
   149→        if not self._pending or self.conn is None:
   150→            return None
   151→        # Compute Merkle root of the raw_hash leaves. Padding to a power
   152→        # of 2 keeps the tree balanced so the proof builder's index
   153→        # arithmetic (sibling_idx = idx ^ 1) is consistent across levels.
   154→        leaves = [h for _, h in self._pending]
   155→        merkle_root = self._merkle_root(leaves)
   156→        # Get previous interval's root (None on first interval → genesis).
   157→        with self.conn.cursor() as cur:
   158→            cur.execute(
   159→                "SELECT merkle_root FROM audit_merkle_intervals "
   160→                "ORDER BY interval_id DESC LIMIT 1"
   161→            )
   162→            prev = cur.fetchone()
   163→        prev_root = prev[0] if prev else GENESIS
   164→        # Insert the interval row + backfill the per-record columns
   165→        # (interval_id + interval_position) so the proof builder can
   166→        # locate a record's interval + leaf index in O(1).
   167→        with self.conn.cursor() as cur:
   168→            cur.execute(
   169→                """
   170→                INSERT INTO audit_merkle_intervals
   171→                  (start_record_id, end_record_id, merkle_root,
   172→                   prev_interval_root, leaf_count)
   173→                VALUES (%s, %s, %s, %s, %s)
   174→                RETURNING interval_id
   175→                """,
   176→                (
   177→                    self._pending[0][0],
   178→                    self._pending[-1][0],
   179→                    merkle_root,
   180→                    prev_root,
   181→                    len(leaves),
   182→                ),
   183→            )
   184→            interval_id = cur.fetchone()[0]
   185→            # currval('audit_merkle_intervals_interval_id_seq') is the
   186→            # just-inserted row's SERIAL value; using it in the UPDATE
   187→            # avoids passing interval_id back from Python (atomic within
   188→            # the same transaction). Per-record backfill is O(M) where
   189→            # M = interval_size (default 1000) — cheap.
   190→            for pos, (rid, _) in enumerate(self._pending):
   191→                cur.execute(
   192→                    """
   193→                    UPDATE audit_records
   194→                       SET interval_id = %s, interval_position = %s
   195→                     WHERE id = %s
   196→                    """,
   197→                    (interval_id, pos, rid),
   198→                )
   199→            self.conn.commit()
   200→        self._pending = []
   201→        self._interval_started_at = None
   202→        return {
   203→            "interval_id": interval_id,
   204→            "merkle_root": merkle_root,
   205→            "prev_root": prev_root,
   206→            "leaf_count": len(leaves),
   207→        }
   208→
   209→    @staticmethod
   210→    def _merkle_root(leaves: list[str]) -> str:
   211→        """Compute Merkle root from leaf hashes. Pad to power of 2.
   212→
   213→        Padding uses the LAST leaf's hash repeated (RFC 6962-style) so
   214→        the tree is balanced without introducing a synthetic zero-leaf
   215→        that a verifier would need to special-case. Empty leaf list →
   216→        GENESIS (the same ``"0"*64`` constant Track E's hash chain uses
   217→        as the genesis ``prev_hash``).
   218→        """
   219→        if not leaves:
   220→            return GENESIS
   221→        # Pad to next power of 2.
   222→        size = 1
   223→        while size < len(leaves):
   224→            size *= 2
   225→        padded = leaves + [leaves[-1]] * (size - len(leaves))
   226→        # Build tree level-by-level: parent = sha256(left + right).
   227→        level = padded
   228→        while len(level) > 1:
   229→            next_level = []
   230→            for i in range(0, len(level), 2):
   231→                combined = hashlib.sha256(
   232→                    (level[i] + level[i + 1]).encode()
   233→                ).hexdigest()
   234→                next_level.append(combined)
   235→            level = next_level
   236→        return level[0]
   237→
   238→    def proof(self, record_id: int) -> dict | None:
   239→        """Generate Merkle proof for a record: path from leaf to interval root.
   240→
   241→        Returns None if the record doesn't exist, has no interval_id
   242→        (i.e. the sealer hasn't reached the threshold + the cron seal
   243→        hasn't fired), or the sealer is in file mode. The proof dict
   244→        contains the sibling hashes from leaf to root, the interval's
   245→        root + prev_interval_root (chain anchor), leaf_count, sealed_at
   246→        — enough to verify inclusion in O(log N) without recomputing
   247→        the whole tree.
   248→        """
   249→        if self.conn is None:
   250→            return None
   251→        # Get the record's interval + position.
   252→        with self.conn.cursor() as cur:
   253→            cur.execute(
   254→                "SELECT interval_id, interval_position, raw_hash "
   255→                "FROM audit_records WHERE id = %s",
   256→                (record_id,),
   257→            )
   258→            row = cur.fetchone()
   259→        if not row or row[0] is None:
   260→            # Record not found, OR interval not yet sealed.
   261→            return None
   262→        interval_id, position, leaf_hash = row
   263→        # Get all leaves in this interval (the proof builder needs to
   264→        # descend from the leaf position; siblings at each level are
   265→        # looked up by index — pre-fetched once per proof for simplicity
   266→        # rather than one query per tree level).
   267→        with self.conn.cursor() as cur:
   268→            cur.execute(
   269→                "SELECT raw_hash FROM audit_records WHERE interval_id = %s "
   270→                "ORDER BY interval_position",
   271→                (interval_id,),
   272→            )
   273→            leaves = [r[0] for r in cur.fetchall()]
   274→        # Build proof: at each tree level, the sibling's hash + whether
   275→        # it's the left or right child of the parent node. Pad leaves
   276→        # to a power of 2 (same padding rule as _merkle_root) so the
   277→        # index arithmetic is consistent.
   278→        size = 1
   279→        while size < len(leaves):
   280→            size *= 2
   281→        level = leaves + [leaves[-1]] * (size - len(leaves))
   282→        proof: list[dict] = []
   283→        idx = position
   284→        while len(level) > 1:
   285→            sibling_idx = idx ^ 1  # XOR 1: pairs (0,1), (2,3), ...
   286→            if sibling_idx < len(level):
   287→                proof.append(
   288→                    {
   289→                        "position": "right" if sibling_idx > idx else "left",
   290→                        "hash": level[sibling_idx],
   291→                    }
   292→                )
   293→            else:
   294→                # Padding sibling (RFC 6962-style last-leaf-repeat).
   295→                proof.append({"position": "right", "hash": level[-1]})
   296→            # Compute next level (parent hashes).
   297→            next_level = []
   298→            for i in range(0, len(level), 2):
   299→                combined = hashlib.sha256(
   300→                    (level[i] + level[i + 1 if i + 1 < len(level) else i]).encode()
   301→                ).hexdigest()
   302→                next_level.append(combined)
   303→            level = next_level
   304→            idx //= 2
   305→        # Get the interval's metadata for the chain anchor.
   306→        with self.conn.cursor() as cur:
   307→            cur.execute(
   308→                """
   309→                SELECT merkle_root, prev_interval_root, leaf_count, sealed_at
   310→                  FROM audit_merkle_intervals WHERE interval_id = %s
   311→                """,
   312→                (interval_id,),
   313→            )
   314→            interval = cur.fetchone()
   315→        if not interval:
   316→            return None
   317→        return {
   318→            "record_id": record_id,
   319→            "leaf_hash": leaf_hash,
   320→            "interval_id": interval_id,
   321→            "position": position,
   322→            "proof": proof,
   323→            "merkle_root": interval[0],
   324→            "prev_interval_root": interval[1],
   325→            "leaf_count": interval[2],
   326→            "sealed_at": interval[3].isoformat() if interval[3] else None,
   327→        }

   330→class AuditLogger:
   331→    """Append-only audit log, tamper-evident hash chain, O(1) indexed reads.
   332→
   333→    Dual-mode (Day 2 Track E):
   334→      * Postgres mode (``settings.database_url`` set): INSERT into
   335→        ``audit_records``; reads via ``SELECT ... WHERE audit_id = %s``;
   336→        chain verification via ``SELECT audit_id, body, raw_hash, prev_hash
   337→        FROM audit_records ORDER BY id ASC``.
   338→      * File mode (no DSN): the original JSONL append + byte-offset index.
   339→
   340→    The hash-chain math is identical between modes (``canonical(body) +
   341→    prev_hash``) so a verifier computes the same hash either way.
   342→
   343→    Day 2 Track H — Merkle interval sealing (V3 §10.3). In Postgres
   344→    mode, ``__init__`` constructs a ``MerkleSealer`` sharing the same
   345→    connection; ``_log_postgres`` calls ``sealer.add(record_id, raw_hash)``
   346→    after every INSERT so the sealer tracks new records in real time.
   347→    File mode skips Merkle (the per-record hash chain is enough there).
   348→    """

   350→    HASH_FIELDS = ("previous_hash", "raw_hash")

   352→    def __init__(self, path: str = "out/audit.jsonl", model_version: str = "dev"):
   353→        from src.config import get_settings
   354→
   355→        self.settings = get_settings()
   356→        self.model_version = model_version
   357→        self._lock = threading.Lock()
   358→
   359→        if self.settings.is_postgres:
   360→            # Postgres mode — open one persistent psycopg connection per logger
   361→            # instance. We use a single connection (not a pool) because the
   362→            # API is single-process + the audit log is the write-hot path;
   363→            # a pool would add latency for no benefit at this scale.
   364→            import psycopg
   365→
   366→            self._conn = psycopg.connect(self.settings.database_url, autocommit=False)
   367→            self.path = Path(path)  # kept for backward-compat (audit_tail reads)
   368→            self.path.touch(exist_ok=True)  # so audit_logger.path.exists() is True
   369→            # Track last_hash from the most recent row so we can chain without
   370→            # a SELECT on every INSERT. Hydrated lazily on first log() call
   371→            # (cheap query; we'd rather not slow __init__ if the table is empty).
   372→            self._last_hash_cached: str | None = None
   373→            # Day 2 Track H — Merkle interval sealer (V3 §10.3). Shares
   374→            # the parent connection so audit INSERT + Merkle seal happen
   375→            # in the same transaction (atomic; partial-failure recovery
   376→            # is left to Postgres's WAL, not Python try/except).
   377→            self.sealer = MerkleSealer(conn=self._conn)
   378→        else:
   379→            # File mode — original behaviour, unchanged. The Merkle
   380→            # sealer is a no-op here (per-record hash chain is enough
   381→            # for tests; Merkle is a Postgres-mode enhancement).
   382→            self._conn = None
   383→            self.path = Path(path)
   384→            self._index: dict[str, int] = {}
   385→            self.last_hash = GENESIS
   386→            self.sealer = None
   387→            if self.path.exists():
   388→                with self.path.open() as f:
   389→                    for offset, line in enumerate(f):
   390→                        rec = json.loads(line)
   391→                        self._index[rec.get("audit_id", "")] = offset
   392→                        self.last_hash = rec.get("raw_hash", self.last_hash)

   450→    def merkle_proof(self, record_id: int) -> dict | None:
   451→        """Generate Merkle inclusion proof for an audit record (V3 §10.3).

   453→        Returns None in file mode (no Merkle layer) OR if the record's

   461→            return None

   464→    def merkle_intervals(self, limit: int = 100) -> list[dict]:
   465→        """Return the last N sealed intervals (compliance audit endpoint).
   466→
   467→        Returns ``[]`` in file mode. Used by the ``/v1/usage`` endpoint

   473→        if self._conn is None:
   474→            return []
   475→        with self._conn.cursor() as cur:
   476→            cur.execute(
   477→                """
   478→                SELECT interval_id, start_record_id, end_record_id,
   479→                       merkle_root, prev_interval_root, leaf_count, sealed_at
   480→                  FROM audit_merkle_intervals
   481→                 ORDER BY interval_id DESC LIMIT %s
   482→                """,
   483→                (limit,),
   484→            )
   485→            rows = cur.fetchall()
   486→        return [
   487→            {
   488→                "interval_id": r[0],
   489→                "start_record_id": r[1],
   490→                "end_record_id": r[2],
   491→                "merkle_root": r[3],
   492→                "prev_interval_root": r[4],
   493→                "leaf_count": r[5],
   494→                "sealed_at": r[6].isoformat() if r[6] else None,

   496→            for r in rows

   499→    def usage_counts(self, since_hours: tuple[int, ...] = (24, 168, 720)) -> dict:
   500→        """Per-window audit-record counts for the ``/v1/usage`` metering
   501→        endpoint. Returns ``{str(h): count, ...}`` for each h in
   502→        ``since_hours`` (default last 24h / 7d / 30d).
   503→
   504→        Postgres mode: ``SELECT count(*) FROM audit_records WHERE
   505→        created_at > now() - interval '%s hours'``. File mode: scan the
   506→        JSONL + filter by the record's ``timestamp`` field. Cheap on
   507→        either side — the audit table is indexed on ``created_at DESC``
   508→        (Track E migration 001), so the count is an index-only scan.
   509→        """
   510→        out: dict[str, int] = {}
   511→        if self._conn is not None:
   512→            with self._conn.cursor() as cur:
   513→                for h in since_hours:
   514→                    cur.execute(
   515→                        "SELECT count(*) FROM audit_records "
   516→                        "WHERE created_at > now() - interval '%s hours'",
   517→                        (h,),
   518→                    )
   519→                    out[str(h)] = int(cur.fetchone()[0])
   520→            return out
   521→        # File mode — scan + filter by timestamp.
   522→        if not self.path.exists():
   523→            return {str(h): 0 for h in since_hours}
   524→        now = datetime.now(timezone.utc)
   525→        cutoffs = {h: now.timestamp() - h * 3600 for h in since_hours}
   526→        counts = {h: 0 for h in since_hours}
   527→        try:
   528→            for line in self.path.read_text().splitlines():
   529→                if not line.strip():
   530→                    continue
   531→                try:
   532→                    rec = json.loads(line)
   533→                    ts = rec.get("timestamp")
   534→                    if not ts:
   535→                        continue
   536→                    parsed = datetime.fromisoformat(ts).timestamp()
   537→                    for h in since_hours:
   538→                        if parsed >= cutoffs[h]:
   539→                            counts[h] += 1
   540→                except (json.JSONDecodeError, ValueError, TypeError):
   541→                    continue
   542→        except OSError:
   543→            return {str(h): 0 for h in since_hours}
   544→        return {str(h): counts[h] for h in since_hours}
   545→

   550→    def _hydrate_last_hash_postgres(self) -> str:
   551→        if self._last_hash_cached is not None:
   552→            return self._last_hash_cached
   553→        with self._conn.cursor() as cur:
   554→            cur.execute(
   555→                "SELECT raw_hash FROM audit_records ORDER BY id DESC LIMIT 1"
   556→            )
   557→            row = cur.fetchone()
   558→            self._last_hash_cached = row[0] if row else GENESIS
   559→            return self._last_hash_cached

   561→    def _log_postgres(self, payload: dict) -> str:
   562→        audit_id = f"aud_{uuid.uuid4().hex[:16]}"
   563→        ts = datetime.now(timezone.utc).isoformat()
   564→        # Body — the full audit record minus the hash fields (which are
   565→        # computed below). Stored as JSONB so consumers can query arbitrary
   566→        # paths. Note the body includes audit_id, timestamp, model_version
   567→        # + every key in payload (per Track D: request, decision,
   568→        # decision_source, cost_breakdown, reason_codes, mandate_*,
   569→        # bh_purpose_code, device_id, user_id, breach_note, rule_fired,
   570→        # degraded, features_used, latency_ms, case_id, probability).
   571→        body = {
   572→            "audit_id": audit_id,
   573→            "timestamp": ts,
   574→            "model_version": self.model_version,
   575→            **payload,
   576→        }
   577→        prev_hash = self._hydrate_last_hash_postgres()
   578→        raw_hash = self._hash(body, prev=prev_hash)
   579→        with self._lock:
   580→            with self._conn.cursor() as cur:
   581→                cur.execute(
   582→                    """
   583→                    INSERT INTO audit_records
   584→                      (audit_id, body, raw_hash, prev_hash, created_at,
   585→                       model_version, mandate_type, bh_purpose_code,
   586→                       device_id, user_id)
   587→                    VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
   588→                    RETURNING id
   589→                    """,
   590→                    (
   591→                        audit_id,
   592→                        json.dumps(body, default=str),
   593→                        raw_hash,
   594→                        prev_hash,
   595→                        ts,
   596→                        self.model_version,
   597→                        payload.get("mandate_type"),
   598→                        payload.get("bh_purpose_code"),
   599→                        payload.get("device_id"),
   600→                        payload.get("user_id"),
   601→                    ),
   602→                )
   603→                # Day 2 Track H — feed the new record to the Merkle
   604→                # sealer so it tracks pending intervals in real time.
   605→                # ``sealer.add`` is a no-op if interval_size hasn't been
   606→                # reached yet (it just appends to the pending list). When
   607→                # the threshold trips, ``seal()`` runs inline + commits
   608→                # the interval row + the per-record backfill in the same
   609→                # transaction (atomic with the audit INSERT above).
   610→                record_id_row = cur.fetchone()
   611→                record_id = record_id_row[0] if record_id_row else None
   612→                self._conn.commit()
   613→            self._last_hash_cached = raw_hash
   614→        if record_id is not None and self.sealer is not None:
   615→            try:
   616→                self.sealer.add(record_id, raw_hash)
   617→            except Exception as e:  # pragma: no cover — best-effort
   618→                # Merkle sealing must never break the audit write (the
   619→                # per-record hash chain is the foundation; Merkle is a
   620→                # coarser layer that can catch up via seal_interval()
   621→                # later). Log + continue.
   622→                print(
   623→                    f"[audit] merkle sealer.add failed: "
   624→                    f"{type(e).__name__}: {e}",
   625→                    file=sys.stderr,
   626→                )
   627→        return audit_id
   628→
   629→    def _read_postgres(self, audit_id: str) -> dict | None:
   630→        with self._conn.cursor() as cur:
   631→            cur.execute(
   632→                """
   633→                SELECT body, raw_hash, prev_hash
   634→                FROM audit_records WHERE audit_id = %s
   635→                """,
   636→                (audit_id,),
   637→            )
   638→            row = cur.fetchone()
   639→        if row is None:
   640→            return None
   641→        body, raw_hash, prev_hash = row
   642→        # Merge hash fields back into the body so the file-mode test
   643→        # assertions (which read ``rec["raw_hash"]``) work unchanged.
   644→        body = dict(body) if isinstance(body, dict) else json.loads(body)
   645→        body["raw_hash"] = raw_hash
   646→        body["previous_hash"] = prev_hash
   647→        return body

   649→    def _verify_chain_postgres(self) -> tuple[bool, int, str]:
   650→        expected_prev = GENESIS
   651→        n = 0
   652→        with self._conn.cursor() as cur:
   653→            cur.execute(
   654→                """
   655→                SELECT audit_id, body, raw_hash, prev_hash
   656→                FROM audit_records ORDER BY id ASC
   657→                """
   658→            )
   659→            rows = cur.fetchall()
   660→        for audit_id, body, raw_hash, prev_hash in rows:
   661→            body_dict = dict(body) if isinstance(body, dict) else json.loads(body)
   662→            stored_prev = prev_hash
   663→            stored_raw = raw_hash
   664→            want_prev = expected_prev
   665→            want_raw = self._hash(body_dict, prev=want_prev)
   666→            ok = stored_prev == want_prev and stored_raw == want_raw
   667→            if not ok:
   668→                return False, n, audit_id
   669→            expected_prev = stored_raw
   670→            n += 1
   671→        return True, n, ""

   677→    def _log_file(self, payload: dict) -> str:
   678→        audit_id = str(uuid.uuid4())
   679→        base = {
   680→            "audit_id": audit_id,
   681→            "timestamp": datetime.now(timezone.utc).isoformat(),
   682→            "model_version": self.model_version,
   683→            **payload,
   684→        }
   685→        with self._lock:
   686→            base["previous_hash"] = self.last_hash
   687→            base["raw_hash"] = self._hash(base)
   688→            self.path.parent.mkdir(parents=True, exist_ok=True)
   689→            with self.path.open("a") as f:
   690→                self._index[audit_id] = f.tell()
   691→                f.write(json.dumps(base, default=str) + "\n")
   692→            self.last_hash = base["raw_hash"]
   693→        return audit_id

   695→    def _read_file(self, audit_id: str) -> dict | None:
   696→        offset = self._index.get(audit_id)
   697→        if offset is None or not self.path.exists():
   698→            return None
   699→        with self.path.open() as f:
   700→            f.seek(offset)
   701→            return json.loads(f.readline())

   703→    def _verify_chain_file(self) -> tuple[bool, int, str]:
   704→        expected_prev = GENESIS
   705→        n = 0
   706→        if not self.path.exists():
   707→            return True, 0, ""
   708→        with self.path.open() as f:
   709→            for line in f:
   710→                rec = json.loads(line)
   711→                stored_hashes = {k: rec.get(k) for k in self.HASH_FIELDS}
   712→                body = {k: v for k, v in rec.items() if k not in self.HASH_FIELDS}
   713→                want_prev = expected_prev
   714→                want_raw = self._hash(body, prev=want_prev)
   715→                hashes_ok = (
   716→                    stored_hashes["previous_hash"] == want_prev
   717→                    and stored_hashes["raw_hash"] == want_raw
   718→                )
   719→                if not hashes_ok:
   720→                    return False, n, rec.get("audit_id", "?")
   721→                expected_prev = stored_hashes["raw_hash"]
   722→                n += 1
   723→        return True, n, ""

   729→    @staticmethod
   730→    def _hash(record: dict, prev: str | None = None) -> str:
   731→        body = {k: v for k, v in record.items() if k not in AuditLogger.HASH_FIELDS}
   732→        prev_hash = prev if prev is not None else record.get("previous_hash", GENESIS)
   733→        return hashlib.sha256((canonical(body) + prev_hash).encode()).hexdigest()
