    59→def test_calibrate_probabilities_bahnsen_eq6_rescales_when_undersampled():
    60→    """If S50 under-sampling inflates the positive prior from 0.05 to 0.50,
    61→    a model predicting p=0.5 should recalibrate to 0.05 (the original prior)."""
    62→    p_orig = 0.05
    63→    p_und = 0.50
    64→    out = calibrate_probabilities(0.5, p_orig, p_und)
    65→    assert abs(out - 0.05) < 1e-9
    66→    # Edge: p_und=0 → division-by-zero guard → returns 0 (not +inf)
    67→    assert calibrate_probabilities(0.9, 0.05, 0.0) == 0.0
    68→    # Edge: p_orig=0 → all calibrated probabilities are 0
    69→    assert calibrate_probabilities([0.5, 0.9], 0.0, 0.5) == [0.0, 0.0]
    70→    # Edge: clip inputs outside [0, 1] before applying the ratio
    71→    assert calibrate_probabilities(1.5, 0.5, 0.5) == 1.0  # clipped to 1, ratio=1

   275→
   276→    This is the headline test from Day 1 Track C: before this change, the
   277→    cost-optimizer was stored as ``policy_hint`` but the actual decision came

   395→# ---------------------------------------------------------------------------
   396→# Day 4 Track N — V3 §11.6 5-way intervention policy (Bahnsen Eq.(5))
   397→# ---------------------------------------------------------------------------
   398→# The 5-way intervention layer extends Track C's 3-way optimal_decision() with
   399→# the full intervention set {ship, otp_verify, partial_cod, address_check,
   400→# hold}. Per Bahnsen 2013 BMR Eq.(5), the FN cost is the per-transaction
   401→# amount (NOT a constant), and per-intervention effectiveness rates come from
   402→# the Pragma 2025 RTO-mitigation benchmark (OTP 0.82, partial COD 0.65,
   403→# address check 0.45). The argmin over the 5 interventions gives the
   404→# cost-optimal next-step recommendation surfaced in /risk/score's
   405→# ``intervention`` field.
   406→#
   407→# Key cost-model property with DEFAULT weights: otp_verify dominates the
   408→# "soft" interventions for any p×amt ≥ 6 INR (because c_otp=5 is cheap AND
   409→# eff_otp=0.82 is the highest effectiveness). For p×amt < 6 INR, ship wins
   410→# (the OTP fee exceeds the residual RTO loss). Hold/partial_cod/address_check
   411→# become optimal when their weights are re-tuned (e.g. higher c_otp in a
   412→# market where SMS is expensive, or lower c_hold + residual ship rate when
   413→# human review is cheap). The tests below verify both the default-weight
   414→# behaviour AND the re-tuned crossover so the cost-model's truthful answer is
   415→# documented — the argmin is a function of (p, amt, weights), not a constant.
   416→# ---------------------------------------------------------------------------
   417→
   418→
   419→def test_optimal_intervention_ship_at_low_p():
   420→    """At very low p × low amount, the per-amount FN cost is smaller than the
   421→    OTP verification fee, so ship (baseline, no intervention) wins.
   422→
   423→    Math: ship = p·amt; otp_verify = c_otp + (1 − eff_otp)·p·amt.
   424→    Ship wins iff p·amt < c_otp / eff_otp = 5 / 0.82 ≈ 6.10 INR.
   425→    Use p=0.01, amt=600 → p·amt = 6 → ship wins (boundary case).
   426→    """
   427→    intervention, costs = optimal_intervention(0.01, 600)
   428→    assert intervention == "ship", (
   429→        f"expected ship at p=0.01 amt=600 (p*amt=6 < 6.10 threshold), "
   430→        f"got {intervention} with costs {costs}"
   431→    )
   432→    # Full 5-way breakdown present
   433→    assert set(costs.keys()) == set(INTERVENTIONS)
   434→    # Ship's cost equals p*amt (Bahnsen Eq.(5): FN = amount)
   435→    assert abs(costs["ship"] - 0.01 * 600) < 0.01
   436→

   438→def test_optimal_intervention_otp_at_medium_p():
   439→    """At medium p × medium amount, the OTP fee is small relative to the
   440→    residual RTO loss, so otp_verify wins.
   441→
   442→    Use p=0.4, amt=12400 → p·amt = 4960 INR.
   443→    otp_verify cost = 5 + 0.18·4960 = 5 + 892.8 = 897.8 INR — far below ship
   444→    (4960) and below partial_cod / address_check / hold. argmin = otp_verify.
   445→    """
   446→    intervention, costs = optimal_intervention(0.4, 12400)
   447→    assert intervention == "otp_verify", (
   448→        f"expected otp_verify at p=0.4 amt=12400, got {intervention}: {costs}"
   449→    )
   450→    # Verify the OTP cost math (Bahnsen Eq.(5) per-amount FN).
   451→    expected_otp = 5.0 + (1 - 0.82) * 0.4 * 12400
   452→    assert abs(costs["otp_verify"] - expected_otp) < 0.05
   453→    # Verify ship cost = full amount at risk

   522→def test_optimal_decision_per_amount_fn_cost_flag():
   523→    """Day 4 Track N — optimal_decision() now accepts an optional amount_inr
   524→    that overrides the constant c_fn with the per-transaction amount
   525→    (Bahnsen Eq.(5)). At the same probability, a low-amount order may ACCEPT
   526→    while a high-amount order REJECTs — this is the headline of the Bahnsen
   527→    2013 paper (FN cost ≠ constant).
   528→    """
   529→    p = 0.4
   530→    # Low amount: c_fn = 600 (constant fallback) → REVIEW
   531→    # Same p with amount=600 (Bahnsen Eq.(5)): cost_accept = 0.4 * 600 = 240
   532→    # vs cost_review = 5 + 0.6*50 + 0.4*0.18*600 = 5+30+43.2 = 78.2 → REVIEW
   533→    d_const, c_const = optimal_decision(p)  # constant c_fn=600
   534→    d_amt_low, c_low = optimal_decision(p, amount_inr=600)
   535→    # The constant path and the per-amount path should agree when amount == c_fn
   536→    assert d_const == d_amt_low, (
   537→        f"constant c_fn=600 vs amount_inr=600 should give same decision: "
   538→        f"{d_const} vs {d_amt_low}"
   539→    )
   540→    # Same p with amount=52000 (Bahnsen Eq.(5)): cost_accept = 0.4*52000 = 20800
   541→    # vs cost_reject = 0.6*1000 = 600 → REJECT wins (block is cheaper than ship)
   542→    d_amt_high, c_high = optimal_decision(p, amount_inr=52000)
   543→    assert d_amt_high == "REJECT", (
   544→        f"at p=0.4 amt=52000 (per-amount FN cost), expected REJECT, "
   545→        f"got {d_amt_high}: {c_high}"
   546→    )
   547→    # The per-amount FN cost produces a DIFFERENT decision than the constant
   548→    # c_fn=600 default — the headline of the Bahnsen 2013 paper.
   549→    assert d_amt_low != d_amt_high, (
   550→        "per-amount FN cost should change the decision at the same probability "
   551→        "(Bahnsen Eq.(5) headline: FN cost ≠ constant)"
   552→    )
