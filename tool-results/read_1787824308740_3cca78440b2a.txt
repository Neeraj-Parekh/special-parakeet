    61→    61→class MerkleSealer:
    62→    62→    """Merkle interval sealing per V3 §10.3.
    63→    63→
    64→    64→    Every N records (default 1000) or T seconds (default 3600), compute
    65→    65→    a Merkle root of the batch's ``raw_hash`` leaves, chain it to the
    66→    66→    previous interval's root, store the interval.
    67→    67→
    68→    68→    This is ADDITIONAL to the per-record hash chain (Track E Day 2) —
    69→    69→    it's a coarser tamper-evidence layer that enables O(log N) proof
    70→    70→    generation per record (vs O(N) full-chain recompute via
    71→    71→    ``AuditLogger.verify_chain()``). The per-record ``raw_hash`` is the
    72→    72→    leaf; ``audit_merkle_intervals.merkle_root`` is the interval root;
    73→    73→    ``audit_merkle_intervals.prev_interval_root`` chains intervals
    74→    74→    together so cross-interval tampering is detected (same model as
    75→    75→    Track E's per-record ``prev_hash``).
    76→    76→
    77→    77→    Source: SoK (Mao 2026) capability ``recommend_layered_defenses``
    78→    78→    layer 5 — "market & compliance monitoring with tamper-evident audit
    79→    79→    trails"; ``audit_agent_mandate_scoping`` (the dual-control override
    80→    80→    in routes.py consumes this tamper-evidence layer so both admin
    81→    81→    signatures are anchored to a verifiable root, not just the live
    82→    82→    hash chain tip).
    83→    83→
    84→    84→    File mode is a no-op: the per-record hash chain is enough for tests
    85→    85→    + local dev without Postgres. Merkle is a Postgres-mode enhancement.
    86→    86→    """
    87→    87→
    88→    88→    def __init__(
    89→    89→        self,
    90→    90→        conn=None,
    91→    91→        interval_size: int = 1000,
    92→    92→        interval_seconds: int = 3600,
    93→    93→    ):
    94→    94→        # Postgres connection (None in file mode → no-op). Shared with
    95→    95→        # the parent AuditLogger's connection — no extra pool needed.
    96→    96→        self.conn = conn
    97→    97→        self.interval_size = interval_size
    98→    98→        self.interval_seconds = interval_seconds
    99→    99→        # Pending (record_id, raw_hash) pairs since the last seal.
   100→   100→        # Bounded by interval_size; flushed to audit_merkle_intervals on
   101→   101→        # add() when len(self._pending) >= interval_size or by an
   102→   102→        # explicit seal_interval() call (e.g. from a cron / shutdown hook).
   103→   103→        self._pending: list[tuple[int, str]] = []
   104→   104→        # First-record timestamp so the time-based seal trigger can compute
   105→   105→        # elapsed since the interval opened (independent of record count).
   106→   106→        self._interval_started_at: datetime | None = None
   107→   107→
   108→   108→    # ------------------------------------------------------------------ #
   109→   109→    # Public API — add / seal / proof                                    #
   110→   110→    # ------------------------------------------------------------------ #
   111→   111→
   112→   112→    def add(self, record_id: int, raw_hash: str) -> dict | None:
   113→   113→        """Add a record to the pending interval. Seal if threshold reached.
   114→   114→
   115→   115→        Called by ``AuditLogger._log_postgres`` after every INSERT so the
   116→   116→        sealer tracks new records in real time. The first record's
   117→   117→        timestamp seeds ``_interval_started_at`` for the time-based
   118→   118→        trigger (so an idle period after a partial interval still flushes
   119→   119→        on the next ``seal_interval()`` even with no fresh traffic).
   120→   120→        """
   121→   121→        if self.conn is None:
   122→   122→            # File mode — no-op (per-record hash chain is enough there).
   123→   123→            return None
   124→   124→        if not self._pending:
   125→   125→            self._interval_started_at = datetime.now(timezone.utc)
   126→   126→        self._pending.append((record_id, raw_hash))
   127→   127→        # Seal when EITHER threshold trips (count OR elapsed). The count
   128→   128→        # trigger fires inline; the time trigger is checked here too so
   129→   129→        # a long quiet gap after a partial batch doesn't leave the
   130→   130→        # interval open forever (the cron hook calls seal_interval() too).
   131→   131→        if len(self._pending) >= self.interval_size:
   132→   132→            return self.seal()
   133→   133→        if self._interval_started_at is not None:
   134→   134→            elapsed = (datetime.now(timezone.utc) - self._interval_started_at).total_seconds()
   135→   135→            if elapsed >= self.interval_seconds and len(self._pending) > 0:
   136→   136→                return self.seal()
   137→   137→        return None
   138→   138→
   139→   139→    def seal(self) -> dict | None:
   140→   140→        """Seal the current pending batch into a Merkle interval.
   141→   141→
   142→   142→        Returns the interval metadata dict (merkle_root, prev_root,
   143→   143→        leaf_count) on success, None if there's nothing to seal or the
   144→   144→        sealer is in file mode (conn is None). After sealing, the
   145→   145→        pending batch is cleared + the next add() starts a fresh
   146→   146→        interval. Idempotent — calling seal() with an empty pending
   147→   147→        list is a no-op.
   148→   148→        """
   149→   149→        if not self._pending or self.conn is None:
   150→   150→            return None
   151→   151→        # Compute Merkle root of the raw_hash leaves. Padding to a power
   152→   152→        # of 2 keeps the tree balanced so the proof builder's index
   153→   153→        # arithmetic (sibling_idx = idx ^ 1) is consistent across levels.
   154→   154→        leaves = [h for _, h in self._pending]
   155→   155→        merkle_root = self._merkle_root(leaves)
   156→   156→        # Get previous interval's root (None on first interval → genesis).
   157→   157→        with self.conn.cursor() as cur:
   158→   158→            cur.execute(
   159→   159→                "SELECT merkle_root FROM audit_merkle_intervals "
   160→   160→                "ORDER BY interval_id DESC LIMIT 1"
   161→   161→            )
   162→   162→            prev = cur.fetchone()
   163→   163→        prev_root = prev[0] if prev else GENESIS
   164→   164→        # Insert the interval row + backfill the per-record columns
   165→   165→        # (interval_id + interval_position) so the proof builder can
   166→   166→        # locate a record's interval + leaf index in O(1).
   167→   167→        with self.conn.cursor() as cur:
   168→   168→            cur.execute(
   169→   169→                """
   170→   170→                INSERT INTO audit_merkle_intervals
   171→   171→                  (start_record_id, end_record_id, merkle_root,
   172→   172→                   prev_interval_root, leaf_count)
   173→   173→                VALUES (%s, %s, %s, %s, %s)
   174→   174→                RETURNING interval_id
   175→   175→                """,
   176→   176→                (
   177→   177→                    self._pending[0][0],
   178→   178→                    self._pending[-1][0],
   179→   179→                    merkle_root,
   180→   180→                    prev_root,
   181→   181→                    len(leaves),
   182→   182→                ),
   183→   183→            )
   184→   184→            interval_id = cur.fetchone()[0]
   185→   185→            # currval('audit_merkle_intervals_interval_id_seq') is the
   186→   186→            # just-inserted row's SERIAL value; using it in the UPDATE
   187→   187→            # avoids passing interval_id back from Python (atomic within
   188→   188→            # the same transaction). Per-record backfill is O(M) where
   189→   189→            # M = interval_size (default 1000) — cheap.
   190→   190→            for pos, (rid, _) in enumerate(self._pending):
   191→   191→                cur.execute(
   192→   192→                    """
   193→   193→                    UPDATE audit_records
   194→   194→                       SET interval_id = %s, interval_position = %s
   195→   195→                     WHERE id = %s
   196→   196→                    """,
   197→   197→                    (interval_id, pos, rid),
   198→   198→                )
   199→   199→            self.conn.commit()
   200→   200→        self._pending = []
   201→   201→        self._interval_started_at = None
   202→   202→        return {
   203→   203→            "interval_id": interval_id,
   204→   204→            "merkle_root": merkle_root,
   205→   205→            "prev_root": prev_root,
   206→   206→            "leaf_count": len(leaves),
   207→   207→        }
   208→   208→
   209→   209→    @staticmethod
   210→   210→    def _merkle_root(leaves: list[str]) -> str:
   211→   211→        """Compute Merkle root from leaf hashes. Pad to power of 2.
   212→   212→
   213→   213→        Padding uses the LAST leaf's hash repeated (RFC 6962-style) so
   214→   214→        the tree is balanced without introducing a synthetic zero-leaf
   215→   215→        that a verifier would need to special-case. Empty leaf list →
   216→   216→        GENESIS (the same ``"0"*64`` constant Track E's hash chain uses
   217→   217→        as the genesis ``prev_hash``).
   218→   218→        """
   219→   219→        if not leaves:
   220→   220→            return GENESIS
   221→   221→        # Pad to next power of 2.
   222→   222→        size = 1
   223→   223→        while size < len(leaves):
   224→   224→            size *= 2
   225→   225→        padded = leaves + [leaves[-1]] * (size - len(leaves))
   226→   226→        # Build tree level-by-level: parent = sha256(left + right).
   227→   227→        level = padded
   228→   228→        while len(level) > 1:
   229→   229→            next_level = []
   230→   230→            for i in range(0, len(level), 2):
   231→   231→                combined = hashlib.sha256(
   232→   232→                    (level[i] + level[i + 1]).encode()
   233→   233→                ).hexdigest()
   234→   234→                next_level.append(combined)
   235→   235→            level = next_level
   236→   236→        return level[0]
   237→   237→
   238→   238→    def proof(self, record_id: int) -> dict | None:
   239→   239→        """Generate Merkle proof for a record: path from leaf to interval root.
   240→   240→
   241→   241→        Returns None if the record doesn't exist, has no interval_id
   242→   242→        (i.e. the sealer hasn't reached the threshold + the cron seal
   243→   243→        hasn't fired), or the sealer is in file mode. The proof dict
   244→   244→        contains the sibling hashes from leaf to root, the interval's
   245→   245→        root + prev_interval_root (chain anchor), leaf_count, sealed_at
   246→   246→        — enough to verify inclusion in O(log N) without recomputing
   247→   247→        the whole tree.
   248→   248→        """
   249→   249→        if self.conn is None:
   250→   250→            return None
   251→   251→        # Get the record's interval + position.
   252→   252→        with self.conn.cursor() as cur:
   253→   253→            cur.execute(
   254→   254→                "SELECT interval_id, interval_position, raw_hash "
   255→   255→                "FROM audit_records WHERE id = %s",
   256→   256→                (record_id,),
   257→   257→            )
   258→   258→            row = cur.fetchone()
   259→   259→        if not row or row[0] is None:
   260→   260→            # Record not found, OR interval not yet sealed.
   261→   261→            return None
   262→   262→        interval_id, position, leaf_hash = row
   263→   263→        # Get all leaves in this interval (the proof builder needs to
   264→   264→        # descend from the leaf position; siblings at each level are
   265→   265→        # looked up by index — pre-fetched once per proof for simplicity
   266→   266→        # rather than one query per tree level).
   267→   267→        with self.conn.cursor() as cur:
   268→   268→            cur.execute(
   269→   269→                "SELECT raw_hash FROM audit_records WHERE interval_id = %s "
   270→   270→                "ORDER BY interval_position",
   271→   271→                (interval_id,),
   272→   272→            )
   273→   273→            leaves = [r[0] for r in cur.fetchall()]
   274→   274→        # Build proof: at each tree level, the sibling's hash + whether
   275→   275→        # it's the left or right child of the parent node. Pad leaves
   276→   276→        # to a power of 2 (same padding rule as _merkle_root) so the
   277→   277→        # index arithmetic is consistent.
   278→   278→        size = 1
   279→   279→        while size < len(leaves):
   280→   280→            size *= 2
   281→   281→        level = leaves + [leaves[-1]] * (size - len(leaves))
   282→   282→        proof: list[dict] = []
   283→   283→        idx = position
   284→   284→        while len(level) > 1:
   285→   285→            sibling_idx = idx ^ 1  # XOR 1: pairs (0,1), (2,3), ...
   286→   286→            if sibling_idx < len(level):
   287→   287→                proof.append(
   288→   288→                    {
   289→   289→                        "position": "right" if sibling_idx > idx else "left",
   290→   290→                        "hash": level[sibling_idx],
   291→   291→                    }
   292→   292→                )
   293→   293→            else:
   294→   294→                # Padding sibling (RFC 6962-style last-leaf-repeat).
   295→   295→                proof.append({"position": "right", "hash": level[-1]})
   296→   296→            # Compute next level (parent hashes).
   297→   297→            next_level = []
   298→   298→            for i in range(0, len(level), 2):
   299→   299→                combined = hashlib.sha256(
   300→   300→                    (level[i] + level[i + 1 if i + 1 < len(level) else i]).encode()
   301→   301→                ).hexdigest()
   302→   302→                next_level.append(combined)
   303→   303→            level = next_level
   304→   304→            idx //= 2
   305→   305→        # Get the interval's metadata for the chain anchor.
   306→   306→        with self.conn.cursor() as cur:
   307→   307→            cur.execute(
   308→   308→                """
   309→   309→                SELECT merkle_root, prev_interval_root, leaf_count, sealed_at
   310→   310→                  FROM audit_merkle_intervals WHERE interval_id = %s
   311→   311→                """,
   312→   312→                (interval_id,),
   313→   313→            )
   314→   314→            interval = cur.fetchone()
   315→   315→        if not interval:
   316→   316→            return None
   317→   317→        return {
   318→   318→            "record_id": record_id,
   319→   319→            "leaf_hash": leaf_hash,
   320→   320→            "interval_id": interval_id,
   321→   321→            "position": position,
   322→   322→            "proof": proof,
   323→   323→            "merkle_root": interval[0],
   324→   324→            "prev_interval_root": interval[1],
   325→   325→            "leaf_count": interval[2],
   326→   326→            "sealed_at": interval[3].isoformat() if interval[3] else None,
   327→   327→        }

   330→   330→class AuditLogger:
   331→   331→    """Append-only audit log, tamper-evident hash chain, O(1) indexed reads.
   332→   332→
   333→   333→    Dual-mode (Day 2 Track E):
   334→   334→      * Postgres mode (``settings.database_url`` set): INSERT into
   335→   335→        ``audit_records``; reads via ``SELECT ... WHERE audit_id = %s``;
   336→   336→        chain verification via ``SELECT audit_id, body, raw_hash, prev_hash
   337→   337→        FROM audit_records ORDER BY id ASC``.
   338→   338→      * File mode (no DSN): the original JSONL append + byte-offset index.
   339→   339→
   340→   340→    The hash-chain math is identical between modes (``canonical(body) +
   341→   341→    prev_hash``) so a verifier computes the same hash either way.
   342→   342→
   343→   343→    Day 2 Track H — Merkle interval sealing (V3 §10.3). In Postgres
   344→   344→    mode, ``__init__`` constructs a ``MerkleSealer`` sharing the same
   345→   345→    connection; ``_log_postgres`` calls ``sealer.add(record_id, raw_hash)``
   346→   346→    after every INSERT so the sealer tracks new records in real time.
   347→   347→    File mode skips Merkle (the per-record hash chain is enough there).

   384→   384→            self._index: dict[str, int] = {}
   385→   385→            self.last_hash = GENESIS
   386→   386→            self.sealer = None
   387→   387→            if self.path.exists():
   388→   388→                with self.path.open() as f:
   389→   389→                    for offset, line in enumerate(f):
   390→   390→                        rec = json.loads(line)
   391→   391→                        self._index[rec.get("audit_id", "")] = offset
   392→   392→                        self.last_hash = rec.get("raw_hash", self.last_hash)

   414→   414→        return self._verify_chain_file()

   416→   416→    def tail(self, limit: int = 300) -> list[dict]:
   417→   417→        """Return the last ``limit`` audit records (most-recent last).
   418→   418→
   419→   419→        Used by the ``/v1/models/drift`` endpoint (recent feature distributions
   420→   420→        for PSI) and the ``/v1/compliance/audit-export`` CSV export. In
   421→   421→        Postgres mode this is a ``SELECT ... ORDER BY id DESC LIMIT %s``;
   422→   422→        in file mode it's the tail of the JSONL.
   423→   423→        """
   424→   424→        if self._conn is not None:
   425→   425→            with self._conn.cursor() as cur:
   426→   426→                cur.execute(
   427→   427→                    "SELECT body FROM audit_records ORDER BY id DESC LIMIT %s",
   428→   428→                    (limit,),
   429→   429→                )

   499→   499→    def usage_counts(self, since_hours: tuple[int, ...] = (24, 168, 720)) -> dict:
   500→   500→        """Per-window audit-record counts for the ``/v1/usage`` metering
   501→   501→        endpoint. Returns ``{str(h): count, ...}`` for each h in
   502→   502→        ``since_hours`` (default last 24h / 7d / 30d).
   503→   503→
   504→   504→        Postgres mode: ``SELECT count(*) FROM audit_records WHERE
   505→   505→        created_at > now() - interval '%s hours'``. File mode: scan the
   506→   506→        JSONL + filter by the record's ``timestamp`` field. Cheap on
   507→   507→        either side — the audit table is indexed on ``created_at DESC``
   508→   508→        (Track E migration 001), so the count is an index-only scan.
   509→   509→        """
   510→   510→        out: dict[str, int] = {}
   511→   511→        if self._conn is not None:
   512→   512→            with self._conn.cursor() as cur:
   513→   513→                for h in since_hours:
   514→   514→                    cur.execute(
   515→   515→                        "SELECT count(*) FROM audit_records "
   516→   516→                        "WHERE created_at > now() - interval '%s hours'",
   517→   517→                        (h,),
   518→   518→                    )
   519→   519→                    out[str(h)] = int(cur.fetchone()[0])
   520→   520→            return out
   521→   521→        # File mode — scan + filter by timestamp.
   522→   522→        if not self.path.exists():
   523→   523→            return {str(h): 0 for h in since_hours}
   524→   524→        now = datetime.now(timezone.utc)
   525→   525→        cutoffs = {h: now.timestamp() - h * 3600 for h in since_hours}
   526→   526→        counts = {h: 0 for h in since_hours}
   527→   527→        try:
   528→   528→            for line in self.path.read_text().splitlines():
   529→   529→                if not line.strip():
   530→   530→                    continue
   531→   531→                try:
   532→   532→                    rec = json.loads(line)
   533→   533→                    ts = rec.get("timestamp")
   534→   534→                    if not ts:
   535→   535→                        continue
   536→   536→                    parsed = datetime.fromisoformat(ts).timestamp()
   537→   537→                    for h in since_hours:
   538→   538→                        if parsed >= cutoffs[h]:
   539→   539→                            counts[h] += 1
   540→   540→                except (json.JSONDecodeError, ValueError, TypeError):
   541→   541→                    continue
   542→   542→        except OSError:
   543→   543→            return {str(h): 0 for h in since_hours}
   544→   544→        return {str(h): counts[h] for h in since_hours}

   561→   561→    def _log_postgres(self, payload: dict) -> str:
   562→   562→        audit_id = f"aud_{uuid.uuid4().hex[:16]}"
   563→   563→        ts = datetime.now(timezone.utc).isoformat()
   564→   564→        # Body — the full audit record minus the hash fields (which are
   565→   565→        # computed below). Stored as JSONB so consumers can query arbitrary
   566→   566→        # paths. Note the body includes audit_id, timestamp, model_version
   567→   567→        # + every key in payload (per Track D: request, decision,
   568→   568→        # decision_source, cost_breakdown, reason_codes, mandate_*,
   569→   569→        # bh_purpose_code, device_id, user_id, breach_note, rule_fired,
   570→   570→        # degraded, features_used, latency_ms, case_id, probability).
   571→   571→        body = {
   572→   572→            "audit_id": audit_id,
   573→   573→            "timestamp": ts,
   574→   574→            "model_version": self.model_version,
   575→   575→            **payload,
   576→   576→        }
   577→   577→        prev_hash = self._hydrate_last_hash_postgres()
   578→   578→        raw_hash = self._hash(body, prev=prev_hash)
   579→   579→        with self._lock:
   580→   580→            with self._conn.cursor() as cur:
   581→   581→                cur.execute(
   582→   582→                    """
   583→   583→                    INSERT INTO audit_records
   584→   584→                      (audit_id, body, raw_hash, prev_hash, created_at,
   585→   585→                       model_version, mandate_type, bh_purpose_code,
   586→   586→                       device_id, user_id)
   587→   587→                    VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
   588→   588→                    RETURNING id
   589→   589→                    """,
   590→   590→                    (
   591→   591→                        audit_id,
   592→   592→                        json.dumps(body, default=str),
   593→   593→                        raw_hash,
   594→   594→                        prev_hash,
   595→   595→                        ts,
   596→   596→                        self.model_version,
   597→   597→                        payload.get("mandate_type"),
   598→   598→                        payload.get("bh_purpose_code"),
   599→   599→                        payload.get("device_id"),
   600→   600→                        payload.get("user_id"),
   601→   601→                    ),
   602→   602→                )
   603→   603→                # Day 2 Track H — feed the new record to the Merkle
   604→   604→                # sealer so it tracks pending intervals in real time.
   605→   605→                # ``sealer.add`` is a no-op if interval_size hasn't been
   606→   606→                # reached yet (it just appends to the pending list). When
   607→   607→                # the threshold trips, ``seal()`` runs inline + commits
   608→   608→                # the interval row + the per-record backfill in the same
   609→   609→                # transaction (atomic with the audit INSERT above).
   610→   610→                record_id_row = cur.fetchone()
   611→   611→                record_id = record_id_row[0] if record_id_row else None
   612→   612→                self._conn.commit()
   613→   613→            self._last_hash_cached = raw_hash
   614→   614→        if record_id is not None and self.sealer is not None:
   615→   615→            try:
   616→   616→                self.sealer.add(record_id, raw_hash)
   617→   617→            except Exception as e:  # pragma: no cover — best-effort
   618→   618→                # Merkle sealing must never break the audit write (the
   619→   619→                # per-record hash chain is the foundation; Merkle is a
   620→   620→                # coarser layer that can catch up via seal_interval()
   621→   621→                # later). Log + continue.
   622→   622→                print(
   623→   623→                    f"[audit] merkle sealer.add failed: "
   624→   624→                    f"{type(e).__name__}: {e}",
   625→   625→                    file=sys.stderr,
   626→   626→                )
   627→   627→        return audit_id

   629→   629→    def _read_postgres(self, audit_id: str) -> dict | None:
   630→   630→        with self._conn.cursor() as cur:
   631→   631→            cur.execute(
   632→   632→                """
   633→   633→                SELECT body, raw_hash, prev_hash
   634→   634→                FROM audit_records WHERE audit_id = %s
   635→   635→                """,
   636→   636→                (audit_id,),
   637→   637→            )
   638→   638→            row = cur.fetchone()
   639→   639→        if row is None:
   640→   640→            return None
   641→   641→        body, raw_hash, prev_hash = row
   642→   642→        # Merge hash fields back into the body so the file-mode test
   643→   643→        # assertions (which read ``rec["raw_hash"]``) work unchanged.
   644→   644→        body = dict(body) if isinstance(body, dict) else json.loads(body)
   645→   645→        body["raw_hash"] = raw_hash
   646→   646→        body["previous_hash"] = prev_hash
   647→   647→        return body

   649→   649→    def _verify_chain_postgres(self) -> tuple[bool, int, str]:
   650→   650→        expected_prev = GENESIS
   651→   651→        n = 0
   652→   652→        with self._conn.cursor() as cur:
   653→   653→            cur.execute(
   654→   654→                """
   655→   655→                SELECT audit_id, body, raw_hash, prev_hash
   656→   656→                FROM audit_records ORDER BY id ASC
   657→   657→                """
   658→   658→            )
   659→   659→            rows = cur.fetchall()
   660→   660→        for audit_id, body, raw_hash, prev_hash in rows:
   661→   661→            body_dict = dict(body) if isinstance(body, dict) else json.loads(body)
   662→   662→            stored_prev = prev_hash
   663→   663→            stored_raw = raw_hash
   664→   664→            want_prev = expected_prev
   665→   665→            want_raw = self._hash(body_dict, prev=want_prev)
   666→   666→            ok = stored_prev == want_prev and stored_raw == want_raw
   667→   667→            if not ok:
   668→   668→                return False, n, audit_id
   669→   669→            expected_prev = stored_raw
   670→   670→            n += 1
   671→   671→        return True, n, ""
