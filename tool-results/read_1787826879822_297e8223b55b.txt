   322→   322→@skip_no_redis
   323→   323→def test_consumer_processes_message():
   324→   324→    """Produce a message, run consumer in a thread, assert handler called."""
   325→   325→    if _skip_if_redis_unreachable():
   326→   326→        pytest.skip("Redis not reachable at " + _REDIS_URL)
   327→   327→    import redis  # type: ignore[import-not-found]
   328→   328→
   329→   329→    from src.stream.consumer import StreamConsumer
   330→   330→
   331→   331→    raw = redis.from_url(_REDIS_URL, decode_responses=True)
   332→   332→    # Clean the stream + the consumer group from any prior run.
   333→   333→    raw.delete(STREAM_RISK_SCORES)
   334→   334→    try:
   335→   335→        raw.delete("rto:stream:hll:orders:0")  # processor HLL (if any)
   336→   336→    except Exception:
   337→   337→        pass
   338→   338→
   339→   339→    received: list[tuple[str, dict]] = []
   340→   340→
   341→   341→    def handler(stream: str, fields: dict) -> None:
   342→   342→        received.append((stream, dict(fields)))
   343→   343→
   344→   344→    consumer = StreamConsumer(_REDIS_URL, group="test-group", consumer="test-consumer")
   345→   345→
   346→   346→    # Run the consumer in a daemon thread; stop it after a short poll.
   347→   347→    def _run():
   348→   348→        # block_ms=500 + count limit; the consumer will exit via _stop
   349→   349→        # signal handler — but in tests we can't SIGTERM a thread, so we
   350→   350→        # use a short block + check a stop flag via the consumer's _stop.
   351→   351→        try:
   352→   352→            client = consumer._connect()
   353→   353→            consumer._ensure_group(STREAM_RISK_SCORES)
   354→   354→            # One-shot read instead of the infinite loop.
   355→   355→            resp = client.xreadgroup(
   356→   356→                groupname="test-group",
   357→   357→                consumername="test-consumer",
   358→   358→                streamcounts={STREAM_RISK_SCORES: ">"},
   359→   359→                count=10,
   360→   360→                block=500,
   361→   361→            )
   362→   362→            if resp:
   363→   363→                for stream, messages in resp:
   364→   364→                    for msg_id, fields in messages:
   365→   365→                        try:
   366→   366→                            handler(stream, dict(fields) if fields else {})
   367→   367→                            client.xack(stream, "test-group", msg_id)
   368→   368→                        except Exception:
   369→   369→                            pass
   370→   370→        except Exception:
   371→   371→            pass
   372→   372→
   373→   373→    # Produce the message BEFORE starting the consumer so XREADGROUP
   374→   374→    # has something to deliver on the first poll.
   375→   375→    producer = StreamProducer(_REDIS_URL)
   376→   376→    msg_id = producer.publish(
   377→   377→        STREAM_RISK_SCORES,
   378→   378→        {
   379→   379→            "prediction_id": "p-cons-1",
   380→   380→            "order_id": "ORD-CONS-1",
   381→   381→            "decision": "REVIEW",
   382→   382→            "score": "0.555",
   383→   383→            "ts": "2026-01-02T00:00:00+00:00",
   384→   384→        },
   385→   385→    )
   386→   386→    assert msg_id is not None
   387→   387→    producer.close()
   388→   388→
   389→   389→    t = threading.Thread(target=_run, daemon=True)
   390→   390→    t.start()
   391→   391→    t.join(timeout=3.0)
   392→   392→    assert not t.is_alive(), "consumer thread should have exited after one poll"
   393→   393→
   394→   394→    assert len(received) >= 1, "consumer handler should have been called once"
   395→   395→    stream, fields = received[0]
   396→   396→    assert stream == STREAM_RISK_SCORES
   397→   397→    assert fields["prediction_id"] == "p-cons-1"
   398→   398→    assert fields["order_id"] == "ORD-CONS-1"
   399→   399→    assert fields["decision"] == "REVIEW"
   400→   400→
   401→   401→    consumer.close()
   402→   402→    raw.delete(STREAM_RISK_SCORES)
