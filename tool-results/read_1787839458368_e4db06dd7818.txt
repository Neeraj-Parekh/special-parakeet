   114→def test_merkle_root_padding_to_power_of_two():
   115→    """The Merkle root of N leaves (N not a power of 2) must equal the
   116→    root of the same leaves padded with the last leaf's hash to the
   117→    next power of 2 (RFC 6962-style padding)."""
   118→    leaves = ["a" * 64, "b" * 64, "c" * 64]  # 3 leaves → pad to 4
   119→    root = MerkleSealer._merkle_root(leaves)
   120→    # Manual recompute with padding.
   121→    padded = leaves + [leaves[-1]]
   122→    level = padded
   123→    while len(level) > 1:
   124→        nxt = []
   125→        for i in range(0, len(level), 2):
   126→            nxt.append(hashlib.sha256((level[i] + level[i + 1]).encode()).hexdigest())
   127→        level = nxt
   128→    assert root == level[0]
   129→    # Empty → GENESIS.
   130→    assert MerkleSealer._merkle_root([]) == "0" * 64

   133→def test_merkle_proof_reconstructs_root():
   134→    """The Merkle proof for a leaf at position P, when combined with the
   135→    leaf hash, must reconstruct the same root as ``MerkleSealer._merkle_root``
   136→    over the whole leaf list. This is the inclusion-proof invariant
   137→    (RFC 6962 §2.1.1).
   138→
   139→    T1.3 FIX: the prior version of this test had an ``or True`` tautology
   140→    at its final assert — it tried both ``sha256(leaf + sibling)`` and
   141→    ``sha256(sibling + leaf)`` at each level but ALWAYS picked the right-
   142→    side form regardless of the leaf's parity, so for odd indices (where
   143→    the sibling is on the LEFT) the reconstruction silently broke + the
   144→    ``or True`` papered over the bug. The proof BUILDER in
   145→    ``MerkleSealer._build_proof_path`` was already correct (it emits the
   146→    sibling's position explicitly); only the test's reconstruction was
   147→    buggy. This rewrite routes through the shared static helper +
   148→    honors each step's ``position`` field so the proof builder is
   149→    exercised for BOTH even and odd leaf indices.
   150→
   151→    Coverage:
   152→      * 5 leaves (odd → exercises the RFC 6962 padding case: pad to 8
   153→        with the last leaf's hash repeated).
   154→      * 4 positions: 0 (even), 1 (odd), 2 (even), 4 (odd, last). Each
   155→        has a different sibling-position pattern across tree levels —
   156→        together they cover every left/right bookkeeping branch.
   157→    """
   158→    leaves = [hashlib.sha256(f"leaf-{i}".encode()).hexdigest() for i in range(5)]
   159→    root = MerkleSealer._merkle_root(leaves)
   160→    # Sanity: 5 leaves pad to 8 (3 duplicates of the last leaf).
   161→    assert root == MerkleSealer._merkle_root(
   162→        leaves + [leaves[-1]] * 3
   163→    ), "padding rule mismatch — _merkle_root must use last-leaf-repeat"
   164→
   165→    for position in (0, 1, 2, 4):
   166→        proof_path = MerkleSealer._build_proof_path(leaves, position)
   167→        # Proof length must be ceil(log2(padded_size)) = 3 (padded to 8).
   168→        assert len(proof_path) == 3, (
   169→            f"position {position}: expected 3 proof steps (8-leaf padded "
   170→            f"tree → log2(8) levels), got {len(proof_path)}"
   171→        )
   172→        # Reconstruct root from leaf hash + proof path. The sibling's
   173→        # position field tells us the order: "right" → parent = H(leaf + sib),
   174→        # "left" → parent = H(sib + leaf). This is the RFC 6962 invariant.
   175→        h = leaves[position]
   176→        for step in proof_path:
   177→            if step["position"] == "right":
   178→                h = hashlib.sha256((h + step["hash"]).encode()).hexdigest()
   179→            else:  # "left"
   180→                h = hashlib.sha256((step["hash"] + h).encode()).hexdigest()
   181→        assert h == root, (
   182→            f"position {position} proof failed to reconstruct root: "
   183→            f"got {h}, want {root}. Proof path was: {proof_path}"
   184→        )
