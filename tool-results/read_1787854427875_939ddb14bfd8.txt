   192→def test_risk_score_endpoint_publishes_to_streams(monkeypatch):
   193→    """POST /risk/score → assert StreamProducer.publish was called with the
   194→    3 stream names + canonical prediction_id.
   195→
   196→    Uses monkeypatch on ``src.api.routes.StreamProducer`` so when
   197→    ``create_app()`` instantiates the producer, it gets a Mock-class
   198→    instance whose ``publish`` is a MagicMock. The handler still calls
   199→    ``publish`` three times per request (risk.scores, audit.records,
   200→    cases.created-on-REVIEW); we capture the calls + assert structure.
   201→    """
   202→    # Create the mock class BEFORE importing create_app so the patch
   203→    # is in place when create_app() resolves the name at call time.
   204→    # The mock instance's ``publish`` is a MagicMock that returns a fake
   205→    # message ID (so the fire-and-forget path looks successful).
   206→    captured_publish_calls: list[tuple] = []
   207→
   208→    class _MockProducer:
   209→        def __init__(self, redis_url=None):
   210→            self.redis_url = redis_url
   211→            self.client = None  # mirroring the real shape
   212→            self.publish = MagicMock(side_effect=self._capture_publish)
   213→
   214→        def _capture_publish(self, stream, fields):
   215→            captured_publish_calls.append((stream, dict(fields)))
   216→            return f"mock-msg-id-{len(captured_publish_calls)}"
   217→
   218→        def close(self):
   219→            pass
   220→
   221→    # Patch the name as imported into routes.py. The import line was:
   222→    #   from src.stream.producer import (... StreamProducer)
   223→    # so the name lives in the ``src.api.routes`` module namespace.
   224→    from src.api import routes as routes_mod
   225→
   226→    monkeypatch.setattr(routes_mod, "StreamProducer", _MockProducer)
   227→
   228→    with TestClient(create_app(scorer_rate_per_min=1000)) as client:
   229→        # Accept decision — should NOT trigger cases.created publish.
   230→        r1 = client.post("/risk/score", json=VALID, headers=SCORER)
   231→        assert r1.status_code == 200
   232→        body1 = r1.json()
   233→        # Find this request's publishes by prediction_id (other tests in the
   234→        # module might share the mock, but TestClient-with-context ensures
   235→        # isolation — the create_app call instantiated a fresh mock).
   236→        pid1 = body1["prediction_id"]
   237→        p1_calls = [
   238→            (s, f) for s, f in captured_publish_calls if f.get("prediction_id") == pid1
   239→        ]
   240→        streams_published_1 = {s for s, _ in p1_calls}
   241→        # risk.scores + audit.records always published.
   242→        assert STREAM_RISK_SCORES in streams_published_1
   243→        assert STREAM_AUDIT_RECORDS in streams_published_1
   244→        # ACCEPT (or REJECT) — NOT REVIEW — so cases.created should be absent.
   245→        # (We can't assert the exact decision because the model is live; but
   246→        # we CAN assert that IF decision != REVIEW, cases.created wasn't
   247→        # called for this prediction_id.)
   248→        if body1["decision"] != "REVIEW":
   249→            assert STREAM_CASES_CREATED not in streams_published_1, (
   250→                "cases.created should only fire on REVIEW, got decision="
   251→                f"{body1['decision']}"
   252→            )
   253→        else:
   254→            # REVIEW → cases.created published with case_id from the response.
   255→            assert STREAM_CASES_CREATED in streams_published_1
   256→            cases_call = next(
   257→                f for s, f in p1_calls if s == STREAM_CASES_CREATED
   258→            )
   259→            assert cases_call["case_id"] == body1["case_id"]
   260→            assert cases_call["prediction_id"] == pid1
   261→
   262→        # Force a REVIEW decision by adding a REVIEW rule. The rules engine
   263→        # is empty by default; we add a rule that always fires (priority 1,
   264→        # REVIEW action) and then assert cases.created was published.
   265→        captured_publish_calls.clear()
   266→        rules_post = client.post(
   267→            "/v1/rules",
   268→            json={
   269→                "rule_id": "force-review-str",
   270→                "name": "Track F test rule",
   271→                "field": "amount_inr",
   272→                "op": "gt",
   273→                "value": 0,  # always true for VALID amount
   274→                "action": "REVIEW",
   275→                "priority": 1,
   276→            },
   277→            headers=ADMIN,
   278→        )
   279→        assert rules_post.status_code == 200
   280→        # POST a fresh order — REVIEW rule will fire + force decision=REVIEW
   281→        # (the cost-optimizer might pick ACCEPT but the REVIEW rule gates
   282→        # it to REVIEW; the decision_source becomes cost_optimal_bmr_review_rule).
   283→        review_payload = {**VALID, "order_id": "STR-REVIEW-1"}
   284→        r2 = client.post("/risk/score", json=review_payload, headers=SCORER)
   285→        assert r2.status_code == 200
   286→        body2 = r2.json()
   287→        assert body2["decision"] == "REVIEW", (
   288→            "REVIEW rule should force decision=REVIEW; got "
   289→            f"{body2['decision']} (source={body2['decision_source']})"
   290→        )
   291→        pid2 = body2["prediction_id"]
   292→        p2_calls = [
   293→            (s, f) for s, f in captured_publish_calls if f.get("prediction_id") == pid2

   295→        streams_published_2 = {s for s, _ in p2_calls}
   296→        assert STREAM_RISK_SCORES in streams_published_2
   297→        assert STREAM_AUDIT_RECORDS in streams_published_2
   298→        assert STREAM_CASES_CREATED in streams_published_2

   300→            f for s, f in p2_calls if s == STREAM_CASES_CREATED

   304→        assert cases_call["order_id"] == "STR-REVIEW-1"

   310→            f for s, f in p2_calls if s == STREAM_RISK_SCORES

   312→        assert risk_call["prediction_id"] == pid2

   315→        assert risk_call["decision"] == "REVIEW"

   322→@skip_no_redis
   323→def test_consumer_processes_message():
   324→    """Produce a message, run consumer in a thread, assert handler called."""
   325→    if _skip_if_redis_unreachable():
   326→        pytest.skip("Redis not reachable at " + _REDIS_URL)
   327→    import redis  # type: ignore[import-not-found]
   328→
   329→    from src.stream.consumer import StreamConsumer
   330→
   331→    raw = redis.from_url(_REDIS_URL, decode_responses=True)
   332→    # Clean the stream + the consumer group from any prior run.
   333→    raw.delete(STREAM_RISK_SCORES)
   334→    try:
   335→        raw.delete("rto:stream:hll:orders:0")  # processor HLL (if any)
   336→    except Exception:
   337→        pass
   338→
   339→    received: list[tuple[str, dict]] = []
   340→
   341→    def handler(stream: str, fields: dict) -> None:
   342→        received.append((stream, dict(fields)))
   343→
   344→    consumer = StreamConsumer(_REDIS_URL, group="test-group", consumer="test-consumer")
   345→
   346→    # Run the consumer in a daemon thread; stop it after a short poll.
   347→    def _run():
   348→        # block_ms=500 + count limit; the consumer will exit via _stop
   349→        # signal handler — but in tests we can't SIGTERM a thread, so we
   350→        # use a short block + check a stop flag via the consumer's _stop.
   351→        try:
   352→            client = consumer._connect()
   353→            consumer._ensure_group(STREAM_RISK_SCORES)
   354→            # One-shot read instead of the infinite loop.
   355→            resp = client.xreadgroup(
   356→                groupname="test-group",
   357→                consumername="test-consumer",
   358→                streamcounts={STREAM_RISK_SCORES: ">"},
   359→                count=10,
   360→                block=500,
   361→            )
   362→            if resp:
   363→                for stream, messages in resp:
   364→                    for msg_id, fields in messages:
   365→                        try:
   366→                            handler(stream, dict(fields) if fields else {})
   367→                            client.xack(stream, "test-group", msg_id)
   368→                        except Exception:
   369→                            pass
   370→        except Exception:
   371→            pass
   372→
   373→    # Produce the message BEFORE starting the consumer so XREADGROUP
   374→    # has something to deliver on the first poll.
   375→    producer = StreamProducer(_REDIS_URL)
   376→    msg_id = producer.publish(
   377→        STREAM_RISK_SCORES,
   378→        {
   379→            "prediction_id": "p-cons-1",
   380→            "order_id": "ORD-CONS-1",
   381→            "decision": "REVIEW",
   382→            "score": "0.555",
   383→            "ts": "2026-01-02T00:00:00+00:00",
   384→        },
   385→    )
   386→    assert msg_id is not None
   387→    producer.close()
   388→
   389→    t = threading.Thread(target=_run, daemon=True)
   390→    t.start()
   391→    t.join(timeout=3.0)
   392→    assert not t.is_alive(), "consumer thread should have exited after one poll"
   393→
   394→    assert len(received) >= 1, "consumer handler should have been called once"
   395→    stream, fields = received[0]
   396→    assert stream == STREAM_RISK_SCORES
   397→    assert fields["prediction_id"] == "p-cons-1"
   398→    assert fields["order_id"] == "ORD-CONS-1"
   399→    assert fields["decision"] == "REVIEW"
   400→
   401→    consumer.close()
   402→    raw.delete(STREAM_RISK_SCORES)
