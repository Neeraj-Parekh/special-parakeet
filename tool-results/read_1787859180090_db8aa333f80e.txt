   285→def test_override_nonce_persistence_survives_simulated_restart(
   286→    tmp_state_dir, no_database_url
   287→):
   288→    """A consumed override nonce hash survives a process restart — a
   289→    captured override request cannot be replayed after a redeploy.
   290→
   291→    The A2 fix (14-d) inserts the nonce hash into the in-memory LRU+TTL
   292→    cache (file-mode) on first sighting → 200; a second sighting →
   293→    409. But the in-memory cache was wiped on every redeploy, opening
   294→    a replay window. The 15-b fix persists the consumed nonce hash to
   295→    ``override_nonces_state.json`` + re-loads on module import so the
   296→    replay protection survives the restart.
   297→    """
   298→    # Access the module-level cache via attribute access (same reason
   299→    # as the mandate tests above — the routes module doesn't get
   300→    # reloaded by the test suite, but consistency is safer).
   301→    import src.api.routes as routes_mod
   302→
   303→    # Clear both the in-memory cache + the file (the fixture's
   304→    # _clear_override_nonce_cache call does both, but the tmp_state_dir
   305→    # fixture runs first so the file path is already tmp_path).
   306→    _clear_override_nonce_cache()
   307→
   308→    # Consume a fresh nonce — first sighting, no exception. The
   309→    # _check_and_consume_override_nonce helper inserts into the
   310→    # in-memory cache + calls _persist_nonce (file-mode path).
   311→    nonce_hash = hashlib.sha256(b"test-nonce-restart-1").hexdigest()
   312→    _check_and_consume_override_nonce({}, nonce_hash, None)
   313→    # The in-memory cache now has the entry.
   314→    assert nonce_hash in routes_mod._override_nonce_cache
   315→
   316→    # Force-flush the persist file (the throttled default might skip
   317→    # the write within 5 sec of the clear's reset). This makes the
   318→    # test robust against throttle-timing flakes.
   319→    _persist_nonce(nonce_hash, force=True)
   320→
   321→    # Verify the file actually has the nonce hash.
   322→    state_path = tmp_state_dir / "override_nonces_state.json"
   323→    assert state_path.exists(), (
   324→        "The force-flush must have written the nonce file; the test "
   325→        "cannot verify restart-persistence without an actual on-disk file."
   326→    )
   327→    import json
   328→    with open(state_path) as f:
   329→        on_disk = json.load(f)
   330→    assert isinstance(on_disk, list)
   331→    assert nonce_hash in on_disk
   332→
   333→    # Simulate a process restart: clear the in-memory cache (but
   334→    # DON'T delete the file — the _clear_override_nonce_cache helper
   335→    # would delete the file too, which is NOT what a real restart
   336→    # does). Then re-run _load_nonces_from_disk (the same function
   337→    # that runs at module import).
   338→    with _override_nonce_cache_lock:
   339→        routes_mod._override_nonce_cache.clear()
   340→    _load_nonces_from_disk()
   341→
   342→    # The cache should now be re-populated from the file — the
   343→    # persisted nonce hash should be present.
   344→    assert nonce_hash in routes_mod._override_nonce_cache, (
   345→        "After a process restart, the override nonce cache should be "
   346→        "re-populated from the file so the same nonce still gets "
   347→        "rejected (replay protection survives the restart — this is "
   348→        "the whole point of the DO BADLY #3 fix on the routes side)."
   349→    )
   350→
   351→    # Replay the same nonce — must get 409 (replay detected). The
   352→    # persisted nonce hash is in the cache, so the second sighting
   353→    # trips the replay check.
   354→    with pytest.raises(HTTPException) as exc:
   355→        _check_and_consume_override_nonce({}, nonce_hash, None)
   356→    assert exc.value.status_code == 409
   357→    assert "replay detected" in str(exc.value.detail), (
   358→        "The replayed nonce must be rejected with 409 'replay detected' "
   359→        "even after a process restart — this is the A2 + 15-b combined "
   360→        "guarantee (no replay window opens on redeploy)."
   361→    )

   365→# Throttle test — 100 rapid mutations → < 20 disk writes (the 5-sec
   366→# throttle suppresses writes within the window).
   367→# ============================================================================

   369→
   370→def test_throttle_limits_disk_writes_under_burst(tmp_state_dir, monkeypatch):
   371→    """100 rapid mutations on a fresh ``_FileState`` → fewer than 20
   372→    actual disk writes (the 5-second throttle skips writes within the
   373→    window).
   374→
   375→    The first mutation writes (since ``_last_persist`` is ``0.0`` on a
   376→    fresh instance); the next 99 mutations are within the 5-second
   377→    throttle window → skipped. The assertion ``< 20`` is generous
   378→    headroom (the expected count is exactly 1).
   379→    """
   380→    # Fresh _FileState instance (not the module-level one) so the
   381→    # throttle window is wide open (no prior persist to set
   382→    # _last_persist to "now").
   383→    fs = _FileState("test_throttle.json")
   384→    # Reset _last_persist explicitly (the __init__ calls _load_from_disk
   385→    # which doesn't touch _last_persist; but a prior test in the same
   386→    # tmp_state_dir — there isn't one, since tmp_path is per-test —
   387→    # could have left a file. This is belt-and-suspenders).
   388→    fs._last_persist = 0.0
   389→    sub = fs.sub("cumulative_monthly")
   390→
   391→    # Count os.replace calls. The _FileState._persist_to_disk uses
   392→    # os.replace(tmp, path) as the atomic-write step — this is the
   393→    # cleanest signal of "actual disk write happened" (vs "persist
   394→    # was called but throttled"). The throttle check is upstream, so
   395→    # only writes that pass the throttle reach os.replace.
   396→    write_count = {"n": 0}
   397→    original_replace = os.replace
   398→
   399→    def counting_replace(src, dst):
   400→        write_count["n"] += 1
   401→        return original_replace(src, dst)
   402→
   403→    monkeypatch.setattr(os, "replace", counting_replace)
   404→
   405→    # 100 rapid mutations (all within ~0.01 sec — well within the
   406→    # 5-second throttle window).
   407→    for i in range(100):
   408→        sub[f"key_{i:03d}"] = float(i)
   409→
   410→    # The 5-second throttle should suppress writes after the first.
   411→    assert write_count["n"] < 20, (
   412→        f"100 rapid mutations should produce fewer than 20 disk writes "
   413→        f"(the 5-second throttle suppresses writes within the window); "
   414→        f"got {write_count['n']} writes. The throttle is broken."
   415→    )
   416→    # Sanity check — at least 1 write happened (the very first
   417→    # mutation, before the throttle window opens).
   418→    assert write_count["n"] >= 1, (
   419→        "At least one write should have happened (the very first "
   420→        "mutation, before the throttle window opens — _last_persist "
   421→        "was 0.0 so the first persist's throttle check passes). "
   422→        "Zero writes means the persist path is broken."
   423→    )

   426→def test_force_persist_bypasses_throttle(tmp_state_dir, monkeypatch):
   427→    """``force=True`` bypasses the 5-second throttle so explicit
   428→    flushes (test isolation, the ``reset_upi_counters`` clear, process
   429→    shutdown hooks) write immediately regardless of the throttle
   430→    window.
   431→
   432→    The 15-b implementation uses ``force=True`` in
   433→    ``reset_upi_counters`` so the post-clear empty state is reflected
   434→    on disk immediately (the 3 individual ``.clear()`` calls each
   435→    trigger a throttled persist; the throttle would skip the 2nd +
   436→    3rd writes, leaving the file with a stale partial-cleared state).
   437→    """
   438→    fs = _FileState("test_force.json")
   439→    # Set _last_persist to "just now" so the throttle would normally
   440→    # suppress the next persist (force=False would skip).
   441→    fs._last_persist = time.time()
   442→    sub = fs.sub("cumulative_monthly")
   443→
   444→    write_count = {"n": 0}
   445→    original_replace = os.replace
   446→
   447→    def counting_replace(src, dst):
   448→        write_count["n"] += 1
   449→        return original_replace(src, dst)
   450→
   451→    monkeypatch.setattr(os, "replace", counting_replace)
   452→
   453→    # 50 force-flushes — all should write (bypass throttle). Each
   454→    # force-flush is a separate _persist_to_disk(force=True) call.
   455→    for i in range(50):
   456→        sub[f"key_{i:03d}"] = float(i)
   457→        fs._persist_to_disk(force=True)
   458→
   459→    # All 50 force-flushes should write (bypass the throttle).
   460→    assert write_count["n"] == 50, (
   461→        f"force=True bypasses the throttle — all 50 force-flushes "
   462→        f"should write; got {write_count['n']} (expected 50). The "
   463→        f"force-bypass is broken."
   464→    )
   465→
   466→    # Counter-test: the same 50 mutations WITHOUT force would all be
   467→    # throttled (since _last_persist was set to "just now" at the top
   468→    # + the force-flushes above kept updating it). Verify the throttle
   469→    # WOULD have skipped them.
   470→    pre_count = write_count["n"]
   471→    for i in range(50, 100):
   472→        sub[f"key_{i:03d}"] = float(i)
   473→        # NO force — the throttle should kick in.
   474→        fs._persist_to_disk()  # force=False (default)
   475→    # All 50 throttled persists should have been skipped.
   476→    assert write_count["n"] == pre_count, (
   477→        f"The 50 throttled (force=False) persists after a "

   491→def test_override_nonce_cache_lru_cap_at_10000(tmp_state_dir):
   492→    """Insert 10_001 nonces into the ``_override_nonce_cache`` — the
   493→    cache must stay at 10_000 entries (LRU eviction per the TTLCache
   494→    ``maxsize=10_000`` configuration).
   495→
   496→    The 14-d implementation already configured ``TTLCache(maxsize=10_000,
   497→    ttl=86400)`` for the override nonce cache (bounded LRU + 1-day TTL).
   498→    This test is a regression guard: any future change that bumps the
   499→    maxsize to unbounded (or removes the TTLCache for a plain dict)
   500→    will fail this test, surfacing the unbounded-growth bug.
   501→
   502→    The mandate of DO BADLY #3 explicitly calls out "bounded LRU cap
   503→    (max 10_000) on any unbounded dict in your territory" — the
   504→    override nonce cache IS in my territory + the cap was already in
   505→    place via 14-d's TTLCache. This test makes the cap ASSERTED, not
   506→    just claimed.
   507→    """
   508→    # Attribute access at call time — gets the CURRENT binding.
   509→    import src.api.routes as routes_mod
   510→
   511→    # Clear first (the autouse fixture in test_override_replay.py
   512→    # would do this if we were running there, but this test runs in
   513→    # its own file — be explicit).
   514→    _clear_override_nonce_cache()
   515→    try:
   516→        cache = routes_mod._override_nonce_cache
   517→        # Defensive: the cache's maxsize should be configured to 10_000
   518→        # (asserting the configuration, not just the behaviour).
   519→        assert cache.maxsize == 10_000, (
   520→            "The override nonce cache must be configured with "
   521→            "maxsize=10_000 (bounded LRU cap per the DO BADLY #3 "
   522→            "mandate). Got a different maxsize — the configuration "
   523→            "was changed."
   524→        )
   525→
   526→        # Insert 10_001 fresh nonce hashes directly into the cache.
   527→        # The TTLCache evicts the oldest (least-recently-used) entry
   528→        # when the 10_001st entry is inserted, keeping the size at
   529→        # 10_000.
   530→        for i in range(10_001):
   531→            cache[f"nonce-hash-{i:05d}"] = True
   532→
   533→        # The cache must be at exactly 10_000 entries (the maxsize cap
   534→        # evicted the oldest entry).
   535→        assert len(cache) == 10_000, (
   536→            f"The override nonce cache is bounded to 10_000 entries "
   537→            f"(TTLCache maxsize=10_000); after inserting 10_001 "
   538→            f"entries, the size should be 10_000 (LRU eviction); got "
   539→            f"{len(cache)}. The cap is broken — the cache would grow "
   540→            f"unbounded under burst traffic, eventually OOMing the "
   541→            f"process."
   542→        )
   543→
   544→        # The first inserted entry (``nonce-hash-00000``) should have
   545→        # been evicted (LRU — it's the least-recently-used after 10_000
   546→        # subsequent inserts).
   547→        assert "nonce-hash-00000" not in cache, (
   548→            "The oldest nonce hash should have been evicted (LRU) "
   549→            "when the 10_001st entry was inserted."
   550→        )
   551→        # The last inserted entry (``nonce-hash-10000``) should still
   552→        # be present.
   553→        assert "nonce-hash-10000" in cache, (
   554→            "The most-recently-inserted nonce hash should still be "
   555→            "present (LRU eviction only evicts the oldest entries)."
   556→        )
   557→    finally:
   558→        _clear_override_nonce_cache()
