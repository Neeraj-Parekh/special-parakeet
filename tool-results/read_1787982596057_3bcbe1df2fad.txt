   463→# --- Circuit Breaker ---
   464→class CircuitBreaker:
   465→    """Fail open if ML is down."""
   466→    def __init__(self, failure_threshold=5, recovery_timeout=30):
   467→        self.failures = 0
   468→        self.state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN
   469→        self.threshold = failure_threshold
   470→        self.timeout = recovery_timeout
   471→        self.last_failure = None
   472→    
   473→    def call(self, func, *args, **kwargs):
   474→        if self.state == "OPEN":
   475→            if (datetime.utcnow() - self.last_failure).seconds > self.timeout:
   476→                self.state = "HALF_OPEN"
   477→            else:
   478→                return None  # Fail open
   479→        
   480→        try:
   481→            result = func(*args, **kwargs)
   482→            if self.state == "HALF_OPEN":
   483→                self.state = "CLOSED"
   484→                self.failures = 0
   485→            return result
   486→        except Exception as e:
   487→            self.failures += 1
   488→            self.last_failure = datetime.utcnow()
   489→            if self.failures >= self.threshold:
   490→                self.state = "OPEN"
   491→                logger.error("circuit_breaker_opened")
   492→            raise

   529→    explanation: list
   530→    rule_fired: Optional[str]
   531→    model_version: str
   532→    latency_ms: float

   536→def compute_hash(record: dict, previous_hash: str) -> str:
   537→    record_str = json.dumps(record, sort_keys=True) + previous_hash
   538→    return hashlib.sha256(record_str.encode()).hexdigest()

   541→    # Get last hash for chain
   542→    last = db.query(AuditRecord).order_by(AuditRecord.timestamp.desc()).first()

   544→    
   545→    record["previous_hash"] = prev_hash
   546→    record["raw_hash"] = compute_hash(record, prev_hash)
   547→    
   548→    db_record = AuditRecord(**record)
   549→    db.add(db_record)
   550→    db.commit()

   577→    merchant: dict = Depends(verify_api_key),
   578→    db: Session = Depends(get_db),

   621→        prob = 0.5
   622→        version = "fallback"
   623→    
   624→    risk_score = prob * 100
   625→    
   626→    # 5. Decision Gates (Dynamic Thresholds)
   627→    if risk_score < 15:
   628→        decision = "ACCEPT"
   629→    elif risk_score < 60:
   630→        decision = "REVIEW"
   631→    else:
   632→        decision = "REJECT"
   633→    
   634→    # 6. SHAP-like Explanation (simplified)

   636→        {"feature": "order_amount", "value": request.amount, "contribution": 0.3},
   637→        {"feature": "is_cod", "value": request.is_cod, "contribution": 0.25},
   638→        {"feature": "prior_returns", "value": online_features.get("prior_returns", 0), "contribution": 0.2}
   639→    ]
   640→    
   641→    # 7. Build Response
   642→    prediction_id = f"pred_{request.order_id}_{int(start.timestamp())}"
   643→    latency = (datetime.utcnow() - start).total_seconds() * 1000
   644→    
   645→    response = RiskResponse(
   646→        prediction_id=prediction_id,
   647→        risk_score=risk_score,
   648→        probability=prob,
   649→        decision=decision,
   650→        explanation=explanation,
   651→        rule_fired=rule_decision,
   652→        model_version=version,
   653→        latency_ms=round(latency, 2),
   654→        timestamp=start
   655→    )
   656→    
   657→    # 8. Async Audit Write
   658→    audit_record = {
   659→        "prediction_id": prediction_id,
   660→        "order_id": request.order_id,
   661→        "merchant_id": merchant["merchant_id"],
   662→        "risk_score": risk_score,

   673→                order_id=request.order_id,
   674→                merchant=merchant["merchant_id"],

  1004→
  1005→You were right to be scared of unconstrained agents. Here's how you add agentic capability **safely**:
  1006→
  1007→```python
  1008→# src/agents/bounded_agent.py
  1009→from typing import List, Literal
  1010→
  1011→class BoundedAgent:
  1012→    """
  1013→    Agent that can ONLY perform pre-approved actions.
  1014→    No LLM decides money movements. Ever.
  1015→    """
  1016→    
  1017→    ALLOWED_ACTIONS = {
  1018→        "score_order": {"params": ["order_id"], "cost": 0},
  1019→        "request_otp": {"params": ["customer_id"], "cost": 1},
  1020→        "flag_review": {"params": ["order_id", "reason"], "cost": 2},
  1021→        "block_order": {"params": ["order_id", "reason"], "cost": 10, "requires_approval": True}
  1022→    }
  1023→    
  1024→    def __init__(self, risk_api_client, approval_queue):
  1025→        self.client = risk_api_client
  1026→        self.approval = approval_queue
  1027→    
  1028→    def act(self, intent: str, params: dict) -> dict:
  1029→        if intent not in self.ALLOWED_ACTIONS:
  1030→            return {"error": "Action not in allowlist", "blocked": True}
  1031→        
  1032→        action = self.ALLOWED_ACTIONS[intent]
  1033→        
  1034→        # High-cost actions require human approval
  1035→        if action.get("requires_approval"):
  1036→            self.approval.submit(intent, params)
  1037→            return {"status": "pending_approval", "ticket_id": str(uuid4())}
  1038→        
  1039→        # Execute bounded action
  1040→        if intent == "score_order":
  1041→            return self.client.score(params["order_id"])
  1042→        elif intent == "request_otp":
  1043→            return self.client.send_otp(params["customer_id"])
  1044→        
  1045→        return {"error": "Unknown execution path"}
  1046→```

  1470→## 4. The Agentic Layer (Phase 2 — Thin Plugin)
  1471→
  1472→**Location:** `Agent Gateway Service` — sits as a consumer of the Risk API, not as a controller.
  1473→
  1474→**Design:** The agent **cannot** call payment APIs, refund APIs, or modify orders. It can only:
  1475→1. `risk.score(order_id)` — get risk score
  1476→2. `risk.explain(prediction_id)` — get explanation
  1477→3. `case.create(order_id, reason)` — create review case
  1478→4. `notify.merchant(merchant_id, message)` — send alert
  1479→5. `rules.suggest(condition, action)` — suggest new rule (pending admin approval)
  1480→
  1481→**Guardrails:**
  1482→- Action allowlist (hardcoded, not LLM-decided)
  1483→- Every agent action is logged via Audit Service with `actor: "agent:{agent_id}"`

  1517→## 6. Security Model
  1518→
  1519→### 6.1 Zero Trust Between Services
  1520→- **Gateway → Services:** JWT with short expiry (5 min), signed with RS256
  1521→- **Service → Service:** mTLS (mutual TLS) certificates
  1522→- **Service → DB:** Connection pooling with dedicated DB user per service
  1523→- **Secrets:** HashiCorp Vault or Docker Secrets (for hackathon)
  1524→
  1525→### 6.2 Data Protection
  1526→- **At Rest:** PostgreSQL encrypted (LUKS), Redis encrypted (Redis ACL + TLS)
  1527→- **In Transit:** TLS 1.3 everywhere
  1528→- **PII:** Address fields hashed (SHA-256) in audit logs. Original address only in encrypted DB column.
  1529→- **API Keys:** Hashed with bcrypt, never stored plaintext
  1530→
  1531→### 6.3 Compliance Posture
  1532→- **RBI PA Guidelines:** Data localization (all data stays in India region)
  1533→- **PCI DSS:** Tokenization of sensitive fields, audit trail completeness
  1534→- **GDPR-style:** Right to explanation (every decision has SHAP), right to deletion (soft delete + anonymization)

  1940→## 6. Load Testing Strategy (k6 vs Locust)
  1941→
  1942→You need to prove your system handles load. Here's the decision and patterns.
  1943→
  1944→| Tool | Language | Best For | Your Use Case |
  1945→|---|---|---|---|
  1946→| **k6** | JavaScript (Go engine) | CI/CD gating, high throughput, Prometheus-native | **Primary choice.** Single binary, threshold-based pass/fail (`p(95)<100ms`), integrates with Grafana Cloud. |
  1947→| **Locust** | Python | Complex logic, distributed load, Python ecosystem | **Secondary.** Use if your team needs to compute HMAC signatures or custom auth in test scripts. |
  1948→
  1949→**Recommended: k6** for your hackathon because:
  1950→- Your agents write JavaScript faster than Python for simple HTTP tests
  1951→- Native threshold gating: if p95 latency > 100ms, the CI pipeline fails 
  1952→- Single binary, no Python environment needed on CI runners
  1953→
  1954→**Load patterns to implement:**
  1955→1. **Steady-state:** 100 concurrent users, 5 minutes → baseline metrics
  1956→2. **Ramp:** 0→1000 users over 2 minutes → autoscaling test
  1957→3. **Spike:** 0→5000 users in 30 seconds → circuit breaker test
  1958→4. **Soak:** 100 users for 30 minutes → memory leak detection
  1959→
  1960→**Target thresholds:**
  1961→- `p(50) < 50ms` (median latency)
  1962→- `p(95) < 100ms` (tail latency)
  1963→- `p(99) < 200ms` (worst case)
  1964→- Error rate < 0.1%
  1965→- ML inference latency < 20ms (model-only)
