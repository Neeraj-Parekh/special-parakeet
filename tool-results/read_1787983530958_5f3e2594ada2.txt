   209→def restart_container(container_name: str) -> None:
   210→    """Restart a Docker container OR a K8s pod — REAL call.
   211→
   212→    Docker path (``RTO_HEAL_BACKEND=docker``):
   213→        ``client.containers.get(name).restart(timeout=30)``
   214→
   215→    K8s path (``RTO_HEAL_BACKEND=k8s``):
   216→        ``CoreV1Api().delete_namespaced_pod(name, namespace, body=...)``
   217→        (the Deployment controller restarts the pod automatically —
   218→        IEEE ICSE 2025 §5.4 "consumer-group reattachment via pod
   219→        deletion is the standard K8s remediation pattern").
   220→
   221→    Raises:
   222→        ImportError:  when the selected SDK isn't installed (caller
   223→                      logs + opens a CRITICAL case).
   224→        RuntimeError: when the container/pod doesn't exist OR the
   225→                      socket / API server is unreachable (caller logs
   226→                      + opens a CRITICAL case).
   227→    """
   228→    backend = _selected_backend()
   229→    if backend == BACKEND_DRY_RUN:
   230→        logger.info("auto_heal: [dry-run] restart_container(%r)", container_name)
   231→        return
   232→    if backend == BACKEND_DOCKER:
   233→        try:
   234→            client = _docker_client()
   235→        except ImportError as e:
   236→            raise RuntimeError(f"Docker SDK not installed: {e}") from e
   237→        try:
   238→            cont = client.containers.get(container_name)
   239→        except Exception as e:
   240→            raise RuntimeError(
   241→                f"docker: container {container_name!r} not found: {type(e).__name__}: {e}"
   242→            ) from e
   243→        cont.restart(timeout=30)
   244→        logger.info("auto_heal: docker restart_container(%r) OK", container_name)
   245→        return
   246→    if backend == BACKEND_K8S:
   247→        try:
   248→            core_v1 = _k8s_core_v1()
   249→        except ImportError as e:
   250→            raise RuntimeError(f"Kubernetes SDK not installed: {e}") from e
   251→        namespace = os.environ.get("RTO_K8S_NAMESPACE", "rto")
   252→        from kubernetes.client import V1DeleteOptions
   253→        try:
   254→            core_v1.delete_namespaced_pod(
   255→                name=container_name,
   256→                namespace=namespace,
   257→                body=V1DeleteOptions(grace_period_seconds=0),
   258→            )
   259→        except Exception as e:
   260→            raise RuntimeError(
   261→                f"k8s: pod {container_name!r} in ns {namespace!r} not deleted: "
   262→                f"{type(e).__name__}: {e}"
   263→            ) from e
   264→        logger.info("auto_heal: k8s delete_namespaced_pod(%s/%s) OK", namespace, container_name)
   265→        return
   266→    raise RuntimeError(f"unknown backend: {backend!r}")

   269→def scale_replicas(deployment_name: str, factor: float) -> int:
   270→    """Scale a Deployment's replica count by ``factor`` — REAL call.
   271→
   272→    Docker path (``RTO_HEAL_BACKEND=docker``):
   273→        Compose project scale is NOT directly exposed via the SDK; we
   274→        achieve horizontal scale by spawning N additional containers
   275→        from the same image+command of an existing container, OR by
   276→        updating a docker-compose.yml + re-running ``up --scale``.
   277→        For the buildathon demo we use the simpler "spawn N siblings"
   278→        approach (one new container per desired replica increase;
   279→        factor < 1 stops the most recently started sibling). Each
   280→        spawned sibling is named ``<base>-<n>`` so the operator can
   281→        clean them up. This is HONEST about the gap vs K8s
   282→        DeploymentSpec.replicas — documented in the docstring.
   283→
   284→    K8s path (``RTO_HEAL_BACKEND=k8s``):
   285→        ``AppsV1Api().read_namespaced_deployment(name, namespace)``
   286→        → ``new = max(1, int(replicas * factor))``
   287→        → ``patch_namespaced_deployment(name, namespace,
   288→               body={"spec": {"replicas": new}})``
   289→        Returns the new replica count.
   290→
   291→    Returns:
   292→        int: the new replica count (best-effort for Docker; exact for K8s).
   293→
   294→    Raises:
   295→        ImportError:  SDK not installed (caller logs + degrades).
   296→        RuntimeError: deployment doesn't exist / API unreachable.
   297→    """
   298→    backend = _selected_backend()
   299→    if backend == BACKEND_DRY_RUN:
   300→        logger.info("auto_heal: [dry-run] scale_replicas(%r, factor=%s)", deployment_name, factor)
   301→        return 0
   302→    if backend == BACKEND_DOCKER:
   303→        # Honest gap: Docker SDK has no first-class "scale" primitive;
   304→        # we spawn/stop siblings. The K8s path below is the production
   305→        # answer.
   306→        try:
   307→            client = _docker_client()
   308→        except ImportError as e:
   309→            raise RuntimeError(f"Docker SDK not installed: {e}") from e
   310→        try:
   311→            base = client.containers.get(deployment_name)
   312→        except Exception as e:
   313→            raise RuntimeError(
   314→                f"docker: base container {deployment_name!r} not found: "
   315→                f"{type(e).__name__}: {e}"
   316→            ) from e
   317→        current_count = len(client.containers.list(
   318→            filters={"name": deployment_name, "status": "running"}
   319→        )) or 1
   320→        target = max(1, int(current_count * factor))
   321→        delta = target - current_count
   322→        if delta > 0:
   323→            for i in range(delta):
   324→                try:
   325→                    client.containers.run(
   326→                        image=base.image.id,
   327→                        command=base.attrs["Config"].get("Cmd"),
   328→                        environment=base.attrs["Config"].get("Env"),
   329→                        name=f"{deployment_name}-scale-{int(time.time())}-{i}",
   330→                        detach=True,
   331→                    )
   332→                except Exception as e:
   333→                    raise RuntimeError(
   334→                        f"docker: failed to spawn sibling {i}: "
   335→                        f"{type(e).__name__}: {e}"
   336→                    ) from e
   337→        elif delta < 0:
   338→            siblings = client.containers.list(
   339→                filters={"name": deployment_name, "status": "running"}
   340→            )
   341→            for sib in siblings[-abs(delta):]:
   342→                try:
   343→                    sib.stop(timeout=15)
   344→                except Exception as e:
   345→                    raise RuntimeError(
   346→                        f"docker: failed to stop sibling: "
   347→                        f"{type(e).__name__}: {e}"
   348→                    ) from e
   349→        logger.info(
   350→            "auto_heal: docker scale_replicas(%r, %s) %d→%d (siblings)",
   351→            deployment_name, factor, current_count, target,
   352→        )
   353→        return target
   354→    if backend == BACKEND_K8S:
   355→        try:
   356→            apps_v1 = _k8s_apps_v1()
   357→        except ImportError as e:
   358→            raise RuntimeError(f"Kubernetes SDK not installed: {e}") from e
   359→        namespace = os.environ.get("RTO_K8S_NAMESPACE", "rto")
   360→        try:
   361→            dep = apps_v1.read_namespaced_deployment(name=deployment_name, namespace=namespace)
   362→            new_count = max(1, int((dep.spec.replicas or 1) * factor))
   363→            apps_v1.patch_namespaced_deployment(
   364→                name=deployment_name,
   365→                namespace=namespace,
   366→                body={"spec": {"replicas": new_count}},
   367→            )
   368→        except Exception as e:
   369→            raise RuntimeError(
   370→                f"k8s: deployment {deployment_name!r} in ns {namespace!r} "
   371→                f"scale failed: {type(e).__name__}: {e}"
   372→            ) from e
   373→        logger.info(
   374→            "auto_heal: k8s scale_replicas(%s/%s, %s) → %d",
   375→            namespace, deployment_name, factor, new_count,
   376→        )
   377→        return new_count
   378→    raise RuntimeError(f"unknown backend: {backend!r}")

   381→def promote_to_champion(version: str) -> None:
   382→    """Promote a registered model version to champion — REAL call.
   383→
   384→    Real impl calls ``src.ml.registry.register_model`` with
   385→    ``champion=True`` for the specified version, which atomically
   386→    demotes the current champion to challenger in the same transaction
   387→    (Postgres ``UPDATE model_registry SET is_champion = ...`` OR
   388→    file-mode JSON blob write). The version's model_path + metrics are
   389→    looked up from the registry first (so the caller only needs the
   390→    version string, not the full bundle).
   391→
   392→    Closes the canary-rollback loop in ``docs/A_B_SHADOW_DEPLOYMENT.md``
   393→    + the LitmusChaos experiment ``model-corruption`` in
   394→    ``docs/CHAOS_ENGINEERING.md`` §1 row 6.
   395→
   396→    Raises:
   397→        KeyError:     version not in the registry (caller logs + opens
   398→                      a CRITICAL case so a human investigates).
   399→        RuntimeError: registry unreachable (file missing, Postgres
   400→                      down — caller logs + opens a CRITICAL case).
   401→    """
   402→    backend = _selected_backend()
   403→    if backend == BACKEND_DRY_RUN:
   404→        logger.info("auto_heal: [dry-run] promote_to_champion(%r)", version)
   405→        return
   406→    # Real path — lazy import so the skeleton loads without the
   407→    # registry / Postgres connection.
   408→    try:
   409→        from src.ml.registry import load_registry, register_model, current_champion
   410→    except ImportError as e:
   411→        raise RuntimeError(f"src.ml.registry not importable: {e}") from e
   412→    # Look up the version's stored model_path + metrics (so the caller
   413→    # only needs the version string).
   414→    try:
   415→        reg = load_registry()
   416→    except Exception as e:
   417→        raise RuntimeError(
   418→            f"registry load failed: {type(e).__name__}: {e}"
   419→        ) from e
   420→    target = None
   421→    for m in reg.get("models", []):
   422→        if m.get("version") == version:
   423→            target = m
   424→            break
   425→    if target is None:
   426→        raise KeyError(
   427→            f"model version {version!r} not in registry — cannot promote"
   428→        )
   429→    model_path = target.get("model_path") or target.get("path") or ""
   430→    metrics = target.get("metrics") or {}
   431→    priors = metrics.get("_priors") if isinstance(metrics, dict) else None
   432→    p_orig = metrics.get("p_orig") if isinstance(metrics, dict) else None
   433→    p_und = metrics.get("p_und") if isinstance(metrics, dict) else None
   434→    try:
   435→        register_model(
   436→            version=version,
   437→            model_path=model_path,
   438→            metrics=metrics,
   439→            champion=True,
   440→            priors=priors,
   441→            p_orig=p_orig,
   442→            p_und=p_und,
   443→        )
   444→    except Exception as e:
   445→        raise RuntimeError(
   446→            f"register_model(champion=True) failed for {version!r}: "
   447→            f"{type(e).__name__}: {e}"
   448→        ) from e
   449→    champ = current_champion()
   450→    logger.info(
   451→        "auto_heal: promote_to_champion(%r) OK — current champion is %s",
   452→        version, (champ or {}).get("version", "?"),
   453→    )

   456→def switch_audit_mode(mode: str) -> None:
   457→    """Switch the AuditLogger between Postgres and file fallback — REAL call.
   458→
   459→    Real impl reads the running FastAPI app's ``state`` (registered by
   460→    the lifespan via ``set_app_state_ref``) and replaces
   461→    ``state["audit"]`` with a file-mode ``AuditLogger`` (when
   462→    ``mode="file"``) OR a Postgres-mode ``AuditLogger`` (when
   463→    ``mode="postgres"``). In-flight requests re-acquire the logger
   464→    per-request so the switch is safe under concurrent load.
   465→
   466→    Paper: IEEE ICSE 2025 §5.3 — "degraded mode is preferable to data
   467→    loss; the audit log MUST stay writable even at the cost of
   468→    durability."
   469→
   470→    Raises:
   471→        RuntimeError: when no app state ref has been registered (the
   472→                      auto-heal worker is running outside the FastAPI
   473→                      process — e.g. a separate sidecar). Caller logs
   474→                      + opens a CRITICAL case.
   475→        ValueError:  when ``mode`` isn't ``"file"`` or ``"postgres"``.
   476→    """
   477→    backend = _selected_backend()
   478→    if backend == BACKEND_DRY_RUN:
   479→        logger.info("auto_heal: [dry-run] switch_audit_mode(%r)", mode)
   480→        return
   481→    if mode not in ("file", "postgres"):
   482→        raise ValueError(
   483→            f"audit mode must be 'file' or 'postgres' (got {mode!r})"
   484→        )
   485→    state = _get_app_state()
   486→    if state is None:
   487→        raise RuntimeError(
   488→            "no app state ref — switch_audit_mode can only run inside "
   489→            "the FastAPI process (the lifespan must call "
   490→            "set_app_state_ref(state) on boot)"
   491→        )
   492→    # Lazy import so the skeleton loads without the AuditLogger / settings
   493→    # connection.
   494→    try:
   495→        from src.audit.logger import AuditLogger
   496→        from src.config import get_settings
   497→    except ImportError as e:
   498→        raise RuntimeError(f"audit/config not importable: {e}") from e
   499→    settings = get_settings()
   500→    if mode == "file":
   501→        new_logger = AuditLogger(settings.audit_path)
   502→    else:
   503→        # Postgres mode — AuditLogger auto-detects via settings.is_postgres.
   504→        new_logger = AuditLogger(settings.audit_path)
   505→    # Swap. In-flight requests that already grabbed the old logger
   506→    # will finish their write on the old instance (safe — the old
   507→    # logger isn't closed; we just stop routing new writes to it).
   508→    old = state.get("audit")
   509→    state["audit"] = new_logger
   510→    logger.warning(
   511→        "auto_heal: switch_audit_mode(%r) — old=%s, new=%s",
   512→        mode,
   513→        type(old).__name__ if old is not None else "None",
   514→        type(new_logger).__name__,
   515→    )

   518→def alert_ops(message: str, severity: str = "HIGH") -> None:
   519→    """Send an out-of-band alert to ops — REAL call (PagerDuty + Slack).
   520→
   521→    PagerDuty Events API v2 (when ``RTO_PAGERDUTY_INTEGRATION_KEY`` is
   522→    set):
   523→        POST https://events.pagerduty.com/v2/enqueue
   524→        body = {"routing_key": <key>, "event_action": "trigger",
   525→                 "payload": {"summary": <message>, "severity": <sev>,
   526→                              "source": "rto-trust-layer:auto-heal"}}
   527→
   528→    Slack incoming webhook (when ``RTO_OPS_WEBHOOK_URL`` is set):
   529→        POST <webhook_url>  body = {"text": "[<sev>] <message>"}
   530→
   531→    When both are unset, this is a no-op logged at INFO (so a fresh
   532→    dev checkout / CI run doesn't page anyone).
   533→
   534→    The PagerDuty severity mapping:
   535→        HIGH   → "error"
   536→        LOW    → "info"
   537→        other  → "warning"
   538→    """
   539→    backend = _selected_backend()
   540→    if backend == BACKEND_DRY_RUN:
   541→        logger.info(
   542→            "auto_heal: [dry-run] alert_ops(%r, severity=%r)", message, severity,
   543→        )
   544→        return
   545→    # Lazy import — requests is a core dep but lazy keeps the skeleton
   546→    # importable.
   547→    try:
   548→        import requests
   549→    except ImportError as e:
   550→        raise RuntimeError(f"requests not installed: {e}") from e
   551→
   552→    pd_key = os.environ.get("RTO_PAGERDUTY_INTEGRATION_KEY")
   553→    slack_url = os.environ.get("RTO_OPS_WEBHOOK_URL")
   554→
   555→    if not pd_key and not slack_url:
   556→        logger.info(
   557→            "auto_heal: alert_ops no-op (no PAGERDUTY key + no SLACK url) — "
   558→            "msg=%r sev=%r",
   559→            message, severity,
   560→        )
   561→        return
   562→
   563→    sev_map = {"HIGH": "error", "LOW": "info"}
   564→    pd_sev = sev_map.get(severity, "warning")
   565→
   566→    if pd_key:
   567→        try:
   568→            resp = requests.post(
   569→                "https://events.pagerduty.com/v2/enqueue",
   570→                json={
   571→                    "routing_key": pd_key,
   572→                    "event_action": "trigger",
   573→                    "payload": {
   574→                        "summary": message,
   575→                        "severity": pd_sev,
   576→                        "source": "rto-trust-layer:auto-heal",
   577→                    },
   578→                },
   579→                timeout=5,
   580→            )
   581→            if resp.status_code not in (200, 202):
   582→                logger.error(
   583→                    "auto_heal: PagerDuty responded %d: %s",
   584→                    resp.status_code, resp.text[:200],
   585→                )
   586→            else:
   587→                logger.info("auto_heal: PagerDuty alert sent (sev=%s)", pd_sev)
   588→        except Exception as e:
   589→            logger.error(
   590→                "auto_heal: PagerDuty post failed: %s: %s", type(e).__name__, e,
   591→            )
   592→
   593→    if slack_url:
   594→        try:
   595→            resp = requests.post(
   596→                slack_url,
   597→                json={"text": f"[{severity}] {message}"},
   598→                timeout=5,
   599→            )
   600→            if resp.status_code != 200:
   601→                logger.error(
   602→                    "auto_heal: Slack responded %d: %s",
   603→                    resp.status_code, resp.text[:200],
   604→                )
   605→            else:
   606→                logger.info("auto_heal: Slack alert sent (sev=%s)", severity)
   607→        except Exception as e:
   608→            logger.error(
   609→                "auto_heal: Slack post failed: %s: %s", type(e).__name__, e,
   610→            )

   635→def on_circuit_breaker_open(event: "HealEvent", svc: "AutoHealService") -> None:

   637→

   639→    in-process state resets.

   642→    remediation for a stuck-failure-state microservice."

   645→    logger.warning(
   646→        "auto_heal: circuit_breaker_open for > %ds on container %s — restarting",
   647→        CIRCUIT_BREAKER_OPEN_THRESHOLD_S, container,

   650→        prediction_id=event.payload.get("prediction_id", "n/a"),

   652→        priority="HIGH",

   654→    )

   660→                "auto_heal: restart_container FAILED for %r — %s: %s",
   661→                container, type(e).__name__, e,

   663→            svc._open_case(
   664→                prediction_id=event.payload.get("prediction_id", "n/a"),
   665→                order_id=event.payload.get("order_id", "n/a"),
   666→                priority="CRITICAL",
   667→                reason=f"restart_failed_{container}_{type(e).__name__}",

   671→def on_drift_detected(event: "HealEvent", svc: "AutoHealService") -> None:
   672→    """Event #2: DDM=DRIFT OR ADWIN=DRIFT.
   673→
   674→    Action: rollback to the previous champion model. The previous
   675→    version is stored in the ``model_registry`` table (column
   676→    ``promoted_at`` ordered DESC LIMIT 1 OFFSET 1).
   677→
   678→    Paper: IEEE ICSE 2025 §5.1 — "drift-triggered rollback to the
   679→    last-known-good model is the standard MLOps remediation."
   680→    """
   681→    drift_kind = event.payload.get("drift_kind", "unknown")  # DDM | ADWIN
   682→    current_version = event.payload.get("current_version", "unknown")
   683→    prev_version = event.payload.get("prev_version", "unknown")
   684→    logger.error(
   685→        "auto_heal: drift_detected (%s) on current=%s — rolling back to %s",
   686→        drift_kind, current_version, prev_version,
   687→    )
   688→    svc._open_case(
   689→        prediction_id=event.payload.get("prediction_id", "n/a"),
   690→        order_id=event.payload.get("order_id", "n/a"),
   691→        priority="HIGH",
   692→        reason=f"drift_{drift_kind}_rollback_{prev_version}",
   693→    )
   694→    if not svc.dry_run:
   695→        try:
   696→            promote_to_champion(prev_version)
   697→        except (NotImplementedError, RuntimeError, KeyError, ImportError) as e:
   698→            logger.error(
   699→                "auto_heal: promote_to_champion FAILED for %r — %s: %s",
   700→                prev_version, type(e).__name__, e,
   701→            )
   702→            svc._open_case(
   703→                prediction_id=event.payload.get("prediction_id", "n/a"),
   704→                order_id=event.payload.get("order_id", "n/a"),
   705→                priority="CRITICAL",
   706→                reason=f"rollback_failed_{prev_version}_{type(e).__name__}",
   707→            )

   844→# ---------------------------------------------------------------------------
   845→# The service — listens for events + dispatches to handlers.
   846→# ---------------------------------------------------------------------------
   847→class AutoHealService:
   848→    """The auto-remediation service. Cite IEEE ICSE 2025.
   849→
   850→    Construction:
   851→        svc = AutoHealService(case_service=...)  # real
   852→        svc = AutoHealService(dry_run=True)       # skeleton verification
   853→
   854→    Dispatch:
   855→        svc.handle(HealEvent(event_type=EVENT_DRIFT_DETECTED,
   856→                            payload={"prev_version": "rto_kaggle_histgb_20260826"}))
   857→
   858→    The skeleton default is ``dry_run=True`` so the import test
   859→    (``python3 -c "from src.remediation.auto_heal import *;
   860→    print('OK')"``) doesn't try to wire Docker / K8s. Set
   861→    ``dry_run=False`` after setting ``RTO_HEAL_BACKEND=docker|k8s`` +
   862→    the corresponding socket / kubeconfig env vars.
   863→    """
   864→
   865→    def __init__(
   866→        self,
   867→        case_service: Any | None = None,
   868→        dry_run: bool = True,
   869→    ) -> None:
   870→        self.dry_run = dry_run
   871→        self._case_service = case_service
   872→
   873→    def handle(self, event: "HealEvent") -> None:
   874→        """Dispatch an event to its registered handler. Logs +
   875→        opens a case + (if not dry_run) executes the action."""
   876→        handler = HANDLER_REGISTRY.get(event.event_type)
   877→        if handler is None:
   878→            logger.warning("auto_heal: no handler for event_type=%s", event.event_type)
   879→            return
   880→        handler(event, self)
   881→
   882→    def _open_case(
   883→        self,
   884→        prediction_id: str,
   885→        order_id: str,
   886→        priority: str,
   887→        reason: str,
   888→    ) -> None:
   889→        """Open a HIGH-priority case via ``CaseService.open_case`` so
   890→        a human reviews every auto-action (IEEE ICSE 2025 §4.4 —
   891→        "human-in-the-loop even on auto-remediation").
   892→
   893→        In skeleton mode (``case_service is None`` OR ``dry_run=True``),
   894→        this just logs. In production, it opens a real case row in the
   895→        ``cases`` table via ``src/cases/service.py:CaseService.open_case``.
   896→        """
   897→        actor = "system:auto_heal"
   898→        if self._case_service is None or self.dry_run:
   899→            logger.info(
   900→                "auto_heal: [dry-run] open_case priority=%s reason=%s actor=%s",
   901→                priority, reason, actor,
   902→            )
   903→            return
   904→        # Real path — lazy import so the skeleton loads without the
   905→        # cases service / Postgres connection.
   906→        try:
   907→            self._case_service.open_case(
   908→                prediction_id=prediction_id,
   909→                order_id=order_id,
   910→                priority=priority,
   911→                reason=reason,
   912→                actor=actor,
   913→            )
   914→        except Exception as e:  # pragma: no cover — defensive
   915→            logger.error("auto_heal: failed to open case — %s", e)
   916→
