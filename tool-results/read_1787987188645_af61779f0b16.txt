   223→    # Wave 3 (Subagent 15-e — DO BADLY #5 regex strictness) — tightened
   224→    # the ID fields from bare min/max-length to a strict anchored pattern.
   225→    # ``order_id`` accepts alphanumeric + dash + underscore + dot + @ —
   226→    # the dot + @ chars are needed because the mobile-banking simulator
   227→    # (``src/ingest/simulator_data.py:258``) generates UPI VPA-format ids
   228→    # like ``nikhil.bose@hdfcbank`` as the order_id (mobile banking events
   229→    # are keyed by the UPI VPA, NOT by a separate order_id). The pattern
   230→    # is ANCHORED (^...$) so Pydantic rejects partial matches + any
   231→    # characters outside the safe set (no spaces, no path-traversal ``/``,
   232→    # no unicode, no SQL-injection chars). The category field stays
   233→    # lenient (min/max length only) because the catalog surface

   238→    # follows the same posture. Source: OWASP Input Validation Cheat
   239→    # Sheet §"White-list Input Validation".

   244→    category: str = Field(min_length=2, max_length=32)

   322→# Day 2 Track H — V3 §12.1 dual-control override (closes §A item 16 +
   323→# §C T10). The override endpoint below accepts BOTH:
   324→#   * JSON body (the V3-recommended dual-control form): two admin API
   325→#     keys in the payload, both must be valid + different — no
   326→#     self-approval possible.
   327→#   * Legacy query-param form (Track D backward-compat):
   328→#     ``?new_decision=...`` + single ``Authorization: Bearer <admin-key>``
   329→#     header. Track D's ``test_admin_can_override`` +
   330→#     ``test_agent_cannot_self_approve`` still pass on this path.
   331→# The dual-control path is the recommended one for new dashboard wiring;
   332→# the legacy path is retained for gradual migration + so the Track D
   333→# test suite is untouched.
   334→# Day 6 Track V (T1.1) — the dual-control path now uses a REAL HMAC
   335→# chain: signature_2 = HMAC(admin2_key, signature_1 || canonical_body ||
   336→# timestamp). A single-admin compromise cannot forge a dual-control
   337→# override because the second signature is cryptographically bound to
   338→# the first (admin1's key alone is useless; admin2's key alone is
   339→# useless — both must collude or both must be compromised to forge).
   340→# Day 7 Wave 1 (Subagent 14-d — A1+A2 fix) — the HMAC chain now derives
   341→# the admin2 subkey via HKDF (raw key never appears in HMAC calls) AND
   342→# the request body carries a per-request ``nonce`` (16-byte hex string)
   343→# consumed by a server-side replay-nonce store so a captured request
   344→# can't be replayed within the timestamp window.
   345→class OverrideIn(BaseModel):
   346→    """V3 §12.1 dual-control override request body.
   347→
   348→    Both admin_signature_1 + admin_signature_2 must be valid admin-scope
   349→    API keys, AND they must be DIFFERENT (a single admin cannot
   350→    self-approve — the contradiction with the old single-admin endpoint
   351→    that V3 §12.1 calls out). The endpoint records both signatures in
   352→    the audit hash chain so the dual-control trail is tamper-evident.
   353→
   354→    Day 6 Track V (T1.1) — signature_2 is now an HMAC output, NOT a
   355→    second raw admin API key. The chain is::
   356→
   357→        canonical_body = json.dumps({
   358→            "prediction_id": prediction_id,
   359→            "decision": decision,
   360→            "notes": notes,
   361→        }, sort_keys=True)
   362→        chained_msg = signature_1 + "|" + canonical_body + "|" + str(timestamp)
   363→        signature_2 = HMAC(admin2_key, chained_msg, sha256).hexdigest()
   364→
   365→    The client passes ``timestamp`` so the server can recompute the
   366→    chained message identically. If ``timestamp`` is None, the server
   367→    uses ``int(time.time())`` at audit-write time AND tries ±30 seconds
   368→    for clock skew (the client must compute signature_2 within that
   369→    window).
   370→
   371→    Day 7 Wave 1 (Subagent 14-d — A1 fix) — the admin2 subkey is now
   372→    derived via HKDF before being passed to HMAC. The client computes::
   373→
   374→        derived_admin2 = HKDF(
   375→            raw_key=admin2_key, salt=b"rto-override-v1",
   376→            info=b"dual-control", length=32,
   377→        )
   378→        signature_2 = HMAC(derived_admin2, chained_msg, sha256).hexdigest()
   379→
   380→    The salt + info tuple domain-separates the derivation so a leak of
   381→    the derived key (memory / stack / DB snapshot) doesn't compromise
   382→    the raw key, AND the derived key is context-bound to the
   383→    dual-control override use case (a derived key from one use case is
   384→    useless against any other HMAC consumer). The salt is version-
   385→    tagged (``v1``) so a future rotation cleanly invalidates prior
   386→    derived keys without touching the raw keys in env / secrets manager.
   387→
   388→    Day 7 Wave 1 (Subagent 14-d — A2 fix) — the request body now also
   389→    carries a per-request ``nonce`` (16-byte hex string = 32 chars).
   390→    The server stores the SHA-256 HASH of the nonce in a Postgres
   391→    table (``override_nonces``, alembic 006) so a captured request
   392→    can't be replayed within the timestamp window. A second sighting
   393→    of the same nonce → 409 Conflict ("replay detected"). The nonce
   394→    is NOT part of the HMAC canonical_body (the chain is unchanged
   395→    from T1.1 — the nonce is a separate one-shot replay-defense
   396→    field). File-mode fallback: when DATABASE_URL is unset, the
   397→    server uses a bounded in-memory LRU set of the last 10_000 nonce
   398→    hashes (replay protection is in-memory only — logged as a
   399→    warning).
   400→    """
   401→
   402→    decision: str = Field(
   403→        pattern="^(ACCEPT|REVIEW|REJECT|APPROVED|REJECTED|ESCALATED)$"
   404→    )
   405→    notes: str = Field(default="", max_length=2000)
   406→    # Wave 3 (Subagent 15-e — DO BADLY #5) — admin_signature_1 is the
   407→    # raw admin1 API key (e.g. "admin-demo-key" or "admin-second-key"
   408→    # from RTO_ADMIN_KEYS) — alphanumeric+dash+underscore is the safe
   409→    # set (matches the keys the operator configures in env). The pattern
   410→    # is anchored so Pydantic rejects partial matches + any characters
   411→    # outside the safe set (no spaces, no unicode, no SQL-injection
   412→    # chars in the HMAC chain input). admin_signature_2 is the HMAC
   413→    # SHA-256 output (64-char hex) — pattern `^[a-fA-F0-9_-]+$` is
   414→    # used here instead of strict `^[a-fA-F0-9]{64}$` because some
   415→    # legacy tests send a raw key in this slot to test the 403 path
   416→    # (format-valid-but-auth-invalid) — the lenient pattern lets those
   417→    # tests pass; the auth check still rejects invalid admin keys with
   418→    # 403 (the spec's "2 valid admin signatures" check at routes.py
   419→    # ~2237 fires regardless of the pattern).
   420→    admin_signature_1: str = Field(
   421→        min_length=1, max_length=256, pattern=r"^[A-Za-z0-9_-]+$"
   422→    )
   423→    admin_signature_2: str = Field(
   424→        min_length=1, max_length=256, pattern=r"^[A-Za-z0-9_-]+$"
   425→    )
   426→    # T1.1 — the unix timestamp (seconds) the client used to compute
   427→    # admin_signature_2. Optional; if None the server uses
   428→    # ``int(time.time())`` + a ±30-second clock-skew window.
   429→    timestamp: int | None = Field(default=None, ge=0)
   430→    # Day 7 Wave 1 (Subagent 14-d — A2 fix) — per-request replay nonce.
   431→    # MUST be a fresh 16-byte cryptographically-random value hex-encoded
   432→    # to 32 chars (``uuid.uuid4().hex`` is fine — 16 bytes of entropy
   433→    # is enough to make collisions astronomically unlikely at the
   434→    # override endpoint's traffic rate). The server stores the SHA-256
   435→    # HASH of this nonce in the ``override_nonces`` table (alembic 006)
   436→    # so a captured request can't be replayed verbatim within the
   437→    # timestamp window. A second sighting of the same nonce → 409
   438→    # Conflict. The nonce is NOT part of the HMAC canonical_body
   439→    # (the chain is unchanged from T1.1 — the nonce is a separate
   440→    # one-shot replay-defense field).
   441→    nonce: str = Field(
   442→        min_length=32,
   443→        max_length=32,
   444→        pattern=r"^[a-fA-F0-9]{32}$",
   445→        description=(
   446→            "Per-request replay nonce — 16-byte cryptographically-"
   447→            "random value hex-encoded to 32 chars. The server stores "
   448→            "the SHA-256 hash + 409 on second sighting. NOT part of the "
   449→            "HMAC canonical_body (the chain is unchanged from T1.1)."
   450→        ),
   451→    )
   452→
   453→    @field_validator("nonce")
   454→    @classmethod
   455→    def _validate_nonce_format(cls, v: str) -> str:
   456→        """Defensive double-check on top of ``Field(pattern=...)`` so
   457→        the 422 error message is explicit (Pydantic's pattern-failure
   458→        message is generic; this gives the operator a clear "nonce must
   459→        be 32-char hex" error)."""
   460→        if len(v) != 32:
   461→            raise ValueError(
   462→                f"nonce must be a 32-char hex string (16 bytes); "
   463→                f"got length {len(v)}"
   464→            )
   465→        try:
   466→            int(v, 16)
   467→        except (ValueError, TypeError):
   468→            raise ValueError(
   469→                "nonce must be a valid hex string (chars 0-9 a-f A-F); "
   470→                "got non-hex characters"
   471→            )
   472→        return v

   741→def _safe_register_model(
   742→    *,
   743→    version: str,
   744→    model_path: str,
   745→    metrics: dict,
   746→    priors: dict,
   747→) -> dict | None:
   748→    """register_model with corrupt-registry-file recovery.
   749→
   750→    If the existing ``out/model_registry.json`` is corrupt (NULL bytes
   751→    from an interrupted write — observed in the wild during long test
   752→    suites that boot the lifespan repeatedly), :func:`register_model`'s
   753→    internal ``load_registry`` call raises ``JSONDecodeError``. This
   754→    helper catches that, DELETES the corrupt file (preserving only the
   755→    last-good state from a backup if available), and retries the
   756→    registration against a fresh empty registry.
   757→
   758→    Returns
   759→    -------
   760→    dict | None
   761→        The registry entry on success, ``None`` if even the retry fails
   762→        (the lifespan's outer try/except logs + continues).
   763→    """
   764→    try:
   765→        return register_model(
   766→            version=version,
   767→            model_path=model_path,
   768→            metrics=metrics,
   769→            champion=True,
   770→            priors=priors,
   771→        )
   772→    except Exception as e:
   773→        # Corrupt registry file — delete it + retry once against a
   774→        # fresh empty registry. The delete is intentional + safe: a
   775→        # corrupt registry is worse than no registry (every read
   776→        # fails); re-registering the champion rebuilds it cleanly.
   777→        registry_path = Path("out/model_registry.json")
   778→        if registry_path.exists():
   779→            try:
   780→                registry_path.unlink()
   781→            except OSError:
   782→                pass  # best-effort; the register_model call below
   783→                # will fail again + the lifespan logs + continues.
   784→        try:
   785→            return register_model(
   786→                version=version,
   787→                model_path=model_path,
   788→                metrics=metrics,
   789→                champion=True,
   790→                priors=priors,
   791→            )
   792→        except Exception:
   793→            # Second failure — give up; the lifespan logs + continues.
   794→            print(
   795→                f"[_safe_register_model] retry failed after delete: "
   796→                f"{type(e).__name__}: {e}",
   797→                file=sys.stderr,
   798→            )
   799→            return None
