    74→# ---------------------------------------------------------------------------
    75→class _FileState:
    76→    """File-backed combined dict-of-dicts for the mandate counter state.
    77→
    78→    Three sub-dicts (``cumulative_monthly``, ``cumulative_24h``,
    79→    ``last_activity``) are stored under ONE JSON file so a single atomic
    80→    write captures a consistent snapshot of all mandate counters. The
    81→    file path is configurable via the ``RTO_STATE_DIR`` env var
    82→    (default ``out/``); the file name is passed at construction (e.g.
    83→    ``mandate_counters_state.json``).
    84→
    85→    Mutating operations on the sub-dicts trigger ``_persist_to_disk()``,
    86→    which is throttled to ``max one disk write per 5 seconds`` to avoid
    87→    I/O thrash under burst traffic. ``force=True`` bypasses the
    88→    throttle for explicit flushes (test isolation, ``reset_upi_counters``,
    89→    process shutdown hooks). The brief window during the 5-second I/O
    90→    gap is the documented trade-off: if the process dies within 5 sec
    91→    of the last persist, the in-memory state since the last persist is
    92→    lost. Acceptable per the spec — the DB path is the authoritative
    93→    production store; this file is only the file-mode / dev / test
    94→    fallback.
    95→
    96→    Atomic write (``tmp + os.replace``) so a crash mid-write never
    97→    leaves a corrupt file. Best-effort — a write failure (read-only
    98→    FS, full disk, permissions) degrades to "no persistence across
    99→    restarts" which is the prior (pre-15-b) behaviour.
   100→    """
   101→
   102→    _SCHEMA: tuple[str, ...] = (
   103→        "cumulative_monthly", "cumulative_24h", "last_activity",
   104→    )
   105→    _THROTTLE_SECONDS: float = 5.0
   106→
   107→    def __init__(self, file_name: str) -> None:
   108→        self._file_name = file_name
   109→        self._data: dict[str, dict[str, Any]] = {k: {} for k in self._SCHEMA}
   110→        self._lock = threading.Lock()
   111→        self._last_persist: float = 0.0
   112→        self._load_from_disk()
   113→
   114→    def _persist_to_disk(self, *, force: bool = False) -> None:
   115→        """Dump the combined state to JSON. Throttled (max once per
   116→        5 sec) unless ``force=True``. Atomic write (``tmp + os.replace``)
   117→        so a crash mid-write never leaves a corrupt file. Best-effort —
   118→        a write failure degrades to "no persistence across restarts"
   119→        (the prior behaviour).
   120→        """
   121→        now = time.time()
   122→        if not force and now - self._last_persist < self._THROTTLE_SECONDS:
   123→            return
   124→        self._last_persist = now
   125→        state_dir = (os.environ.get("RTO_STATE_DIR") or "out").strip() or "out"
   126→        path = os.path.join(state_dir, self._file_name)
   127→        try:
   128→            os.makedirs(state_dir, exist_ok=True)
   129→            tmp = path + ".tmp"
   130→            with open(tmp, "w") as f:
   131→                json.dump(self._data, f)
   132→            os.replace(tmp, path)
   133→        except Exception:
   134→            # File persistence is best-effort — the in-memory dict is
   135→            # still the source of truth for the current process. A
   136→            # write failure (read-only FS, full disk, permissions)
   137→            # degrades to "no persistence across restarts" which is the
   138→            # prior (pre-15-b) behaviour.
   139→            pass
   140→
   141→    def _load_from_disk(self) -> None:
   142→        """Populate the sub-dicts from the JSON file if it exists.
   143→
   144→        Best-effort — a read failure (file missing, corrupt JSON, read
   145→        error) silently starts with empty sub-dicts (the prior
   146→        behaviour; the cap is enforced within the current process
   147→        regardless).
   148→        """
   149→        state_dir = (os.environ.get("RTO_STATE_DIR") or "out").strip() or "out"
   150→        path = os.path.join(state_dir, self._file_name)
   151→        try:
   152→            with open(path, "r") as f:
   153→                loaded = json.load(f)
   154→            if isinstance(loaded, dict):
   155→                for k in self._SCHEMA:
   156→                    v = loaded.get(k)
   157→                    if isinstance(v, dict):
   158→                        self._data[k] = v
   159→        except FileNotFoundError:
   160→            pass  # fresh state — start empty (the common case at first boot)
   161→        except Exception:
   162→            # Corrupt JSON or read error — start empty (don't crash
   163→            # the process at import time; the file can be repaired
   164→            # later). The in-memory dicts remain the source of truth.
   165→            pass
   166→
   167→    def sub(self, key: str) -> "_SubStateView":
   168→        """Return a ``MutableMapping`` view over a single sub-dict.
   169→        Mutations trigger ``_persist_to_disk()`` (throttled)."""
   170→        if key not in self._data:
   171→            raise KeyError(key)
   172→        return _SubStateView(self, key)
   173→
   174→
   175→class _SubStateView(MutableMapping):
   176→    """``MutableMapping`` view over a single sub-dict of a ``_FileState``.
   177→
   178→    All mutating operations (``__setitem__``, ``__delitem__``,
   179→    ``setdefault``, ``clear``, ``pop``, ``update``) trigger the parent
   180→    ``_FileState._persist_to_disk()`` so changes are persisted (throttled).
   181→    Read operations (``__getitem__``, ``__contains__``, ``get``,
   182→    ``__iter__``, ``__len__``) do NOT trigger persist (they're cheap
   183→    + don't change the on-disk state).
   184→
   185→    ``setdefault`` is overridden so the persist fires when a NEW key is
   186→    inserted (the default ``MutableMapping.setdefault`` calls
   187→    ``__setitem__`` if missing — which would fire persist — but the
   188→    returned object might then be mutated in place (e.g.
   189→    ``d.setdefault(k, []).append(v)``) WITHOUT a subsequent
   190→    ``__setitem__`` call, so the persist wouldn't capture the in-place
   191→    mutation. The ``verify_mandate`` path was refactored to use a pure
   192→    ``__setitem__`` instead of ``setdefault+append`` to ensure the
   193→    persist fires after the in-place mutation — see line ~718 below.)
   194→    """
   195→
   196→    __slots__ = ("_parent", "_key")
   197→
   198→    def __init__(self, parent: "_FileState", key: str) -> None:
   199→        self._parent = parent
   200→        self._key = key
   201→
   202→    def _sub(self) -> dict[str, Any]:
   203→        return self._parent._data[self._key]
   204→
   205→    def __getitem__(self, k):
   206→        return self._sub()[k]
   207→
   208→    def __setitem__(self, k, v) -> None:
   209→        with self._parent._lock:
   210→            self._sub()[k] = v
   211→            self._parent._persist_to_disk()
   212→
   213→    def __delitem__(self, k) -> None:
   214→        with self._parent._lock:
   215→            del self._sub()[k]
   216→            self._parent._persist_to_disk()
   217→
   218→    def __iter__(self):
   219→        return iter(self._sub())
   220→
   221→    def __len__(self) -> int:
   222→        return len(self._sub())
   223→
   224→    def __contains__(self, k) -> bool:
   225→        return k in self._sub()
   226→
   227→    def get(self, k, default=None):
   228→        return self._sub().get(k, default)
   229→
   230→    def setdefault(self, k, default=None):  # type: ignore[override]
   231→        with self._parent._lock:
   232→            sub = self._sub()
   233→            if k not in sub:
   234→                sub[k] = default
   235→                self._parent._persist_to_disk()
   236→            return sub[k]
   237→
   238→    def clear(self) -> None:  # type: ignore[override]
   239→        with self._parent._lock:
   240→            self._sub().clear()
   241→            self._parent._persist_to_disk()
   242→
   243→    def update(self, *args, **kwargs) -> None:  # type: ignore[override]
   244→        with self._parent._lock:
   245→            self._sub().update(*args, **kwargs)
   246→            self._parent._persist_to_disk()
   247→
   248→    def pop(self, k, *default_args):
   249→        with self._parent._lock:
   250→            sub = self._sub()
   251→            if k in sub:
   252→                v = sub.pop(k)
   253→                self._parent._persist_to_disk()
   254→                return v
   255→            if default_args:
   256→                return default_args[0]
   257→            raise KeyError(k)
   258→
   259→    def __repr__(self) -> str:
   260→        return f"_SubStateView({self._key!r}, {self._sub()!r})"

   286→def _get_counters_conn() -> Any:
   287→    """Return a lazy shared psycopg connection for the mandate counters path.
   288→
   289→    Returns ``None`` when:
   290→      * ``DATABASE_URL`` is unset (file-mode / test path), or
   291→      * it does not point at a Postgres DSN (defensive — same logic as
   292→        ``src.config.Settings.is_postgres``), or
   293→      * psycopg is not importable (shouldn't happen — requirements.txt pins it;
   294→        the ``try/except ImportError`` is a defensive guard for stripped test
   295→        envs where the package was yanked).
   296→
   297→    The connection is opened once and cached on the module; subsequent calls
   298→    return the cached connection. ``reset_upi_counters()`` does NOT close it —
   299→    connection lifecycle is the process's responsibility, not the test's. The
   300→    cache is keyed off ``DATABASE_URL`` at first-read time; if the env var
   301→    changes mid-process (tests that mutate it), call ``_reset_counters_conn()``
   302→    to clear the cache.
   303→    """
   304→    global _counters_conn
   305→    if _counters_conn is not None:
   306→        return _counters_conn

   308→    if not db_url or not db_url.startswith(
   309→        ("postgresql://", "postgres://", "postgresql+psycopg://")

   313→        import psycopg
   314→    except ImportError:  # pragma: no cover — defensive; psycopg is in requirements
   315→        return None

   317→        if _counters_conn is None:
   318→            _counters_conn = psycopg.connect(db_url, autocommit=False)

   338→# ---------------------------------------------------------------------------
   339→# C8/C9/C10 fix (Subagent 14-b) — the prior code split the counter read
   340→# (``_read_db_counters``) and the counter write (``_write_db_counters``)
   341→# across TWO separate transactions. Two concurrent ``/risk/score`` calls
   342→# could both read the counter below the ₹15k/month cap, both decrement,
   343→# and blow the ceiling. The new ``_begin_db_counter_txn`` opens a single
   344→# transaction, takes a per-mandate row lock with ``SELECT ... FOR UPDATE``,
   345→# does the C9 month-boundary reset (if needed), reads the 24h cooling
   346→# events, and returns a ``_DbCounterTxn`` handle holding the open txn.
   347→# The caller then runs the cap-checks while the lock is held; on VALID
   348→# it calls ``commit_increment`` (UPDATE + INSERT event + C10 90-day
   349→# prune + COMMIT all in the same txn); on any non-VALID verdict it calls
   350→# ``rollback`` to release the lock. Concurrent verifies serialize on the
   351→# row lock — the second caller blocks at the FOR UPDATE until the first
   352→# commits/rolls back.
   353→# ---------------------------------------------------------------------------
   354→class _DbCounterTxn:
   355→    """Open Postgres transaction holding a FOR UPDATE lock on a mandate counter row.
   356→
   357→    Constructed by ``_begin_db_counter_txn``; the caller runs the cap-checks
   358→    (inactivity, per-txn, monthly, device_id, user_id, cooling 24h) while
   359→    the lock is held, then either:
   360→
   361→      * ``commit_increment(new_cumulative_monthly=..., ...)`` on VALID —
   362→        UPDATEs the counter row, INSERTs the 24h event, runs the C10
   363→        90-day prune DELETE, and COMMITs. Closes the transaction.
   364→      * ``rollback()`` on BREACH/REVIEW/EXPIRED — releases the FOR UPDATE
   365→        lock without advancing the counter (a rejected txn does not
   366→        consume the monthly cap; a REVIEW txn does not re-trigger the
   367→        cooling window). Closes the transaction.
   368→
   369→    Both methods are idempotent — a second call is a no-op (the txn is
   370→    already closed). This guards against double-commit in error paths
   371→    where the caller's try/except fires both the rollback and a second
   372→    cleanup.
   373→    """
   374→
   375→    __slots__ = (
   376→        "conn", "cur", "mid", "cumulative_monthly", "last_activity_ts",
   377→        "recent_24h", "current_month_key", "_closed",
   378→    )
   379→
   380→    def __init__(
   381→        self,
   382→        *,
   383→        conn: Any,
   384→        cur: Any,
   385→        mid: str,
   386→        cumulative_monthly: float,
   387→        last_activity_ts: float,
   388→        recent_24h: list[tuple[float, float]],
   389→        current_month_key: str,
   390→    ) -> None:
   391→        self.conn = conn
   392→        self.cur = cur
   393→        self.mid = mid
   394→        self.cumulative_monthly = cumulative_monthly
   395→        self.last_activity_ts = last_activity_ts
   396→        self.recent_24h = recent_24h
   397→        self.current_month_key = current_month_key
   398→        self._closed = False
   399→
   400→    def commit_increment(
   401→        self,
   402→        *,
   403→        new_cumulative_monthly: float,
   404→        last_activity_ts: float,
   405→        txn_ts: float,
   406→        txn_amount: float,
   407→    ) -> None:
   408→        """UPDATE counter + INSERT 24h event + C10 90-day prune + COMMIT.
   409→
   410→        All four statements run in the SAME transaction (the FOR UPDATE
   411→        lock is held throughout). On ANY DB error this rolls back — the
   412→        verify path has ALREADY updated the in-memory dicts by the time
   413→        this is called (so the request succeeds even if the persistence
   414→        layer fails; the cap is enforced within the process). A future
   415→        improvement is to surface the failure to the audit logger so
   416→        ops sees it (the silent-fall-back is the safe-but-debuggable
   417→        trade-off — same as the prior code's behaviour).
   418→        """
   419→        if self._closed:
   420→            return
   421→        try:
   422→            # C8 — UPDATE the counter while still holding the FOR UPDATE
   423→            # lock (acquired in _begin_db_counter_txn). The
   424→            # ``month_key`` is written back too so the C9 reset branch
   425→            # fires only once per month rollover (subsequent verifies in
   426→            # the same month see the matching month_key and skip the
   427→            # reset).
   428→            self.cur.execute(
   429→                "UPDATE mandate_counters "
   430→                "SET cumulative_monthly = %s, "
   431→                "    last_activity_ts = %s, "
   432→                "    month_key = %s, "
   433→                "    updated_at = NOW() "
   434→                "WHERE mandate_sub = %s",
   435→                (
   436→                    new_cumulative_monthly,
   437→                    last_activity_ts,
   438→                    self.current_month_key,
   439→                    self.mid,
   440→                ),
   441→            )
   442→            self.cur.execute(
   443→                "INSERT INTO mandate_counter_events "
   444→                "(mandate_sub, ts, amount_inr, created_at) "
   445→                "VALUES (%s, %s, %s, NOW())",
   446→                (self.mid, txn_ts, txn_amount),
   447→            )
   448→            # C10 — retention prune. Keep only the last 90 days of
   449→            # mandate_counter_events so the table stays bounded under
   450→            # steady-state traffic. The 24h cooling window only needs
   451→            # the last 24h; 90 days is generous headroom for compliance
   452→            # audit export. Runs on EVERY counter-event INSERT (inline
   453→            # prune-on-write — no scheduler dep). Uses the
   454→            # ``ix_mandate_counter_events_created_at`` index added by
   455→            # alembic migration 004 for a fast range scan.
   456→            self.cur.execute(
   457→                "DELETE FROM mandate_counter_events "
   458→                "WHERE created_at < NOW() - INTERVAL '90 days'"
   459→            )
   460→            self.conn.commit()
   461→        except Exception:
   462→            try:
   463→                self.conn.rollback()
   464→            except Exception:
   465→                pass
   466→        finally:
   467→            self._close()
   468→
   469→    def rollback(self) -> None:
   470→        """Release the FOR UPDATE lock without advancing the counter.
   471→
   472→        Called on BREACH/REVIEW/EXPIRED — a rejected txn does not consume
   473→        the monthly cap; a REVIEW txn does not re-trigger the cooling
   474→        window. Idempotent — a second call (e.g. from a try/finally
   475→        after the cap-check already returned) is a no-op.
   476→        """
   477→        if self._closed:
   478→            return
   479→        try:
   480→            self.conn.rollback()
   481→        except Exception:
   482→            pass
   483→        finally:
   484→            self._close()
   485→
   486→    def _close(self) -> None:
   487→        self._closed = True
   488→        try:
   489→            self.cur.close()
   490→        except Exception:
   491→            pass

   494→def _current_month_key(now: float) -> str:
   495→    """Return the current UTC month as a ``YYYY-MM`` string.
   496→
   497→    Centralised here so ``_begin_db_counter_txn`` (C9 reset) + the test
   498→    that asserts the C9 reset branch uses the same key computation.
   499→    Uses ``time.gmtime`` rather than ``datetime.utcnow`` so we don't
   500→    pull a ``datetime`` import into the verify hot path (the rest of
   501→    the file uses ``time.time()`` consistently).
   502→    """
   503→    return time.strftime("%Y-%m", time.gmtime(now))

   506→def _begin_db_counter_txn(
   507→    mandate_sub: str,
   508→    now: float,
   509→    current_month_key: str | None = None,
   510→) -> "_DbCounterTxn | None":
   511→    """Open a transaction + acquire FOR UPDATE on the per-mandate counter row.
   512→
   513→    Sequence (all inside ONE transaction — the FOR UPDATE lock is held
   514→    throughout):
   515→      1. ``INSERT ... ON CONFLICT DO NOTHING`` to ensure the row exists
   516→         (race-safe against a concurrent verifier that just inserted
   517→         the same ``mandate_sub``).
   518→      2. ``SELECT cumulative_monthly, last_activity_ts, month_key ...
   519→         FOR UPDATE`` — takes the per-mandate row lock. Concurrent
   520→         verifies block here until the first commits/rolls back.
   521→      3. **C9 month-boundary reset** — if the stored ``month_key``
   522→         differs from the current ``YYYY-MM`` string, the monthly cap
   523→         has rolled over: ``UPDATE mandate_counters SET
   524→         cumulative_monthly = 0, month_key = %s`` (still holding the
   525→         FOR UPDATE lock). If ``month_key`` is empty (the migration
   526→         just landed + the row hasn't been touched since), back-fill
   527→         the current month_key without resetting (the counter is
   528→         already the legacy cumulative value — preserving it would
   529→         be wrong, but resetting would lose prior-month spend; the
   530→         pragmatic choice is to back-fill month_key + let the next
   531→         month rollover trigger the reset on its own). Tests cover
   532→         both branches.
   533→      4. ``SELECT ts, amount_inr FROM mandate_counter_events WHERE
   534→         mandate_sub = %s AND ts > now - 86400`` — the 24h cooling
   535→         window, read while holding the lock so the cooling check
   536→         sees a consistent snapshot.
   537→
   538→    Returns a ``_DbCounterTxn`` handle (open transaction, lock held) on
   539→    success. Returns ``None`` on ANY failure (no DB connection, table
   540→    missing, query error) — the caller falls back to the in-memory
   541→    dicts in that case (file-mode / test path). NEVER raise — the
   542→    verify path must degrade to the in-memory fallback rather than
   543→    fail the request.
   544→    """
   545→    conn = _get_counters_conn()
   546→    if conn is None:
   547→        return None
   548→    if current_month_key is None:
   549→        current_month_key = _current_month_key(now)
   550→    try:
   551→        cur = conn.cursor()
   552→        # Step 1 — ensure the row exists (race-safe upsert; ON CONFLICT
   553→        # DO NOTHING so a concurrent verifier that just inserted the
   554→        # same mandate_sub doesn't trip a PK violation).
   555→        cur.execute(
   556→            "INSERT INTO mandate_counters "
   557→            "(mandate_sub, cumulative_monthly, last_activity_ts, month_key, updated_at) "
   558→            "VALUES (%s, 0, NULL, %s, NOW()) "
   559→            "ON CONFLICT (mandate_sub) DO NOTHING",
   560→            (mandate_sub, current_month_key),
   561→        )
   562→        # Step 2 — SELECT FOR UPDATE (C8). Concurrent verifies block
   563→        # here until the first commits/rolls back. The lock is held
   564→        # until commit_increment/rollback is called by the caller.
   565→        cur.execute(
   566→            "SELECT cumulative_monthly, last_activity_ts, month_key "
   567→            "FROM mandate_counters WHERE mandate_sub = %s FOR UPDATE",
   568→            (mandate_sub,),
   569→        )
   570→        row = cur.fetchone()
   571→        if row is None:
   572→            # Defensive — INSERT ON CONFLICT DO NOTHING should have
   573→            # materialised the row above. Roll back + fall back.
   574→            try:
   575→                conn.rollback()
   576→            except Exception:
   577→                pass
   578→            return None
   579→        cumulative_monthly = float(row[0] or 0.0)
   580→        last_activity_ts = float(row[1]) if row[1] is not None else -1.0
   581→        stored_month_key = row[2] or ""
   582→
   583→        # Step 3 — C9 month-boundary reset. If the stored month_key is
   584→        # non-empty AND differs from the current YYYY-MM, the monthly
   585→        # cap has rolled over. Reset cumulative_monthly to 0 + update
   586→        # month_key to the current month. If stored_month_key is empty
   587→        # (the migration just landed; the row hasn't been touched
   588→        # since), back-fill the current month_key WITHOUT resetting
   589→        # (pragmatic choice — see docstring above).
   590→        if stored_month_key and stored_month_key != current_month_key:
   591→            cur.execute(
   592→                "UPDATE mandate_counters "
   593→                "SET cumulative_monthly = 0, month_key = %s, updated_at = NOW() "
   594→                "WHERE mandate_sub = %s",
   595→                (current_month_key, mandate_sub),
   596→            )
   597→            cumulative_monthly = 0.0
   598→        elif not stored_month_key:
   599→            cur.execute(
   600→                "UPDATE mandate_counters "
   601→                "SET month_key = %s, updated_at = NOW() "
   602→                "WHERE mandate_sub = %s",
   603→                (current_month_key, mandate_sub),
   604→            )
   605→
   606→        # Step 4 — 24h cooling window, read while holding the lock so
   607→        # the cooling check sees a consistent snapshot.
   608→        cur.execute(
   609→            "SELECT ts, amount_inr FROM mandate_counter_events "
   610→            "WHERE mandate_sub = %s AND ts > %s ORDER BY ts ASC",
   611→            (mandate_sub, now - 86400),
   612→        )
   613→        recent_24h = [(float(r[0]), float(r[1])) for r in cur.fetchall()]
   614→
   615→        return _DbCounterTxn(
   616→            conn=conn,
   617→            cur=cur,
   618→            mid=mandate_sub,
   619→            cumulative_monthly=cumulative_monthly,
   620→            last_activity_ts=last_activity_ts,
   621→            recent_24h=recent_24h,
   622→            current_month_key=current_month_key,
   623→        )
   624→    except Exception:
   625→        # Degrade to in-memory fallback. The verify path will use the
   626→        # dicts; the cap is enforced within the process (real but not
   627→        # cross-restart). Same trade-off as the prior code.
   628→        try:
   629→            conn.rollback()
   630→        except Exception:
   631→            pass
   632→        return None

   643→def issue_mandate(
   644→    customer_ref: str,
   645→    max_amount_inr: float,
   646→    ttl_seconds: int,
   647→    scope: str = "cod_order",
   648→    *,
   649→    mandate_type: str | None = None,
   650→    device_ids: list[str] | None = None,
   651→    user_id: str | None = None,
   652→    bh_purpose_code: str | None = None,
   653→    max_per_txn_inr: float | None = None,
   654→    max_per_month_inr: float | None = None,
   655→    cooling_24h_inr: float | None = None,
   656→    inactivity_revoke_days: int | None = None,
   657→) -> str:
   658→    """Mint a bounded, HMAC-signed mandate.
   659→
   660→    Backward-compat: 3-positional-arg form (``customer_ref, max_amount_inr,
   661→    ttl_seconds``) still issues a ``cod_order`` mandate — the original system
   662→    agent 1-b built. No existing caller breaks.
   663→
   664→    New keyword-only fields opt in to the UPI Circle / delegated-payments mandate
   665→    type (NPCI OC-201B). When ``mandate_type="upi_circle_delegation"`` (or
   666→    ``scope="upi_circle"`` for short), the payload carries the circular's hard
   667→    caps + the device/user identity chain the issuer must validate per txn.
   668→    """
   669→    payload: dict[str, Any] = {
   670→        "sub": hashlib.sha256(
   671→            f"{customer_ref}:{self_salt()}".encode()
   672→        ).hexdigest()[:16],
   673→        "scope": scope,
   674→        "max_amount_inr": round(float(max_amount_inr), 2),
   675→        "exp": int(time.time()) + int(ttl_seconds),
   676→        "iat": int(time.time()),
   677→    }
   678→
   679→    # Normalize mandate_type: if not provided, infer from scope (back-compat).
   680→    if mandate_type is None:
   681→        mandate_type = "upi_circle_delegation" if scope == "upi_circle" else "cod_order"
   682→    payload["mandate_type"] = mandate_type
   683→
   684→    if mandate_type == "upi_circle_delegation":
   685→        # OC-201B: max 5 IoT devices/software per user. Reject at mint time
   686→        # rather than at verify time — fail-loud doctrine (V3 §4 principle 3).
   687→        if device_ids is None:
   688→            device_ids = []
   689→        if len(device_ids) > 5:
   690→            raise ValueError("OC-201B: max 5 devices per delegation")
   691→        payload["device_ids"] = list(device_ids)
   692→        payload["user_id"] = user_id or ""
   693→        # BH purpose code: "90" = commercial payment per NPCI BH list. The
   694→        # ``bh_purpose_code_reconciliation`` skill surfaces this for raw-file
   695→        # audit trail tagging (Track E Day 2 will persist + emit the
   696→        # secondary-details raw file per OC-201B §3.8).
   697→        payload["bh_purpose_code"] = bh_purpose_code or "90"
   698→        payload["max_per_txn_inr"] = float(
   699→            max_per_txn_inr if max_per_txn_inr is not None else 5000.0
   700→        )
   701→        payload["max_per_month_inr"] = float(
   702→            max_per_month_inr if max_per_month_inr is not None else 15000.0
   703→        )
   704→        payload["cooling_24h_inr"] = float(
   705→            cooling_24h_inr if cooling_24h_inr is not None else 5000.0
   706→        )
   707→        payload["inactivity_revoke_days"] = int(
   708→            inactivity_revoke_days if inactivity_revoke_days is not None else 180
   709→        )
   710→
   711→    body = urlsafe_b64encode(
   712→        json.dumps(payload, sort_keys=True).encode()
   713→    ).decode().rstrip("=")
   714→    sig = hmac.new(_secret(), body.encode(), hashlib.sha256).hexdigest()
   715→    return f"{body}.{sig}"

   718→class MandateVerdict:
   719→    """Outcome categories the mandate verifier can return.
   720→
   721→    Backward-compat: VALID / TAMPERED / EXPIRED / BREACH are the original four
   722→    (agent 1-b). REVIEW is new (Day 1 Track D) — UPI Circle 24h cooling period
   723→    requires human approval but is not a hard REJECT (the txn is allowed in
   724→    principle, just gated). REVIEW routes to the case queue like other REVIEW
   725→    decisions; the difference is the audit carries ``verdict_reason=
   726→    "cooling_period_active"`` for compliance traceability.
   727→    """
   728→
   729→    VALID = "valid"
   730→    TAMPERED = "tampered"
   731→    EXPIRED = "expired"
   732→    BREACH = "breach"
   733→    REVIEW = "review"
   734→
   735→
   736→def verify_mandate(
   737→    token: str | None,
   738→    amount_inr: float,
   739→    *,
   740→    device_id: str | None = None,
   741→    user_id: str | None = None,
   742→) -> tuple[str, dict]:
   743→    """Verify a mandate token + the per-txn request context.
   744→
   745→    Returns ``(verdict, payload)`` where ``payload`` is the decoded mandate
   746→    body (with the new ``verdict_reason`` key explaining non-VALID outcomes)
   747→    or a minimal ``{"verdict_reason": ...}`` dict for TAMPERED tokens.
   748→
   749→    Backward-compat: the original 2-arg signature ``verify_mandate(token,
   750→    amount_inr)`` still works for ``cod_order`` mandates — ``device_id`` and
   751→    ``user_id`` default to ``None`` and are simply not consulted.
   752→
   753→    Enforcement precedence (consumer = ``src/api/routes.py``):
   754→      * ``TAMPERED`` with header present → REJECT (mandate_invalid)
   755→      * ``EXPIRED`` with header present   → REJECT (mandate_invalid) — covers
   756→        both TTL expiry and OC-201B 6-month inactivity auto-revoke
   757→      * ``BREACH``                          → REJECT (mandate_breach) — covers
   758→        per-txn cap, monthly cap, device_id_not_allowed, user_id_mismatch
   759→      * ``REVIEW``                          → REVIEW (mandate_review_required)
   760→        — currently only the OC-201B 24h cooling-period gate
   761→      * ``VALID``                           → fall through to the cost-optimizer
   762→        decision path (Track C's optimal_decision / Bahnsen BMR)
   763→    """
   764→    if not token:
   765→        return MandateVerdict.TAMPERED, {"verdict_reason": "missing_mandate"}
   766→    try:
   767→        body, sig = token.rsplit(".", 1)
   768→        expected = hmac.new(_secret(), body.encode(), hashlib.sha256).hexdigest()
   769→        if not hmac.compare_digest(sig, expected):
   770→            return MandateVerdict.TAMPERED, {"verdict_reason": "hmac_signature_mismatch"}
   771→        pad = "=" * (-len(body) % 4)
   772→        payload = json.loads(urlsafe_b64decode(body + pad))
   773→
   774→        if payload.get("exp", 0) < time.time():
   775→            return MandateVerdict.EXPIRED, {**payload, "verdict_reason": "expired_ttl"}
   776→
   777→        mtype = payload.get("mandate_type", "cod_order")
   778→        # Use ``sub`` (the salted customer_ref digest) as the per-mandate key
   779→        # for cumulative counters. Stable across verifies for the same mandate.
   780→        mid = payload.get("sub", "")
   781→        now = time.time()
   782→
   783→        if mtype == "upi_circle_delegation":
   784→            # --- Track P (Task 11-a) — read persisted counters from DB    ---
   785→            # --- when available; fall back to in-memory dicts in file mode.---
   786→            # C8/C9/C10 fix (Subagent 14-b): the prior code split the
   787→            # read (``_read_db_counters``) + write (``_write_db_counters``)
   788→            # across two transactions — concurrent verifies could both read
   789→            # below the cap + both decrement, blowing the ₹15k ceiling.
   790→            # The new ``_begin_db_counter_txn`` opens ONE transaction,
   791→            # acquires ``SELECT ... FOR UPDATE`` on the per-mandate row
   792→            # (serialising concurrent verifies), does the C9 month-boundary
   793→            # reset (if stored ``month_key != current YYYY-MM``), reads the
   794→            # 24h cooling events, and returns a ``_DbCounterTxn`` handle
   795→            # holding the open txn. The cap-checks run with the lock held;
   796→            # on VALID we call ``commit_increment`` (UPDATE counter +
   797→            # INSERT event + C10 90-day prune DELETE + COMMIT, all in the
   798→            # same txn); on BREACH/REVIEW/EXPIRED we call ``rollback`` to
   799→            # release the lock without advancing the counter.
   800→            #
   801→            # ``_begin_db_counter_txn`` returns None when no Postgres
   802→            # connection is configured OR when the query fails — the
   803→            # sentinel ``last_activity_ts = -1.0`` from the legacy code is
   804→            # replaced by the None return; the caller falls back to the
   805→            # in-memory dicts (file mode / test path).
   806→            current_month_key = _current_month_key(now)
   807→            db_txn = _begin_db_counter_txn(mid, now, current_month_key)
   808→            if db_txn is not None:
   809→                # DB is the source of truth in prod. The C9 reset has
   810→                # already fired inside ``_begin_db_counter_txn`` if needed
   811→                # (the FOR UPDATE lock was held throughout), so
   812→                # ``db_txn.cumulative_monthly`` is the post-reset value.
   813→                cumulative_monthly = db_txn.cumulative_monthly
   814→                # If the row exists but last_activity_ts is NULL (mandate
   815→                # never spent), fall back to the mandate's iat — same
   816→                # baseline as the in-memory path.
   817→                last_act = (
   818→                    db_txn.last_activity_ts
   819→                    if db_txn.last_activity_ts > 0
   820→                    else payload.get("iat", now)
   821→                )
   822→                # ``db_txn.recent_24h`` is already filtered to ts >
   823→                # now-86400 by the DB query inside the txn — no need to
   824→                # re-prune.
   825→                recent = db_txn.recent_24h
   826→                # Mirror to in-memory dicts so ops introspection + the
   827→                # file-mode fallback path see the same values.
   828→                _cumulative_monthly[mid] = cumulative_monthly
   829→                _cumulative_24h[mid] = list(recent)
   830→                _last_activity[mid] = last_act
   831→            else:
   832→                cumulative_monthly = _cumulative_monthly.get(mid, 0.0)
   833→                last_act = _last_activity.get(mid, payload.get("iat", now))
   834→                recent = _cumulative_24h.get(mid, [])
   835→                # Prune txns older than 24h (rolling window).
   836→                recent = [(ts, amt) for ts, amt in recent if now - ts < 86400]
   837→                _cumulative_24h[mid] = recent
   838→
   839→            # 1. OC-201B §3.8: 6-month inactivity auto-revoke. Baseline last
   840→            #    activity is the mandate's iat (so a freshly-issued mandate
   841→            #    passes); once a txn lands, _last_activity is updated.
   842→            inactivity_days = int(payload.get("inactivity_revoke_days", 180))
   843→            if now - last_act > inactivity_days * 86400:
   844→                # Release the FOR UPDATE lock on the non-VALID path — a
   845→                # rejected txn does not consume the monthly cap.
   846→                if db_txn is not None:
   847→                    db_txn.rollback()
   848→                return MandateVerdict.EXPIRED, {
   849→                    **payload,
   850→                    "verdict_reason": "inactivity_auto_revoke",
   851→                }
   852→
   853→            # 2. OC-201B: per-txn cap (default ₹5,000).
   854→            max_per_txn = float(payload.get("max_per_txn_inr", 5000.0))
   855→            if float(amount_inr) > max_per_txn:
   856→                if db_txn is not None:
   857→                    db_txn.rollback()
   858→                return MandateVerdict.BREACH, {
   859→                    **payload,
   860→                    "verdict_reason": "per_txn_cap_exceeded",
   861→                }
   862→
   863→            # 3. OC-201B: monthly cumulative cap (default ₹15,000).
   864→            # C8 fix: the cap-check uses the post-FOR-UPDATE-LOCK
   865→            # cumulative_monthly value (read inside the txn while the
   866→            # row was locked). A concurrent verifier that just landed a
   867→            # ₹15k txn will have committed before our SELECT FOR UPDATE
   868→            # acquired the lock, so we'll see its committed value + trip
   869→            # the cap ourselves — the cap is now race-safe.
   870→            max_per_month = float(payload.get("max_per_month_inr", 15000.0))
   871→            projected = cumulative_monthly + float(amount_inr)
   872→            if projected > max_per_month:
   873→                if db_txn is not None:
   874→                    db_txn.rollback()
   875→                return MandateVerdict.BREACH, {
   876→                    **payload,
   877→                    "verdict_reason": "monthly_cap_exceeded",
   878→                }
   879→
   880→            # 4. OC-201B §3.7 (Issuer Bank duty): validate device_id before
   881→            #    debiting. If the mandate enumerates device_ids and the
   882→            #    request carries an X-Device-Id, it must be in the list.
   883→            allowed_devices = payload.get("device_ids", [])
   884→            if device_id is not None and allowed_devices and device_id not in allowed_devices:
   885→                if db_txn is not None:
   886→                    db_txn.rollback()
   887→                return MandateVerdict.BREACH, {
   888→                    **payload,
   889→                    "verdict_reason": "device_id_not_allowed",
   890→                }
   891→
   892→            # 5. OC-201B §3.3 (Secondary App/PSP duty): user_id captured at
   893→            #    registration and validated per txn.
   894→            expected_uid = payload.get("user_id", "")
   895→            if user_id is not None and expected_uid and user_id != expected_uid:
   896→                if db_txn is not None:
   897→                    db_txn.rollback()
   898→                return MandateVerdict.BREACH, {
   899→                    **payload,
   900→                    "verdict_reason": "user_id_mismatch",
   901→                }
   902→
   903→            # 6. OC-201B: 24h cooling period. If a prior txn in the last 24h
   904→            #    was >= cooling_24h_inr (default ₹5,000), require human
   905→            #    approval — REVIEW, not REJECT. The txn is permitted in
   906→            #    principle; the cooling gate is a fraud-control circuit
   907→            #    breaker, not a hard cap.
   908→            cooling_24h = float(payload.get("cooling_24h_inr", 5000.0))
   909→            for _ts, amt in recent:
   910→                if amt >= cooling_24h:
   911→                    if db_txn is not None:
   912→                        db_txn.rollback()
   913→                    return MandateVerdict.REVIEW, {
   914→                        **payload,
   915→                        "verdict_reason": "cooling_period_active",
   916→                    }
   917→
   918→            # All checks passed — record the txn for cumulative tracking.
   919→            # Update BOTH the in-memory dicts (cheap, file-mode source of
   920→            # truth + shadow for ops introspection) AND the DB counters
   921→            # (cross-restart persistence, Track P). On a VALID verdict only —
   922→            # BREACH/REVIEW/EXPIRED do not advance the counters (a rejected
   923→            # txn does not consume the monthly cap; a REVIEW txn does not
   924→            # re-trigger the cooling window).
   925→            new_cumulative = cumulative_monthly + float(amount_inr)
   926→            _cumulative_monthly[mid] = new_cumulative
   927→            # 15-b — refactor from ``setdefault(mid, []).append(...)`` (which
   928→            # mutated the stored list in place WITHOUT triggering the
   929→            # ``_FileState._persist_to_disk()`` hook — the in-memory state
   930→            # would be advanced but the file would be stale). The new form
   931→            # uses a pure ``__setitem__`` so the persist fires after the
   932→            # in-place append, capturing the new event on disk.
   933→            prior_events = _cumulative_24h.get(mid, [])
   934→            _cumulative_24h[mid] = prior_events + [(now, float(amount_inr))]
   935→            _last_activity[mid] = now
   936→            if db_txn is not None:
   937→                # C8/C9/C10 — the FOR UPDATE lock has been held since
   938→                # ``_begin_db_counter_txn``; this UPDATE + INSERT event +
   939→                # C10 90-day prune DELETE + COMMIT all run in the SAME
   940→                # transaction, so concurrent verifies serialize.
   941→                db_txn.commit_increment(
   942→                    new_cumulative_monthly=new_cumulative,
   943→                    last_activity_ts=now,
   944→                    txn_ts=now,
   945→                    txn_amount=float(amount_inr),
   946→                )
   947→            return MandateVerdict.VALID, {**payload, "verdict_reason": "ok"}
   948→
   949→        # --- cod_order (legacy) path ---
   950→        if float(amount_inr) > float(payload.get("max_amount_inr", 0)):
   951→            return MandateVerdict.BREACH, {**payload, "verdict_reason": "amount_exceeds_max"}
   952→        return MandateVerdict.VALID, {**payload, "verdict_reason": "ok"}
   953→    except Exception:
   954→        return MandateVerdict.TAMPERED, {"verdict_reason": "decode_error"}

   957→def decode_mandate(token: str) -> dict:
   958→    """Decode a mandate token's body without verifying the HMAC.
   959→
   960→    Convenience helper for tests + dashboards that want to *inspect* a
   961→    mandate's payload (e.g. read the ``mandate_type``, ``bh_purpose_code``,
   962→    or ``device_ids``) without going through ``verify_mandate``'s
   963→    enforcement path. **Never** use this for authorization decisions —
   964→    always go through ``verify_mandate`` for those.
   965→    """
   966→    body = token.rsplit(".", 1)[0]
   967→    pad = "=" * (-len(body) % 4)
   968→    return json.loads(urlsafe_b64decode(body + pad))

   971→def reset_upi_counters() -> None:
   972→    """Test helper — wipe the UPI Circle cumulative counters.
   973→
   974→    Wipes BOTH surfaces so test isolation holds regardless of mode:
   975→      * in-memory dicts (file-mode / test path) — always cleared.
   976→      * Postgres ``mandate_counters`` + ``mandate_counter_events`` tables
   977→        (Track P, prod path) — TRUNCATED when ``DATABASE_URL`` is set.
   978→      * cached module-level psycopg connection — closed + cleared so the
   979→        next ``_get_counters_conn()`` call re-reads ``DATABASE_URL`` (handles
   980→        tests that flip the env var between cases).
   981→
   982→    Track E Day 2 will replace this with a Postgres truncate + transactional
   983→    rollback; this implementation already does the truncate so a future Track
   984→    E refactor just removes the in-memory fallback.
   985→    """
   986→    _cumulative_monthly.clear()
   987→    _cumulative_24h.clear()
   988→    _last_activity.clear()
   989→    # 15-b — force-flush the combined cleared state. The 3 individual
   990→    # ``.clear()`` calls each trigger a throttled persist; the throttle
   991→    # would skip the 2nd + 3rd writes (within 5 sec of the first), so
   992→    # the file would have a STALE partial-cleared state (monthly
   993→    # cleared, 24h + last_activity still populated). The force=True
   994→    # bypass ensures the file reflects the post-clear empty state.
   995→    _mandate_state._persist_to_disk(force=True)
   996→    # Close + drop the cached connection so DATABASE_URL changes between
   997→    # tests are picked up. Re-opening is cheap (lazy on next verify call).
   998→    _reset_counters_conn()
   999→    conn = _get_counters_conn()
  1000→    if conn is None:
  1001→        return
  1002→    try:
  1003→        with conn.cursor() as cur:
  1004→            # Events first (no FK to counters, but order is still safer).
  1005→            cur.execute("TRUNCATE TABLE mandate_counter_events")
  1006→            cur.execute("DELETE FROM mandate_counters")
  1007→        conn.commit()
  1008→    except Exception:
  1009→        # The tables may not exist (migration not run yet — e.g. CI without
  1010→        # DATABASE_URL fixture). Silently fall through; the in-memory dicts
  1011→        # are the test's source of truth in that case.
  1012→        try:
  1013→            conn.rollback()
  1014→        except Exception:
  1015→            pass

  1018→def simulate_inactivity(token: str, days: int) -> None:
  1019→    """Test helper — backdate the per-mandate ``_last_activity`` by ``days``.
  1020→
  1021→    Used to drive the OC-201B 6-month inactivity auto-revoke path without
  1022→    having to actually sleep 180 days. Mutates module state for the mandate
  1023→    identified by the token's ``sub``. No-op if the token is invalid.
  1024→
  1025→    Track P — also persists the backdated timestamp to the
  1026→    ``mandate_counters`` row so the auto-revoke check fires in Postgres mode
  1027→    too (the DB row is the source of truth there; in-memory mutation alone
  1028→    would be invisible to the next ``verify_mandate`` call).
  1029→    """
  1030→    try:
  1031→        payload = decode_mandate(token)
  1032→    except Exception:
  1033→        return
  1034→    mid = payload.get("sub", "")
  1035→    if not mid:
  1036→        return
  1037→    backdated_ts = time.time() - (int(days) * 86400)
  1038→    _last_activity[mid] = backdated_ts
  1039→    # Persist to the mandate_counters row so the inactivity check fires in
  1040→    # Postgres mode. UPSERT — the mandate may not have a row yet (fresh
  1041→    # mandate with no prior txn).
  1042→    conn = _get_counters_conn()
  1043→    if conn is None:
  1044→        return
  1045→    try:
  1046→        with conn.cursor() as cur:
  1047→            cur.execute(
  1048→                "INSERT INTO mandate_counters "
  1049→                "(mandate_sub, cumulative_monthly, last_activity_ts, updated_at) "
  1050→                "VALUES (%s, 0, %s, NOW()) "
  1051→                "ON CONFLICT (mandate_sub) DO UPDATE SET "
  1052→                "last_activity_ts = EXCLUDED.last_activity_ts, "
  1053→                "updated_at = NOW()",
  1054→                (mid, backdated_ts),
  1055→            )
  1056→        conn.commit()
  1057→    except Exception:
  1058→        try:
  1059→            conn.rollback()
  1060→        except Exception:
  1061→            pass
