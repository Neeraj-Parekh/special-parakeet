   262→function fakeHash(seed: string): string {
   263→  // Deterministic 64-hex-char hash so the chain looks valid but is
   264→  // obviously synthetic. NOT real SHA-256 — preview-only.
   265→  let h = 0x811c9dc5;
   266→  for (let i = 0; i < seed.length; i++) {
   267→    h ^= seed.charCodeAt(i);
   268→    h = (h * 0x01000193) >>> 0;
   269→  }
   270→  let out = "";
   271→  let n = h;
   272→  while (out.length < 64) {
   273→    n = (n * 0x01000193 + 0xb5297a4d) >>> 0;
   274→    out += n.toString(16).padStart(8, "0").slice(-8);
   275→  }
   276→  return out.slice(0, 64);
   277→}

   521→// Merkle proof mock — single interval sealing all 8 records above.
   522→export interface MerkleProof {
   523→  record_id: number;
   524→  leaf_hash: string;
   525→  interval_id: number;
   526→  position: number;
   527→  proof: Array<{ position: "left" | "right"; hash: string }>;
   528→  merkle_root: string;
   529→  prev_interval_root: string;
   530→  leaf_count: number;
   531→  sealed_at: string | null;
   532→}
   533→
   534→export function mockMerkleProof(recordId: number): MerkleProof | null {
   535→  const idx = SAMPLE_AUDIT_RECORDS.findIndex(
   536→    (r) => r.audit_id === `aud_${(1000 + recordId - 1).toString(36)}${(recordId - 1)
   537→      .toString()
   538→      .padStart(2, "0")}`
   539→  );
   540→  if (idx < 0) return null;
   541→  const record = SAMPLE_AUDIT_RECORDS[idx];
   542→  // RFC 6962-style proof: walk the tree, recording siblings.
   543→  const size = 8;
   544→  const leaves = SAMPLE_AUDIT_RECORDS.slice(0, size).map((r) => r.raw_hash);
   545→  let level = leaves;
   546→  let i = idx;
   547→  const proof: MerkleProof["proof"] = [];
   548→  while (level.length > 1) {
   549→    const siblingIdx = i ^ 1;
   550→    const sibling = level[Math.min(siblingIdx, level.length - 1)];
   551→    proof.push({
   552→      position: siblingIdx > i ? "right" : "left",
   553→      hash: sibling,
   554→    });
   555→    const next: string[] = [];
   556→    for (let k = 0; k < level.length; k += 2) {
   557→      const a = level[k];
   558→      const b = level[Math.min(k + 1, level.length - 1)];
   559→      next.push(fakeHash(a + b));
   560→    }
   561→    level = next;
   562→    i = Math.floor(i / 2);
   563→  }
   564→  return {
   565→    record_id: recordId,
   566→    leaf_hash: record.raw_hash,
   567→    interval_id: 1,
   568→    position: idx,
   569→    proof,
   570→    merkle_root: level[0],
   571→    prev_interval_root: "0".repeat(64),
   572→    leaf_count: size,
   573→    sealed_at: new Date(Date.now() - 1000 * 60 * 30).toISOString(),
   574→  };
   575→}

   606→// ----------------------------------------------------------------------------
   607→// Cost-curve slider (demo moment #6) — Bahnsen Bayes Minimum Risk helpers.
   608→//
   609→// The slider lets a judge dial `C_fn` (false-negative cost) + `probability`
   610→// (model P(RTO)) and watch the cost-optimal decision flip live. These
   611→// helpers mirror `src/business/cost_optimizer.py::optimal_decision` 1:1 so
   612→// the live `CostBreakdownTable` shown in the Verdict card + the chart in
   613→// the cost-curve slider agree at every input combination — no "demo math"
   614→// drift between the slider and the real /risk/score response.
   615→//
   616→//   cost_accept = p · c_fn                                          (ship)
   617→//   cost_review = c_otp + (1 − p)·c_fp + p·(1 − otp_eff)·c_fn       (OTP gate)
   618→//   cost_reject = (1 − p) · c_block                                 (block)
   619→//   decision    = argmin over the three (Bahnsen Eq.5 BMR, ICMLA 2013)
   620→//
   621→// Defaults match the deployed weights (c_fp=50, c_fn=600, c_otp=5,
   622→// c_block=1000, otp_eff=0.82) used by both the Python `optimal_decision`
   623→// and the `mockScore` mock response — see docs/cost_table.md.
   624→// ----------------------------------------------------------------------------
   625→export interface BmrWeights {
   626→  c_fp: number;
   627→  c_fn: number;
   628→  c_otp: number;
   629→  c_block: number;
   630→  otp_effectiveness: number;
   631→}
   632→
   633→export const DEFAULT_BMR_WEIGHTS: BmrWeights = {
   634→  c_fp: 50,
   635→  c_fn: 600,
   636→  c_otp: 5,
   637→  c_block: 1000,
   638→  otp_effectiveness: 0.82,
   639→};

   641→/** Single sample point on the Bahnsen BMR cost-curve surface. */
   642→export interface CostCurveSample {
   643→  probability: number;
   644→  ACCEPT: number;
   645→  REVIEW: number;
   646→  REJECT: number;
   647→  decision: Decision;
   648→}
   649→
   650→/** Bahnsen Eq.(5) — per-decision expected cost at probability `p`. */
   651→export function bmrExpectedCosts(
   652→  p: number,
   653→  weights: Partial<BmrWeights> = {},
   654→): { ACCEPT: number; REVIEW: number; REJECT: number } {
   655→  const w: BmrWeights = { ...DEFAULT_BMR_WEIGHTS, ...weights };
   656→  const cost_accept = p * w.c_fn;
   657→  const cost_review =
   658→    w.c_otp + (1 - p) * w.c_fp + p * (1 - w.otp_effectiveness) * w.c_fn;
   659→  const cost_reject = (1 - p) * w.c_block;
   660→  return {
   661→    ACCEPT: Math.round(cost_accept * 100) / 100,
   662→    REVIEW: Math.round(cost_review * 100) / 100,
   663→    REJECT: Math.round(cost_reject * 100) / 100,
   664→  };
   665→}
   666→
   667→/** Argmin decision ∈ {ACCEPT, REVIEW, REJECT} at probability `p`. */
   668→export function bmrDecisionAt(
   669→  p: number,
   670→  weights: Partial<BmrWeights> = {},
   671→): { decision: Decision; costs: { ACCEPT: number; REVIEW: number; REJECT: number } } {
   672→  const costs = bmrExpectedCosts(p, weights);
   673→  const min = Math.min(costs.ACCEPT, costs.REVIEW, costs.REJECT);
   674→  let decision: Decision = "REVIEW";
   675→  // Tie-break by lowest friction: ACCEPT > REVIEW > REJECT (matches the
   676→  // Python `optimal_intervention` insertion-order tie-break, less-is-more).
   677→  if (costs.ACCEPT === min) decision = "ACCEPT";
   678→  else if (costs.REVIEW === min) decision = "REVIEW";
   679→  else decision = "REJECT";
   680→  return { decision, costs };
   681→}
   682→
   683→/**
   684→ * Sample the cost-curve surface over P(RTO) ∈ [0,1]. Returns N evenly
   685→ * spaced points with the three expected-cost curves so the cost-curve
   686→ * slider can render them with Recharts. Drummond-Holte 2006 §3.6.
   687→ */
   688→export function sampleCostCurve(
   689→  weights: Partial<BmrWeights> = {},
   690→  n = 41,
   691→): CostCurveSample[] {
   692→  const out: CostCurveSample[] = [];
   693→  for (let i = 0; i < n; i++) {
   694→    const p = i / (n - 1);
   695→    const { costs, decision } = bmrDecisionAt(p, weights);
   696→    out.push({
   697→      probability: Math.round(p * 1000) / 1000,
   698→      ACCEPT: costs.ACCEPT,
   699→      REVIEW: costs.REVIEW,
   700→      REJECT: costs.REJECT,
   701→      decision,
   702→    });
   703→  }
   704→  return out;
   705→}
   706→
   707→/**
   708→ * Find the crossover probabilities where the cost-optimal decision flips.
   709→ * Returns the ascending list of P(RTO) at which argmin changes — so the
   710→ * slider can render vertical "flip points" on the chart.
   711→ */
   712→export function findDecisionCrossovers(
   713→  weights: Partial<BmrWeights> = {},
   714→): number[] {
   715→  const curve = sampleCostCurve(weights, 201);
   716→  const crossovers: number[] = [];
   717→  for (let i = 1; i < curve.length; i++) {
   718→    if (curve[i].decision !== curve[i - 1].decision) {
   719→      crossovers.push(curve[i].probability);
   720→    }
   721→  }
   722→  return crossovers;
   723→}

   828→function ruleMatches(order: OrderInput, rule: Rule): boolean {
   829→  const v = (order as unknown as Record<string, unknown>)[rule.field];
   830→  if (v === undefined) return false;
   831→  const target = rule.value;
   832→  switch (rule.op) {
   833→    case "gt":
   834→      return typeof v === "number" && typeof target === "number" && v > target;
   835→    case "lt":
   836→      return typeof v === "number" && typeof target === "number" && v < target;
   837→    case "eq":
   838→      return String(v) === String(target);
   839→    case "in":
   840→      return Array.isArray(target) && target.some((t) => String(t) === String(v));
   841→    default:
   842→      return false;
   843→  }
   844→}

  1037→      verdict_reason: mandate_verdict_reason,
  1038→      mandate_type: mandateType,
  1039→      bh_purpose_code,
  1040→    },
  1041→  );
  1042→}
