    55→// ----------------------------------------------------------------------------
    56→// Recent-decisions table (in-session local state — survives nav?)
    57→// We use a Zustand-style module-level store so the list survives a
    58→// page navigation away and back. Persisted to sessionStorage so a
    59→// refresh doesn't wipe a 30-second judge demo mid-flight.
    60→// ----------------------------------------------------------------------------
    61→
    62→interface RecentDecision {
    63→  prediction_id: string;
    64→  order_id: string;
    65→  amount_inr: number;
    66→  payment_method: string;
    67→  decision: Decision;
    68→  probability: number | null;
    69→  decision_source: string;
    70→  mock: boolean;
    71→  ts: number;
    72→}
    73→
    74→const RECENT_KEY = "rto-recent-decisions";
    75→let recentListeners: Array<() => void> = [];
    76→let recentCache: RecentDecision[] = [];
    77→
    78→function loadRecent(): RecentDecision[] {
    79→  if (typeof window === "undefined") return [];
    80→  try {
    81→    const raw = window.sessionStorage.getItem(RECENT_KEY);
    82→    if (raw) return JSON.parse(raw) as RecentDecision[];
    83→  } catch {
    84→    /* ignore */
    85→  }
    86→  return [];
    87→}
    88→
    89→function saveRecent(v: RecentDecision[]) {
    90→  if (typeof window === "undefined") return;
    91→  try {
    92→    window.sessionStorage.setItem(RECENT_KEY, JSON.stringify(v.slice(0, 50)));
    93→  } catch {
    94→    /* ignore */
    95→  }
    96→  recentCache = v;
    97→  recentListeners.forEach((fn) => fn());
    98→}
    99→
   100→function useRecentDecisions(): [RecentDecision[], (d: RecentDecision) => void, () => void] {
   101→  const [, force] = React.useReducer((x) => x + 1, 0);
   102→  React.useEffect(() => {
   103→    recentCache = loadRecent();
   104→    const fn = () => force();
   105→    recentListeners.push(fn);
   106→    return () => {
   107→      recentListeners = recentListeners.filter((f) => f !== fn);
   108→    };
   109→  }, []);
   110→  const add = React.useCallback((d: RecentDecision) => {
   111→    const next = [d, ...loadRecent()].slice(0, 50);
   112→    saveRecent(next);
   113→  }, []);
   114→  const clear = React.useCallback(() => {
   115→    saveRecent([]);
   116→  }, []);
   117→  return [recentCache, add, clear];
   118→}

   139→export default function RiskConsolePage() {
   140→  const keys = useApiKeys();
   141→  const [order, setOrder] = React.useState<OrderInput>(EMPTY_ORDER);
   142→  const [result, setResult] = React.useState<ScoreResponse | null>(null);
   143→  const [loading, setLoading] = React.useState(false);
   144→  const [error, setError] = React.useState<string | null>(null);
   145→  const [mock, setMock] = React.useState(false);
   146→  const [recent, addRecent, clearRecent] = useRecentDecisions();
   147→
   148→  // Fetch live rules on mount so the score button's "BLOCK rule fired"
   149→  // branch reflects the user's current rule set (Track I demo moment #4).
   150→  const rulesQuery = useQuery({
   151→    queryKey: ["rules"],
   152→    queryFn: async () => {
   153→      const r = await fetch("/api/v1/rules", {
   154→        headers: buildAuthHeader(keys, "scorer"),
   155→      });
   156→      const data = await r.json().catch(() => ({}));
   157→      return { rules: data.rules || [], mock: r.headers.get("X-Mock-Mode") === "true" };
   158→    },
   159→    staleTime: 30_000,
   160→  });
   161→
   162→  async function score() {
   163→    setLoading(true);
   164→    setError(null);
   165→    setResult(null);
   166→    try {
   167→      const idemKey = `${order.order_id}:${order.amount_inr}:${Date.now()}`;
   168→      const r = await fetch("/api/risk/score", {
   169→        method: "POST",
   170→        headers: {
   171→          "Content-Type": "application/json",
   172→          ...buildAuthHeader(keys, "scorer"),
   173→          "Idempotency-Key": idemKey,
   174→        },
   175→        body: JSON.stringify(order),
   176→      });
   177→      const data = (await r.json().catch(() => null)) as ScoreResponse | null;
   178→      if (!r.ok || !data) {
   179→        const detail = (data as unknown as { detail?: string })?.detail || `HTTP ${r.status}`;
   180→        setError(detail);
   181→      } else {
   182→        setResult(data);
   183→        setMock(r.headers.get("X-Mock-Mode") === "true");
   184→        addRecent({
   185→          prediction_id: data.prediction_id || "—",
   186→          order_id: order.order_id,
   187→          amount_inr: order.amount_inr,
   188→          payment_method: order.payment_method,
   189→          decision: (data.decision || "REVIEW") as Decision,
   190→          probability: data.probability,
   191→          decision_source: data.decision_source,
   192→          mock: r.headers.get("X-Mock-Mode") === "true",
   193→          ts: Date.now(),
   194→        });
   195→      }
   196→    } catch (e) {
   197→      setError(String(e));
   198→    } finally {
   199→      setLoading(false);
   200→    }
   201→  }
   202→

   204→    setOrder({ ...d.order, order_id: d.order.order_id });
   205→    setResult(null);
   206→    setError(null);
   207→  }
   208→
   209→  return (
   210→    <div className="space-y-6">
   211→      <PageHeader />
   212→      <div className="grid gap-6 lg:grid-cols-[420px_minmax(0,1fr)]">
   213→        <div className="space-y-4">
   214→          <OrderFormCard
   215→            order={order}

   227→            mock={mock}
   228→          />

   231→            probability={result?.probability ?? null}
   232→            mock={mock}
   233→          />
   234→          <RulesToggleCard
   235→            order={order}
   236→            lastDecision={result?.decision ?? null}
   237→            lastProbability={result?.probability ?? null}
   238→            onRescore={score}
   239→          />

   242→            rulesCount={rulesQuery.data?.rules.length ?? 0}
   243→          />
   244→          <RecentDecisionsCard
   245→            rows={recent}

   251→      <span className="sr-only" aria-hidden>

   621→function summarizeOrder(result: ScoreResponse): string {
   622→  // Compose the "why" summary line the judge reads aloud — mirrors
   623→  // demo moment #2: "73% risk because: COD + ₹12,400, new customer,
   624→  // vague address in Tier-3 city".
   625→  const flags: string[] = [];
   626→  // Read off the reasons array's most-upvoted features + flag the
   627→  // classic Indian e-commerce RTO drivers. We use the actual value
   628→  // (not the feature name) so the summary reflects the order, not a
   629→  // hardcoded "COD" label.
   630→  const reasons = result.explanation || [];
   631→  const reasonValue = (f: string): string | undefined =>
   632→    reasons.find((r) => r.feature === f)?.value as string | undefined;
   633→  const payment = reasonValue("payment_method");
   634→  if (payment === "COD") flags.push("COD");
   635→  else if (payment === "Prepaid") flags.push("Prepaid payment");
   636→  const addr = reasonValue("address_quality");
   637→  if (addr === "vague") flags.push("vague address");
   638→  else if (addr === "partial") flags.push("partial address");
   639→  const tier = reasonValue("city_tier");
   640→  if (tier === "tier_3") flags.push("Tier-3 city");
   641→  else if (tier === "tier_2") flags.push("Tier-2 city");
   642→  const priorOrders = reasonValue("prior_orders");
   643→  if (priorOrders === 0) flags.push("new customer");
   644→  else if (Number(priorOrders) >= 10) flags.push("loyal customer");
   645→  if (reasonValue("prior_returns") !== undefined && Number(reasonValue("prior_returns")) > 0) {
   646→    flags.push("repeat returner");
   647→  }
   648→  if (reasonValue("amount_inr") !== undefined && Number(reasonValue("amount_inr")) > 10000) {
   649→    flags.push("high-value order");
   650→  }
   651→  return flags.length ? flags.join(" + ") : "no dominant risk factor";
   652→}
