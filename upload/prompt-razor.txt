```python
# --- Rules Engine ---
class RulesEngine:
    """Deterministic rules evaluated before ML. 1-5ms."""
    
    def evaluate(self, order: dict, db: Session) -> Optional[str]:
        rules = db.query(Rule).filter(Rule.is_active == "true").order_by(Rule.priority).all()
        for rule in rules:
            if self._match(rule.condition, order):
                logger.info("rule_fired", rule_id=rule.rule_id, action=rule.action)
                return rule.action
        return None
    
    def _match(self, condition: str, order: dict) -> bool:
        # Simplified: parse JSON logic
        cond = json.loads(condition)
        field = cond.get("field")
        op = cond.get("operator")
        value = cond.get("value")
        actual = order.get(field)
        
        if op == "gt": return actual > value
        if op == "eq": return actual == value
        if op == "in": return actual in value
        return False

rules_engine = RulesEngine()

# --- Circuit Breaker ---
class CircuitBreaker:
    """Fail open if ML is down."""
    def __init__(self, failure_threshold=5, recovery_timeout=30):
        self.failures = 0
        self.state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN
        self.threshold = failure_threshold
        self.timeout = recovery_timeout
        self.last_failure = None
    
    def call(self, func, *args, **kwargs):
        if self.state == "OPEN":
            if (datetime.utcnow() - self.last_failure).seconds > self.timeout:
                self.state = "HALF_OPEN"
            else:
                return None  # Fail open
        
        try:
            result = func(*args, **kwargs)
            if self.state == "HALF_OPEN":
                self.state = "CLOSED"
                self.failures = 0
            return result
        except Exception as e:
            self.failures += 1
            self.last_failure = datetime.utcnow()
            if self.failures >= self.threshold:
                self.state = "OPEN"
                logger.error("circuit_breaker_opened")
            raise

def compute_hash(record: dict, previous_hash: str) -> str:
    record_str = json.dumps(record, sort_keys=True) + previous_hash
    return hashlib.sha256(record_str.encode()).hexdigest()

def write_audit(record: dict, db: Session):

    record["raw_hash"] = compute_hash(record, prev_hash)
    
    db_record = AuditRecord(**record)
    db.add(db_record)
```

```python
# src/agents/bounded_agent.py
from typing import List, Literal

class BoundedAgent:
    """
    Agent that can ONLY perform pre-approved actions.
    No LLM decides money movements. Ever.
    """
    
    ALLOWED_ACTIONS = {
        "score_order": {"params": ["order_id"], "cost": 0},
        "request_otp": {"params": ["customer_id"], "cost": 1},
        "flag_review": {"params": ["order_id", "reason"], "cost": 2},
        "block_order": {"params": ["order_id", "reason"], "cost": 10, "requires_approval": True}
    }
    
    def __init__(self, risk_api_client, approval_queue):
        self.client = risk_api_client
        self.approval = approval_queue
    
    def act(self, intent: str, params: dict) -> dict:
        if intent not in self.ALLOWED_ACTIONS:
            return {"error": "Action not in allowlist", "blocked": True}
        
        action = self.ALLOWED_ACTIONS[intent]
        
        # High-cost actions require human approval
        if action.get("requires_approval"):
            self.approval.submit(intent, params)
            return {"status": "pending_approval", "ticket_id": str(uuid4())}
        
        # Execute bounded action
        if intent == "score_order":
            return self.client.score(params["order_id"])
        elif intent == "request_otp":
            return self.client.send_otp(params["customer_id"])
        
        return {"error": "Unknown execution path"}
```

3. Call Rules Engine (fast path, <5ms)
4. If rule fires → return immediately with rule trace
5. If no rule → fetch online features from Feature Store
6. Run model inference (HistGB, cached model object)
7. Generate SHAP-like explanation (one-at-a-time perturbation)
8. Log to Audit Service (async, via message queue)
9. Return structured response

**Dependencies:** Feature Store, Model Registry, Rules Engine, Audit Service, Message Queue

**Database:** None (stateless, reads model from shared volume/registry)

### 3.2 Rules Engine Service
**Responsibility:** Deterministic, sub-millisecond rule evaluation. Written in Go or Rust for speed. Python fallback acceptable for hackathon.

**Data Model:**
```yaml
rules:
  - rule_id: "RULE-001"
    name: "High Value COD New Customer"
    condition:
      operator: "AND"
      clauses:
        - { field: "is_cod", op: "eq", value: true }
        - { field: "amount", op: "gt", value: 50000 }
        - { field: "prior_orders", op: "eq", value: 0 }

    priority: 1
    is_active: true
    created_by: "admin"
    merchant_scope: "global"  # or specific merchant_id
```

**API Endpoints:**
- `POST /v1/rules/evaluate` — evaluate single order
- `GET /v1/rules` — list active rules
- `POST /v1/rules` — create new rule (admin only)
- `PUT /v1/rules/{id}` — update rule
- `DELETE /v1/rules/{id}` — soft delete

**Performance Target:** p99 < 5ms for rule evaluation

### 3.3 Feature Store Service
**Responsibility:** Unified feature serving. Eliminates training-serving skew.

**Architecture Pattern:** Virtual Feature Store (maps logical features to physical storage)

**Components:**
- **Online Store:** Redis (latest values, <5ms retrieval)
- **Offline Store:** PostgreSQL + Parquet files (historical, point-in-time correct)
- **Registry:** PostgreSQL table (feature metadata, lineage, owner)

**Feature Types:**
| Type | Storage | Latency | Example |
|---|---|---|---|
| Static | Online + Offline | <1ms | `is_cod`, `product_category` |
| On-Demand | Computed at request | <10ms | `address_length`, `hour_of_day` |
| Streaming | Online (Redis) + Offline (Parquet) | <5ms | `rolling_7d_return_rate` |
| Batch | Offline only (training) | N/A | `customer_lifetime_value` |

**API Endpoints:**
- `GET /v1/features/{entity_type}/{entity_id}` — fetch online features

- `GET /v1/features/registry` — list all registered features

**Critical:** Every feature has `feature_name`, `data_type`, `owner`, `freshness_sla`, `online_store`, `offline_store`, `description`.

### 3.5 Audit Service
**Responsibility:** Tamper-evident, append-only audit trail. Court-admissible.

**Data Model:**
```sql
audit_records:
  - prediction_id (PK, UUID v4)
  - order_id
  - merchant_id
  - request_payload (JSONB)
  - feature_snapshot (JSONB)  # exact features used at inference time
  - model_version
  - probability
  - risk_score
  - decision
  - rule_fired (nullable)
  - explanation (JSONB array of SHAP-like values)
  - timestamp (UTC, indexed)
  - previous_hash (SHA-256 of previous record)
  - raw_hash (SHA-256 of this record's content + previous_hash)
  - integrity_verified (boolean, computed on read)
```

**Tamper Evidence:** Each record hashes the previous record's hash. Breaking one record breaks the chain.

**API Endpoints:**

- `GET /v1/audit/export` — export to CSV/Parquet for compliance

**Storage:** PostgreSQL (transactional) + daily Parquet rotation to MinIO/S3 for long-term.

## 4. The Agentic Layer (Phase 2 — Thin Plugin)

**Location:** `Agent Gateway Service` — sits as a consumer of the Risk API, not as a controller.

**Design:** The agent **cannot** call payment APIs, refund APIs, or modify orders. It can only:
1. `risk.score(order_id)` — get risk score
2. `risk.explain(prediction_id)` — get explanation
3. `case.create(order_id, reason)` — create review case
4. `notify.merchant(merchant_id, message)` — send alert
5. `rules.suggest(condition, action)` — suggest new rule (pending admin approval)

**Guardrails:**
- Action allowlist (hardcoded, not LLM-decided)
- Every agent action is logged via Audit Service with `actor: "agent:{agent_id}"`
- High-cost actions (block, refund suggestions) require human approval queue
- Agent has no direct DB access — only API calls
- Agent prompt is deterministic (no open-ended LLM reasoning on money)

**Why this wins:** Other teams show chatbots that "intelligently" refund orders. You show an agent that **physically cannot** perform unbounded actions, with cryptographic proof in the audit trail.

## 6. Security Model

### 6.1 Zero Trust Between Services
- **Gateway → Services:** JWT with short expiry (5 min), signed with RS256
- **Service → Service:** mTLS (mutual TLS) certificates
- **Service → DB:** Connection pooling with dedicated DB user per service
- **Secrets:** HashiCorp Vault or Docker Secrets (for hackathon)

### 6.2 Data Protection
- **At Rest:** PostgreSQL encrypted (LUKS), Redis encrypted (Redis ACL + TLS)
- **In Transit:** TLS 1.3 everywhere
- **PII:** Address fields hashed (SHA-256) in audit logs. Original address only in encrypted DB column.
- **API Keys:** Hashed with bcrypt, never stored plaintext

### 6.3 Compliance Posture
- **RBI PA Guidelines:** Data localization (all data stays in India region)
- **PCI DSS:** Tokenization of sensitive fields, audit trail completeness
- **GDPR-style:** Right to explanation (every decision has SHAP), right to deletion (soft delete + anonymization)

## 9. Cost Model Framework (The Mathematical Business Logic)

Your original doc mentioned cost-optimal threshold but didn't formalize it. Here's the complete framework.

**Variables:**
- `C_FP` = Cost of false positive (reviewing a good order) = ₹50 (ops time)
- `C_FN` = Cost of false negative (missing an RTO) = ₹600 (forward shipping + reverse shipping + repackaging + blocked inventory)
- `C_OTP` = Cost of OTP verification = ₹5 (SMS cost + conversion friction)
- `C_BLOCK` = Cost of blocking a good order = ₹1000 (lost lifetime value)

**Decision matrix:**

| Decision | Condition | Expected Cost |
|---|---|---|
| **ACCEPT** | `p < τ_accept` | `p × C_FN` (accept and hope) |
| **REVIEW + OTP** | `τ_accept ≤ p < τ_reject` | `C_OTP + (1-p) × C_FP + p × (1-0.82) × C_FN` | *(OTP reduces fraud by 82%)* |
| **REJECT** | `p ≥ τ_reject` | `(1-p) × C_BLOCK` |

**Optimal thresholds (from [Mathematics 2026 paper]):**
```
τ* = C_FP / (C_FP + C_FN) = 50 / (50 + 600) ≈ 0.077
```

But because you have a **three-way decision** (Accept / Review / Reject), the optimization becomes:

```python
def optimal_decision(p, c_fp=50, c_fn=600, c_otp=5, c_block=1000, otp_effectiveness=0.82):
    cost_accept = p * c_fn
    cost_review = c_otp + (1-p) * c_fp + p * (1-otp_effectiveness) * c_fn
    cost_reject = (1-p) * c_block
    
    return min([
        ("ACCEPT", cost_accept),
        ("REVIEW", cost_review),
        ("REJECT", cost_reject)
    ], key=lambda x: x[1])
```

**Business insight for pitch:**
- At `p = 0.10`: Cost(Accept) = ₹60, Cost(Review) = ₹5 + ₹45 + ₹10.8 = ₹60.8, Cost(Reject) = ₹900 → **ACCEPT**
- At `p = 0.40`: Cost(Accept) = ₹240, Cost(Review) = ₹5 + ₹30 + ₹43.2 = ₹78.2, Cost(Reject) = ₹600 → **REVIEW**
- At `p = 0.80`: Cost(Accept) = ₹480, Cost(Review) = ₹5 + ₹10 + ₹86.4 = ₹101.4, Cost(Reject) = ₹200 → **REJECT**

**This is your competitive moat:** You don't just predict risk. You **optimize merchant profit** by choosing the cheapest intervention for each risk level. No other hackathon team will have this math.
